前置说明

1. 基于 vnpy 官方 CtaStrategy 模板（vnpy 2.x / 3.x 通用，Python）；
2. 先单独写【ADX+ATR分位 市场状态过滤器模块】，这个可以被下面所有策略直接调用；
3. 然后依次：双均线、海龟、布林突破、SAR波段、多因子复合CTA；
4. 全部默认 5分钟K线、日内模式、收盘强制清仓，适配国内CTP；
5. 风控：单笔风险1%、ATR动态止损；仅学习仿真，严禁直接实盘；
6. 统一约定参数：
  ○ ATR周期：14
  ○ ADX周期：14
  ○ ATR分位回看窗口：60根K
  ○ ADX阈值：>25 趋势环境；<20 震荡关停
  ○ ATR低分位阈值：20%，低于则禁止开仓
一、独立市场状态过滤器（MarketStateFilter.py，单独文件，所有策略导入使用）
import numpy as np
from vnpy.trader.constant import Interval
from vnpy.trader.object import BarData

class MarketStateFilter:
    """
    ADX + ATR历史分位 市场状态过滤器
    输出 is_trend_env: True=允许趋势策略开仓，False=关停
    """
    def __init__(
        self,
        adx_length: int = 14,
        atr_length: int = 14,
        lookback_window: int = 60,
        adx_trend_threshold: float = 25,
        adx_flat_threshold: float = 20,
        atr_low_pct: float = 0.2
    ):
        self.adx_length = adx_length
        self.atr_length = atr_length
        self.lookback_window = lookback_window
        self.adx_trend_threshold = adx_trend_threshold
        self.adx_flat_threshold = adx_flat_threshold
        self.atr_low_pct = atr_low_pct

        # K线缓存
        self.highs: list[float] = []
        self.lows: list[float] = []
        self.closes: list[float] = []
        self.atr_list: list[float] = []
        self.adx_list: list[float] = []

        # ADX计算缓存
        self.plus_dm_list: list[float] = []
        self.minus_dm_list: list[float] = []
        self.tr_list: list[float] = []

        self.prev_high: float = 0
        self.prev_low: float = 0
        self.prev_close: float = 0

    def update_bar(self, bar: BarData):
        """每一根K更新数据"""
        h = bar.high_price
        l = bar.low_price
        c = bar.close_price

        self.highs.append(h)
        self.lows.append(l)
        self.closes.append(c)
        # 控制缓存长度
        if len(self.highs) > self.lookback_window + self.adx_length:
            self.highs.pop(0)
            self.lows.pop(0)
            self.closes.pop(0)

        if len(self.closes) < 2:
            self.prev_high = h
            self.prev_low = l
            self.prev_close = c
            return

        # True Range
        tr1 = h - l
        tr2 = abs(h - self.prev_close)
        tr3 = abs(l - self.prev_close)
        tr = max(tr1, tr2, tr3)
        self.tr_list.append(tr)
        if len(self.tr_list) > self.adx_length:
            self.tr_list.pop(0)

        # DM
        up_move = h - self.prev_high
        down_move = self.prev_low - l
        plus_dm = up_move if (up_move > down_move and up_move > 0) else 0
        minus_dm = down_move if (down_move > up_move and down_move > 0) else 0
        self.plus_dm_list.append(plus_dm)
        self.minus_dm_list.append(minus_dm)
        if len(self.plus_dm_list) > self.adx_length:
            self.plus_dm_list.pop(0)
            self.minus_dm_list.pop(0)

        self.prev_high = h
        self.prev_low = l
        self.prev_close = c

    def calculate(self):
        """计算ADX、ATR、ATR分位，返回状态"""
        if len(self.tr_list) < self.adx_length:
            return False, 0, 0, 0

        # ATR
        atr = np.mean(self.tr_list[-self.atr_length:])
        self.atr_list.append(atr)
        if len(self.atr_list) > self.lookback_window:
            self.atr_list.pop(0)

        # +DI -DI
        sum_tr = np.sum(self.tr_list[-self.adx_length:])
        sum_plus_dm = np.sum(self.plus_dm_list[-self.adx_length:])
        sum_minus_dm = np.sum(self.minus_dm_list[-self.adx_length:])
        plus_di = 100 * sum_plus_dm / sum_tr if sum_tr != 0 else 0
        minus_di = 100 * sum_minus_dm / sum_tr if sum_tr != 0 else 0
        dx = 100 * abs(plus_di - minus_di) / (plus_di + minus_di) if (plus_di + minus_di) != 0 else 0
        self.adx_list.append(dx)
        if len(self.adx_list) > self.lookback_window:
            self.adx_list.pop(0)
        adx = np.mean(self.adx_list[-self.adx_length:]) if len(self.adx_list)>=self.adx_length else 0

        # ATR历史分位
        if len(self.atr_list) >= self.lookback_window:
            atr_pct = np.percentile(self.atr_list, self.atr_low_pct * 100)
            atr_percent_rank = (atr - np.min(self.atr_list)) / (np.max(self.atr_list) - np.min(self.atr_list)) if (np.max(self.atr_list)-np.min(self.atr_list))>0 else 0
        else:
            atr_percent_rank = 1.0

        # 开关规则：ADX>25 并且 ATR不在最低20%分位
        is_trend_env = True
        if adx < self.adx_flat_threshold or atr_percent_rank < self.atr_low_pct:
            is_trend_env = False

        return is_trend_env, adx, atr, atr_percent_rank
使用方式：在每个策略的__init__里实例化 self.filter = MarketStateFilter()，on_bar第一行先 self.filter.update_bar(bar)，然后 env_ok, adx, atr, atr_rank = self.filter.calculate()；if not env_ok: return 直接跳过开仓信号。

二、策略1：双均线趋势策略（CtaStrategy）
from vnpy.trader.constant import Direction, Offset, Interval
from vnpy.trader.object import BarData
from vnpy.app.cta_strategy import CtaTemplate, StopOrder
from MarketStateFilter import MarketStateFilter

class Ma20Ma60CtaStrategy(CtaTemplate):
    """双均线 5min日内趋势策略 MA5/MA20，ADX+ATR过滤，收盘清仓"""
    author = "demo"
    fast_window = 5
    slow_window = 20
    atr_stop_multiple = 1.5
    close_before_minute = 5  # 收盘前5分钟强制平仓

    parameters = ["fast_window", "slow_window", "atr_stop_multiple", "close_before_minute"]
    variables = ["ma_fast", "ma_slow", "atr"]

    def __init__(self, cta_engine, strategy_name, vt_symbol, setting):
        super().__init__(cta_engine, strategy_name, vt_symbol, setting)
        self.filter = MarketStateFilter()
        self.bar_list = []
        self.long_stop = 0
        self.short_stop = 0

    def on_bar(self, bar: BarData):
        self.cancel_all()
        self.filter.update_bar(bar)
        env_ok, adx, atr, atr_rank = self.filter.calculate()
        if not env_ok:
            return

        self.bar_list.append(bar.close_price)
        if len(self.bar_list) < self.slow_window:
            return
        if len(self.bar_list) > self.slow_window:
            self.bar_list.pop(0)

        self.ma_fast = np.mean(self.bar_list[-self.fast_window:])
        self.ma_slow = np.mean(self.bar_list[-self.slow_window:])

        # 收盘清仓判断
        if bar.datetime.hour == 14 and bar.datetime.minute >= (60 - self.close_before_minute):
            if self.pos > 0:
                self.sell(self.price, self.pos, stop=False)
            elif self.pos < 0:
                self.cover(self.price, abs(self.pos), stop=False)
            return

        # 多头：快上穿慢，价格在慢均线上方
        if self.ma_fast > self.ma_slow and bar.close_price > self.ma_slow:
            if self.pos == 0:
                self.buy(bar.close_price, 1)
                self.long_stop = bar.close_price - atr * self.atr_stop_multiple
            elif self.pos > 0:
                if bar.low_price < self.long_stop:
                    self.sell(self.long_stop, self.pos, stop=True)
        # 空头
        elif self.ma_fast < self.ma_slow and bar.close_price < self.ma_slow:
            if self.pos == 0:
                self.short(bar.close_price, 1)
                self.short_stop = bar.close_price + atr * self.atr_stop_multiple
            elif self.pos < 0:
                if bar.high_price > self.short_stop:
                    self.cover(self.short_stop, abs(self.pos), stop=True)
        self.put_event()

    def on_stop_order(self, stop_order: StopOrder):
        pass
三、策略2：海龟交易简化版（5min日内，禁止加仓，收盘清仓）
from vnpy.trader.constant import Direction, Offset, Interval

import numpy as np

class TurtleIntraStrategy(CtaTemplate):
    """海龟简化版 5min：20突破入场，10突破出场，ATR止损，不加仓"""
    author = "demo"
    entry_window = 20
    exit_window = 10
    atr_stop_multiple = 2.0
    close_before_minute = 5

    parameters = ["entry_window", "exit_window", "atr_stop_multiple", "close_before_minute"]
    variables = ["high_entry", "low_entry", "high_exit", "low_exit"]

    def __init__(self, cta_engine, strategy_name, vt_symbol, setting):
        super().__init__(cta_engine, strategy_name, vt_symbol, setting)
        self.filter = MarketStateFilter()
        self.high_buf = []
        self.low_buf = []

    def on_bar(self, bar: BarData):
        self.cancel_all()
        self.filter.update_bar(bar)
        env_ok, adx, atr, atr_rank = self.filter.calculate()
        if not env_ok:
            return

        self.high_buf.append(bar.high_price)
        self.low_buf.append(bar.low_price)
        if len(self.high_buf) > self.entry_window:
            self.high_buf.pop(0)
            self.low_buf.pop(0)
        if len(self.high_buf) < self.entry_window:
            return

        self.high_entry = np.max(self.high_buf[-self.entry_window:])
        self.low_entry = np.min(self.low_buf[-self.entry_window:])
        self.high_exit = np.max(self.high_buf[-self.exit_window:])
        self.low_exit = np.min(self.low_buf[-self.exit_window:])

        # 收盘清仓
        if bar.datetime.hour ==14 and bar.datetime.minute >= (60 - self.close_before_minute):
            if self.pos>0:
                self.sell(bar.close_price, self.pos)
            elif self.pos<0:
                self.cover(bar.close_price, abs(self.pos))
            return

        # 多头突破
        if bar.close_price > self.high_entry and self.pos == 0:
            self.buy(bar.close_price, 1)
            self.long_stop = bar.close_price - atr * self.atr_stop_multiple
        # 多头离场：跌破10周期低点
        elif self.pos>0:
            if bar.close_price < self.low_exit or bar.low_price < self.long_stop:
                self.sell(bar.close_price, self.pos)
        # 空头突破
        if bar.close_price < self.low_entry and self.pos ==0:
            self.short(bar.close_price,1)
            self.short_stop = bar.close_price + atr * self.atr_stop_multiple
        elif self.pos <0:
            if bar.close_price > self.high_exit or bar.high_price> self.short_stop:
                self.cover(bar.close_price, abs(self.pos))
        self.put_event()
    def on_stop_order(self, stop_order: StopOrder):
        pass
四、策略3：布林带趋势突破策略（顺中轨方向突破）

class BollingerTrendBreakStrategy(CtaTemplate):
    """布林趋势突破：只顺着中轨方向突破上/下轨"""
    author = "demo"
    boll_window = 20
    boll_dev = 2.0
    atr_stop_multiple = 1.5
    close_before_minute = 5

    parameters = ["boll_window", "boll_dev", "atr_stop_multiple","close_before_minute"]
    variables = ["mid", "up", "down"]

    def __init__(self, cta_engine, strategy_name, vt_symbol, setting):
        super().__init__(cta_engine, strategy_name, vt_symbol, setting)
        self.filter = MarketStateFilter()
        self.close_buf = []

    def on_bar(self, bar: BarData):
        self.cancel_all()
        self.filter.update_bar(bar)
        env_ok, adx, atr, atr_rank = self.filter.calculate()
        if not env_ok:
            return

        self.close_buf.append(bar.close_price)
        if len(self.close_buf) > self.boll_window:
            self.close_buf.pop(0)
        if len(self.close_buf) < self.boll_window:
            return
        arr = np.array(self.close_buf)
        self.mid = arr.mean()
        std = arr.std()
        self.up = self.mid + self.boll_dev * std
        self.down = self.mid - self.boll_dev * std

        # 收盘清仓
        if bar.datetime.hour ==14 and bar.datetime.minute >= (60 - self.close_before_minute):
            if self.pos>0:
                self.sell(bar.close_price, self.pos)
            elif self.pos<0:
                self.cover(bar.close_price, abs(self.pos))
            return

        # 中轨向上，只做多，突破上轨开多
        if bar.close_price > self.mid:
            if bar.close_price>self.up and self.pos ==0:
                self.buy(bar.close_price, 1)
                self.long_stop = self.mid
            elif self.pos>0 and bar.close_price < self.mid:
                self.sell(bar.close_price, self.pos)
        # 中轨向下，只做空，突破下轨开空
        elif bar.close_price < self.mid:
            if bar.close_price < self.down and self.pos ==0:
                self.short(bar.close_price, 1)
                self.short_stop = self.mid
            elif self.pos <0 and bar.close_price> self.mid:
                self.cover(bar.close_price, abs(self.pos))
        self.put_event()
    def on_stop_order(self, stop_order: StopOrder):
        pass
五、策略4：SAR波段趋势策略（叠加MA20方向过滤）
from vnpy.trader.constant import Direction, Offset, Interval
from vnpy.trader.object import BarData
from vnpy.app.cta_strategy import CtaTemplate, StopOrder
from MarketStateFilter import MarketStateFilter
import numpy as np

class SarTrendStrategy(CtaTemplate):
    """SAR趋势策略 + MA20方向过滤，ADX+ATR环境开关"""
    author = "demo"
    sar_af_start = 0.02
    sar_af_inc = 0.02
    sar_af_max = 0.2
    ma_filter_window = 20
    close_before_minute = 5

    parameters = ["sar_af_start","sar_af_inc","sar_af_max","ma_filter_window","close_before_minute"]
    variables = ["sar_val","ep","af"]

    def __init__(self, cta_engine, strategy_name, vt_symbol, setting):
        super().__init__(cta_engine, strategy_name, vt_symbol, setting)
        self.filter = MarketStateFilter()
        self.close_buf = []
        self.sar = 0.0
        self.ep = 0.0
        self.af = self.sar_af_start
        self.is_long = False
        self.init_flag = False

    def _calc_sar(self, bar: BarData):
        if not self.init_flag:
            self.sar = bar.low_price
            self.ep = bar.high_price
            self.af = self.sar_af_start
            self.is_long = True
            self.init_flag = True
            return self.sar
        if self.is_long:
            self.sar = self.sar + self.af * (self.ep - self.sar)
            if bar.high_price > self.ep:
                self.ep = bar.high_price
                self.af = min(self.af + self.sar_af_inc, self.sar_af_max)
            if bar.low_price < self.sar:
                self.is_long = False
                self.sar = self.ep
                self.ep = bar.low_price
                self.af = self.sar_af_start
        else:
            self.sar = self.sar + self.af * (self.ep - self.sar)
            if bar.low_price < self.ep:
                self.ep = bar.low_price
                self.af = min(self.af + self.sar_af_inc, self.sar_af_max)
            if bar.high_price > self.sar:
                self.is_long = True
                self.sar = self.ep
                self.ep = bar.high_price
                self.af = self.sar_af_start
        return self.sar

    def on_bar(self, bar: BarData):
        self.cancel_all()
        self.filter.update_bar(bar)
        env_ok, adx, atr, atr_rank = self.filter.calculate()
        if not env_ok:
            return
        self.close_buf.append(bar.close_price)
        if len(self.close_buf) > self.ma_filter_window:
            self.close_buf.pop(0)
        if len(self.close_buf) < self.ma_filter_window:
            return
        ma20 = np.mean(self.close_buf)
        sar_val = self._calc_sar(bar)

        # 收盘清仓
        if bar.datetime.hour ==14 and bar.datetime.minute >= (60 - self.close_before_minute):
            if self.pos>0:
                self.sell(bar.close_price, self.pos)
            elif self.pos<0:
                self.cover(bar.close_price, abs(self.pos))
            return

        # SAR翻多，并且价格在MA20之上，做多
        if self.is_long and bar.close_price>ma20 and self.pos ==0:
            self.buy(bar.close_price, 1)
        # SAR翻空，并且价格在MA20之下，做空
        elif not self.is_long and bar.close_price < ma20 and self.pos ==0:
            self.short(bar.close_price, 1)
        # SAR反向触碰，平仓
        if self.pos>0 and not self.is_long:
            self.sell(bar.close_price, self.pos)
        if self.pos<0 and self.is_long:
            self.cover(bar.close_price, abs(self.pos))
        self.put_event()
    def on_stop_order(self, stop_order: StopOrder):
        pass
六、策略5：多因子复合CTA策略（双均线+海龟突破+布林+SAR打分）
from vnpy.trader.constant import Direction, Offset, Interval
from vnpy.trader.object import BarData
from vnpy.app.cta_strategy import CtaTemplate, StopOrder
from MarketStateFilter import MarketStateFilter
import numpy as np

class MultiFactorCtaStrategy(CtaTemplate):
    """多因子打分CTA：均线、唐奇安突破、布林、SAR 加权打分"""
    author = "demo"
    weight_ma = 0.3
    weight_turtle = 0.25
    weight_boll = 0.25
    weight_sar = 0.2
    score_thresh = 0.6
    close_before_minute = 5

    parameters = ["weight_ma","weight_turtle","weight_boll","weight_sar","score_thresh","close_before_minute"]
    variables = ["total_score"]

    def __init__(self, cta_engine, strategy_name, vt_symbol, setting):
        super().__init__(cta_engine, strategy_name, vt_symbol, setting)
        self.filter = MarketStateFilter()
        self.c_buf = []
        self.h_buf = []
        self.l_buf = []
        # sar state
        self.sar = 0
        self.ep =0
        self.af=0.02
        self.sar_long = True
        self.sar_init = False

    def _sar_score(self, bar):
        if not self.sar_init:
            self.sar = bar.low_price
            self.ep = bar.high_price
            self.sar_long = True
            self.sar_init = True
            return 0
        if self.sar_long:
            self.sar = self.sar + self.af*(self.ep - self.sar)
            if bar.high_price>self.ep:
                self.ep = bar.high_price
                self.af = min(self.af+0.02,0.2)
            if bar.low_price < self.sar:
                self.sar_long = False
                self.sar = self.ep
                self.ep = bar.low_price
                self.af =0.02
        else:
            self.sar = self.sar + self.af*(self.ep - self.sar)
            if bar.low_price < self.ep:
                self.ep = bar.low_price
                self.af = min(self.af+0.02,0.2)
            if bar.high_price> self.sar:
                self.sar_long = True
                self.sar = self.ep
                self.ep = bar.high_price
                self.af=0.02
        return 1 if self.sar_long else -1

    def on_bar(self, bar: BarData):
        self.cancel_all()
        self.filter.update_bar(bar)
        env_ok, adx, atr, atr_rank = self.filter.calculate()
        if not env_ok:
            return

        self.c_buf.append(bar.close_price)
        self.h_buf.append(bar.high_price)
        self.l_buf.append(bar.low_price)
        if len(self.c_buf)>20:
            self.c_buf.pop(0)
            self.h_buf.pop(0)
            self.l_buf.pop(0)
        if len(self.c_buf)<20:
            return
        c_arr = np.array(self.c_buf)
        ma5 = np.mean(c_arr[-5:])
        ma20 = np.mean(c_arr[-20:])
        # ma score
        s_ma = 1 if ma5>ma20 and bar.close_price>ma20 else (-1 if ma5<ma20 and bar.close_price<ma20 else 0)
        # turtle
        h20 = np.max(self.h_buf)
        l20 = np.min(self.l_buf)
        s_turtle = 1 if bar.close_price>h20 else (-1 if bar.close_price<l20 else 0)
        # boll
        mid = c_arr.mean()
        std = c_arr.std()
        up = mid + 2*std
        dn = mid -2*std
        s_boll = 1 if bar.close_price>up and bar.close_price>mid else (-1 if bar.close_price<dn and bar.close_price<mid else 0)
        # sar
        s_sar = self._sar_score(bar)

        total = s_ma * self.weight_ma + s_turtle*self.weight_turtle + s_boll*self.weight_boll + s_sar*self.weight_sar
        self.total_score = total

        # 收盘清仓
        if bar.datetime.hour ==14 and bar.datetime.minute >= (60 - self.close_before_minute):
            if self.pos>0:
                self.sell(bar.close_price, self.pos)
            elif self.pos<0:
                self.cover(bar.close_price, abs(self.pos))
            return

        if self.pos ==0:
            if total >= self.score_thresh:
                self.buy(bar.close_price,1)
            elif total <= -self.score_thresh:
                self.short(bar.close_price,1)
        elif self.pos>0 and total <= -self.score_thresh:
            self.sell(bar.close_price, self.pos)
        elif self.pos<0 and total >= self.score_thresh:
            self.cover(bar.close_price, abs(self.pos))
        self.put_event()
    def on_stop_order(self, stop_order: StopOrder):
