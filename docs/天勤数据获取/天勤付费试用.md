### 一句话总结
**历史Tick下载（DataDownloader / get_tick_data_series）是付费专业版功能，免费版不能回溯下载历史Tick**
- ✅ 免费版：只能盘中实时订阅Tick（就是前面给你的挂机录制脚本），只能往后攒，**不能指定起止日期拉过去的Tick**；免费版可以免费下载长周期日线、1分钟K线（对你海龟/DualThrust够用）
- ✅ 专业版：才开放历史Tick下载，国内商品期货Tick从**2016年起**，只有快照一档盘口，不含逐笔成交/逐笔委托
- 价格（官网公开）：月付原价1988元/月，年付原价14888元/年，经常有优惠；**可以申请15天免费试用专业版**，试用期间可以下载历史Tick，用来测试批量下载脚本
- ❗ 没有单独按流量/按品种打包买Tick数据的选项，是整套专业版权限打包卖

### 容易混淆的两点
1. 免费版 `get_tick_serial()`：是实时行情流，最多拿到一小段最近Tick缓存，不是历史下载接口，不能用来批量补历史
2. 专业版下载的Tick：**只能下真实合约，不支持主连**，需要自己写脚本换月拼接（就是前面那份批量拼接脚本）

```python
"""
天勤专业版：批量下载【燃油、纯碱、玉米】历史主力Tick
自动查询历史主力合约、分段下载、最后拼接连续主力tick
"""
import os
import pandas as pd
from datetime import datetime, timedelta
from dotenv import load_dotenv
from tqsdk import TqApi, TqAuth
from tqsdk.tools import DataDownloader
from contextlib import closing

load_dotenv()
USER = os.getenv("TQ_USERNAME")
PWD = os.getenv("TQ_PASSWORD")

# ======================【这里是你修改其他品种的地方！】======================
# 格式："KQ.m@交易所.品种代码"
# 燃油：KQ.m@SHFE.fu
# 纯碱：KQ.m@CZCE.SA
# 玉米：KQ.m@DCE.c
# 想加别的品种，直接在列表追加，例如 "KQ.m@SHFE.rb"（螺纹）
MAIN_CONT_LIST = [
    "KQ.m@SHFE.fu",
    "KQ.m@CZCE.SA",
    "KQ.m@DCE.c"
]

# 下载起止时间，按需修改
START_DATE = datetime(2020, 1, 1)
END_DATE = datetime(2025, 12, 31)

# 数据保存根目录
SAVE_ROOT = "./tick_batch_main"
# =========================================================================

os.makedirs(SAVE_ROOT, exist_ok=True)


def download_single_tick(api, symbol, sdt, edt, out_fp):
    """下载单个真实合约一段区间Tick"""
    if os.path.exists(out_fp):
        print(f"✅ 文件已存在，跳过：{out_fp}")
        return out_fp

    print(f"\n===== 开始下载 {symbol}  [{sdt} ~ {edt}] =====")
    dl = DataDownloader(
        api=api,
        symbol_list=symbol,
        dur_sec=0,  # 0 = Tick，固定不要改
        start_dt=sdt,
        end_dt=edt,
        csv_file_name=out_fp
    )
    while not dl.is_finished():
        api.wait_update()
        progress = dl.get_progress()
        print(f"进度: {progress:.2%}", end="\r")
    print(f"\n✅ 下载完成：{out_fp}")
    return out_fp


def download_and_concat_one_main(api, main_cont):
    """
    处理一个主连品种：
    1. 查询历史主力映射表
    2. 按主力切换点分段下载真实合约tick
    3. 全部下载后合并、按时间排序输出连续tick
    """
    # 取出品种名，用于文件夹命名
    _, code = main_cont.split("@")
    sub_folder = os.path.join(SAVE_ROOT, code)
    os.makedirs(sub_folder, exist_ok=True)

    print(f"\n========== 开始处理品种：{main_cont} ==========")
    # 查询历史主力合约序列（返回日期->对应主力合约）
    df_his_main = api.query_his_cont_quotes(symbol=main_cont, n=9999)
    # 过滤时间区间内的数据
    df_his_main = df_his_main[
        (df_his_main["date"] >= START_DATE) & (df_his_main["date"] <= END_DATE)
    ].copy()
    df_his_main = df_his_main.sort_values("date").reset_index(drop=True)

    if len(df_his_main) == 0:
        print(f"⚠️ {main_cont} 在指定区间没有历史主力数据，跳过")
        return

    downloaded_files = []
    # 遍历每一段主力周期
    for idx, row in df_his_main.iterrows():
        real_symbol = row[main_cont]
        day_start = row["date"]
        # 本段结束日 = 下一条的日期，最后一条用END_DATE
        if idx + 1 < len(df_his_main):
            day_end = df_his_main.iloc[idx + 1]["date"]
        else:
            day_end = END_DATE + timedelta(days=1)

        if not real_symbol:
            print(f"⚠️ {day_start} 无主力合约，跳过")
            continue

        fp = os.path.join(sub_folder, f"{real_symbol}_{day_start.date()}_{day_end.date()}.csv")
        f = download_single_tick(api, real_symbol, day_start, day_end, fp)
        downloaded_files.append(f)

    # 拼接所有分段tick
    print(f"\n===== 开始合并 {main_cont} 全部tick =====")
    df_list = []
    for f in downloaded_files:
        df = pd.read_csv(f, parse_dates=["datetime"])
        df_list.append(df)
    total_df = pd.concat(df_list, ignore_index=True).sort_values("datetime").reset_index(drop=True)
    concat_out = os.path.join(SAVE_ROOT, f"{code}_concat_tick.csv")
    total_df.to_csv(concat_out, index=False)
    print(f"✅ {main_cont} 合并完成！连续文件：{concat_out}")
    print(f"总行数：{len(total_df)}")


def main():
    with closing(TqApi(auth=TqAuth(USER, PWD))) as api:
        for cont in MAIN_CONT_LIST:
            download_and_concat_one_main(api, cont)
    print("\n===== 全部品种下载+拼接完成 =====")


if __name__ == "__main__":
    main()
```
