## 核心一句话
**TqSdk免费版不能下载历史Tick快照；Tick历史下载（DataDownloader，dur_sec=0）是【天勤专业版专属功能】，期货Tick从2016年起提供**
> 免费版 `get_tick_serial()` 只能拿**实时向后一小段Tick（最多8964条）**，只能盘中订阅，不能指定起止日期下载历史Tick，不适合批量回测用

### 1、权限区分（重点）
- ✅ 免费版：
  - `get_tick_serial(symbol, data_length=xxx)`：实时/最近一小段Tick快照，上限8964条，无法指定start/end日期，只能跟着行情流拿
  - 免费版**可以下载长周期1分钟K线**（你之前做DualThrust、海龟够用）
- ✅ 专业版（付费，可申请15天试用）：
  - `tqsdk.tools.DataDownloader`，**dur_sec=0 代表Tick**，指定起止时间，直接导出CSV快照Tick（买一卖一+最新价、持仓、成交量）
  - 时间范围：国内商品期货Tick从2016年开始；支持燃油、纯碱、玉米等
  - 只有**快照Tick（盘口快照）**，**没有逐笔成交/逐笔委托**（天勤不提供逐笔）

### 2、专业版下载Tick历史代码（直接复制）
```python
from tqsdk import TqApi, TqAuth
from tqsdk.tools import DataDownloader
from contextlib import closing
from datetime import datetime

# 填入你的快期账号密码（专业版权限）
api = TqApi(auth=TqAuth("你的账号", "你的密码"))

# dur_sec=0 就是Tick
dl = DataDownloader(
    api=api,
    symbol_list="SHFE.fu2601",  # 燃油合约，注意：不能用主连！只能是真实交割合约
    dur_sec=0,
    start_dt=datetime(2026, 11, 1),
    end_dt=datetime(2026, 11, 2),
    csv_file_name="fu2601_tick.csv"
)

with closing(api):
    while not dl.is_finished():
        api.wait_update()
        print(f"下载进度：{dl.get_progress():.2%}")

print("Tick下载完成")
```
> ⚠️ 重要限制：**不支持主连合约下载Tick**，只能单个真实合约；需要自己写脚本拼接换月

## 1. 免费版盘中自动落盘保存Tick脚本（挂机攒Tick库）
特点：按【合约代码】分文件，追加写入csv；自动去重；捕获异常断线自动重连；使用dotenv放账号密码，不要硬编码。

`.env` 文件（和脚本放同一目录）
```env
TQ_USERNAME=你的快期账号
TQ_PASSWORD=你的快期密码
```

`tick_recorder.py`
```python
import os
import time
from datetime import datetime
import pandas as pd
from dotenv import load_dotenv
from tqsdk import TqApi, TqAuth, TqException

load_dotenv()
USER = os.getenv("TQ_USERNAME")
PWD = os.getenv("TQ_PASSWORD")

# ========== 配置区 ==========
SYMBOL_LIST = ["SHFE.fu", "DCE.c", "CZCE.SA"]  # 燃油、玉米、纯碱，填合约前缀
SAVE_DIR = "./tick_data"
LOG_INTERVAL = 30  # 打印日志间隔秒
# ============================

os.makedirs(SAVE_DIR, exist_ok=True)

def get_csv_path(symbol: str):
    name = symbol.replace(".", "_")
    return os.path.join(SAVE_DIR, f"{name}_tick.csv")

def save_tick_to_csv(tick, symbol: str):
    fp = get_csv_path(symbol)
    row = {
        "datetime": tick["datetime"],
        "last_price": tick["last_price"],
        "average": tick["average"],
        "highest": tick["highest"],
        "lowest": tick["lowest"],
        "ask_price1": tick["ask_price1"],
        "ask_volume1": tick["ask_volume1"],
        "bid_price1": tick["bid_price1"],
        "bid_volume1": tick["bid_volume1"],
        "volume": tick["volume"],
        "open_interest": tick["open_interest"]
    }
    df = pd.DataFrame([row])
    if os.path.exists(fp):
        df.to_csv(fp, mode="a", header=False, index=False)
    else:
        df.to_csv(fp, mode="w", header=True, index=False)

def main():
    while True:
        try:
            api = TqApi(auth=TqAuth(USER, PWD))
            tick_map = {}
            for sym in SYMBOL_LIST:
                tick_map[sym] = api.get_tick_serial(sym)
            print(f"✅ 订阅成功：{SYMBOL_LIST}，开始录制tick，保存目录：{SAVE_DIR}")
            last_log = time.time()

            while True:
                api.wait_update()
                now = time.time()
                for sym, tick_series in tick_map.items():
                    if tick_series.is_changed():
                        latest = tick_series.iloc[-1]
                        save_tick_to_csv(latest, sym)
                if now - last_log > LOG_INTERVAL:
                    print(f"⏰ {datetime.now()}, tick录制中...")
                    last_log = now

        except TqException as e:
            print(f"❌ 天勤异常：{e}, 5s后重连")
        except Exception as e:
            print(f"❌ 未知异常：{e}, 5s后重连")
        time.sleep(5)

if __name__ == "__main__":
    main()
```

> 注意：这里写的是合约前缀 `SHFE.fu`，天勤会自动订阅当前主力合约；主力换月时，tqsdk会自动切换到新主力，csv文件会继续追加，天然实现换月接续，但**不会自动复权**。

## 2. Tick转vnpy导入格式（Tick / 合成Bar两种函数）
vnpy内置的 `CsvTickLoader`、`CsvBarLoader` 有固定字段格式。
输出两份：
- tick 原始csv（vnpy tick格式）
- 可以按周期合成bar（比如1min bar，供DualThrust/海龟回测）

`tick_to_vnpy.py`
```python
import pandas as pd
from datetime import datetime

def convert_tick_for_vnpy(src_csv: str, out_csv: str):
    """
    天勤原始tick csv → vnpy标准Tick csv
    vnpy tick字段：datetime,exchange,symbol,last_price,volume,open_interest,
    bid_price_1,bid_volume_1,ask_price_1,ask_volume_1
    """
    df = pd.read_csv(src_csv, parse_dates=["datetime"])
    # 从文件名解析合约
    base = src_csv.split("/")[-1].replace("_tick.csv","")
    exchange, symbol = base.split("_")

    out = pd.DataFrame()
    out["datetime"] = df["datetime"]
    out["exchange"] = exchange
    out["symbol"] = symbol
    out["last_price"] = df["last_price"]
    out["volume"] = df["volume"]
    out["open_interest"] = df["open_interest"]
    out["bid_price_1"] = df["bid_price1"]
    out["bid_volume_1"] = df["bid_volume1"]
    out["ask_price_1"] = df["ask_price1"]
    out["ask_volume_1"] = df["ask_volume1"]

    out.to_csv(out_csv, index=False)
    print(f"✅ vnpy tick导出完成：{out_csv}")
    return out

def tick_to_bar(df_tick: pd.DataFrame, period: str = "1min") -> pd.DataFrame:
    """
    tick合成bar，period: 1min/5min/15min
    返回：datetime,open,high,low,close,volume,open_interest
    """
    df = df_tick.copy()
    df["datetime"] = pd.to_datetime(df["datetime"])
    df = df.set_index("datetime")
    agg_rule = {
        "last_price": ["first", "max", "min", "last"],
        "volume": "sum",
        "open_interest": "last"
    }
    bar_df = df.resample(period).agg(agg_rule).dropna()
    bar_df.columns = ["open", "high", "low", "close", "volume", "open_interest"]
    bar_df = bar_df.reset_index()
    return bar_df

if __name__ == "__main__":
    # 示例
    src = "./tick_data/SHFE_fu_tick.csv"
    vnpy_tick_out = "./tick_data/SHFE_fu_vnpy_tick.csv"
    df_vnpy_tick = convert_tick_for_vnpy(src, vnpy_tick_out)
    bar1 = tick_to_bar(df_vnpy_tick, period="1min")
    bar1.to_csv("./tick_data/SHFE_fu_1min_bar.csv", index=False)
    print("✅ 1min K线合成完成")
```

> vnpy导入：在vnpy【数据管理】→【导入csv】，选择对应loader：
> - vnpy tick文件 → CsvTickLoader
> - bar文件 → CsvBarLoader

## 3. 合约批量循环下载 + 自动换月拼接脚本（⚠️ 这个是【专业版DataDownloader版本】）
> 重申：免费版**不能用DataDownloader下载历史tick**，这个脚本是给你以后开天勤专业版时用的，批量遍历合约、自动识别主力、分段下载、最后拼接成一个连续tick文件。
> 逻辑：遍历一段时间区间，自动识别每个时段的主力合约，分段下载真实合约tick，最后按时间合并排序，形成连续序列（仅拼接，不复权）

`tick_batch_download_concat.py`
```python
from tqsdk import TqApi, TqAuth
from tqsdk.tools import DataDownloader
from datetime import datetime, timedelta
import pandas as pd
import os
from contextlib import closing

# ==========配置区==========
USER = "你的账号"
PWD = "你的密码"
SYMBOL_PREFIX = "SHFE.fu"  # fu燃油 / c玉米 / SA纯碱
START_DATE = datetime(2025,1,1)
END_DATE = datetime(2025,3,1)
SAVE_FOLDER = "./tick_batch"
# ==========================

os.makedirs(SAVE_FOLDER, exist_ok=True)

def get_main_contract(api, prefix, dt: datetime):
    """获取指定时间点的主力合约"""
    quote = api.get_quote(prefix)
    return quote.underlying_symbol

def download_one_contract(api, symbol, sdt, edt, fp):
    print(f"开始下载 {symbol} | {sdt} ~ {edt}")
    dl = DataDownloader(
        api,
        symbol_list=symbol,
        dur_sec=0,
        start_dt=sdt,
        end_dt=edt,
        csv_file_name=fp
    )
    while not dl.is_finished():
        api.wait_update()
        print(f"进度: {dl.get_progress():.2%}")
    return fp

def batch_download_and_concat():
    api = TqApi(auth=TqAuth(USER, PWD))
    day = START_DATE
    all_files = []
    while day <= END_DATE:
        main_sym = get_main_contract(api, SYMBOL_PREFIX, day)
        day_end = day + timedelta(days=1)
        fname = os.path.join(SAVE_FOLDER, f"{main_sym}_{day.date()}.csv")
        if not os.path.exists(fname):
            download_one_contract(api, main_sym, day, day_end, fname)
        all_files.append(fname)
        day += timedelta(days=1)
    api.close()

    # 拼接所有tick、按时间排序
    dfs = []
    for f in all_files:
        df = pd.read_csv(f, parse_dates=["datetime"])
        dfs.append(df)
    total = pd.concat(dfs).sort_values("datetime").reset_index(drop=True)
    out_path = os.path.join(SAVE_FOLDER, f"{SYMBOL_PREFIX.replace('.','_')}_concat_tick.csv")
    total.to_csv(out_path, index=False)
    print(f"✅ 拼接完成，连续tick文件：{out_path}")

if __name__ == "__main__":
    batch_download_and_concat()
```

## 配套使用建议（适配你vnpy CTA回测）
1. 挂机脚本 `tick_recorder.py` 放服务器/电脑后台长期跑，每天攒tick；**不要用家用电脑随便关机**，关机就断数据。
2. 攒一段时间后，用 `tick_to_vnpy.py` 转成vnpy格式，同时合成1min bar，直接给DualThrust、海龟策略回测。
3. 批量下载脚本，**只有你开专业版之后才生效**，免费版运行会报错。
4. 滑点仿真：如果你用tick回测，注意：天勤只有一档盘口，做滑点仿真会偏乐观；高频盘口策略会失真，趋势策略影响很小。
5. 数据冗余：建议定期备份tick原始csv，避免磁盘损坏丢库。

如果你想要，我再给你一个：**tick库的去重+清洗脚本（剔除重复时间戳、剔除无效价格）**，回测前清洗数据用。
