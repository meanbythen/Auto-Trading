# 总览
天勤 `DataDownloader` 输出原生是 **CSV**，字段：`datetime,last_price,average,highest,lowest,ask_price1,ask_volume1,bid_price1,bid_volume1,volume,open_interest`
你可以转成：
1. **vnpy 格式（Tick / Bar）**（前面已经给过）
2. Parquet（推荐，回测首选，压缩、读写快）
3. SQLite（小数据库，方便按时间查询）
4. HDF5
5. 通达信/文华文本格式
6. 直接写入 vnpy 的 sqlite 数据库（vnpy内置data.db）

> 底层统一流程：**读取天勤CSV → 清洗（前面清洗脚本）→ 字段映射 → 写入目标格式**
> 下面所有示例都基于【清洗后的天勤tick csv】

## 1. 转 Parquet（强烈推荐，CTA回测首选）
优点：体积远小于csv、读取速度快、支持分区存储，tqsdk/vnpy回测都很适合
```python
import pandas as pd

def tick_csv_to_parquet(src_csv: str, out_parquet: str):
    df = pd.read_csv(src_csv, parse_dates=["datetime"])
    # 可选：按日期分区，方便后续切片
    # df["date"] = df["datetime"].dt.date
    df.to_parquet(out_parquet, engine="pyarrow", index=False)
    print(f"✅ 保存parquet：{out_parquet}")

def read_parquet_tick(path: str):
    return pd.read_parquet(path)

if __name__ == "__main__":
    tick_csv_to_parquet("./tick_batch_main/SHFE_fu_concat_tick_clean.csv", "./fu_tick.parquet")
```

## 2. 写入 SQLite（本地轻量数据库，按时间范围查询）
适合你写策略时，按时间段灵活取tick，不用反复读大文件
```python
import pandas as pd
import sqlite3

def tick_csv_to_sqlite(src_csv: str, db_path: str, table_name: str = "tick"):
    df = pd.read_csv(src_csv, parse_dates=["datetime"])
    with sqlite3.connect(db_path) as conn:
        df.to_sql(table_name, conn, if_exists="append", index=False)
        # 给时间建索引，查询提速
        conn.execute(f"CREATE INDEX IF NOT EXISTS idx_dt ON {table_name}(datetime);")
    print(f"✅ 写入sqlite {db_path}，表：{table_name}")

if __name__ == "__main__":
    tick_csv_to_sqlite("./tick_batch_main/SHFE_fu_concat_tick_clean.csv", "./tick_data.db", "fu_tick")
```
查询示例：
```python
with sqlite3.connect("tick_data.db") as conn:
    df = pd.read_sql("SELECT * FROM fu_tick WHERE datetime >= '2024-01-01' AND datetime < '2024-01-02'", conn, parse_dates=["datetime"])
```

## 3. 转 HDF5
适合超大tick库，pandas生态常用
```python
import pandas as pd

def tick_csv_to_hdf5(src_csv: str, out_h5: str, key="tick"):
    df = pd.read_csv(src_csv, parse_dates=["datetime"])
    df.to_hdf(out_h5, key=key, mode="w", format="table")
    print(f"✅ 保存hdf5：{out_h5}")
```

## 4. 文华财经文本格式（wh8导入格式）
文华tick文本格式示例：`时间,价格,成交量,持仓,买一价,买一量,卖一价,卖一量`
```python
import pandas as pd

def tick_to_wh_txt(src_csv: str, out_txt: str):
    df = pd.read_csv(src_csv, parse_dates=["datetime"])
    df["dt_str"] = df["datetime"].dt.strftime("%Y-%m-%d %H:%M:%S.%f")
    out = pd.DataFrame({
        "time": df["dt_str"],
        "price": df["last_price"],
        "volume": df["volume"],
        "oi": df["open_interest"],
        "bid1": df["bid_price1"],
        "bid_vol1": df["bid_volume1"],
        "ask1": df["ask_price1"],
        "ask_vol1": df["ask_volume1"]
    })
    out.to_csv(out_txt, index=False, header=False)
    print(f"✅ 文华tick文本导出：{out_txt}")
```

## 5. 直接导入 vnpy 的 data.db（不用导出csv再手动导入）
vnpy底层就是sqlite，你可以直接把tick写入vnpy数据库，在vnpy界面直接读到
> vnpy tick表字段：datetime,exchange,symbol,last_price,volume,open_interest,bid_price_1,bid_volume_1,ask_price_1,ask_volume_1
```python
import pandas as pd
import sqlite3

def tick_to_vnpy_db(src_csv, vnpy_db_path, exchange, symbol):
    """
    vnpy默认data.db路径示例：
    Windows: C:/users/xxx/.vnpy/data.db
    Mac/Linux: ~/.vnpy/data.db
    """
    df = pd.read_csv(src_csv, parse_dates=["datetime"])
    df_out = pd.DataFrame()
    df_out["datetime"] = df["datetime"]
    df_out["exchange"] = exchange
    df_out["symbol"] = symbol
    df_out["last_price"] = df["last_price"]
    df_out["volume"] = df["volume"]
    df_out["open_interest"] = df["open_interest"]
    df_out["bid_price_1"] = df["bid_price1"]
    df_out["bid_volume_1"] = df["bid_volume1"]
    df_out["ask_price_1"] = df["ask_price1"]
    df_out["ask_volume_1"] = df["ask_volume1"]

    with sqlite3.connect(vnpy_db_path) as conn:
        df_out.to_sql("tick_data", conn, if_exists="append", index=False)
    print(f"✅ 写入vnpy data.db tick表：{exchange}.{symbol}")

if __name__ == "__main__":
    tick_to_vnpy_db(
        src_csv="./tick_batch_main/SHFE_fu_concat_tick_clean.csv",
        vnpy_db_path=r"C:\Users\你的用户名\.vnpy\data.db",
        exchange="SHFE",
        symbol="fu2509"
    )
```

## 6. 通用：Tick合成Bar（1min/5min，前面版本增强版）
```python
import pandas as pd

def tick_to_bar(df_tick: pd.DataFrame, period="1min") -> pd.DataFrame:
    df = df_tick.copy()
    df["datetime"] = pd.to_datetime(df["datetime"])
    df = df.set_index("datetime")
    bar = df.resample(period).agg(
        open=("last_price", "first"),
        high=("last_price", "max"),
        low=("last_price", "min"),
        close=("last_price", "last"),
        volume=("volume", "sum"),
        open_interest=("open_interest", "last")
    ).dropna()
    return bar.reset_index()
```

# 选型建议（适配你的CTA回测）
- 本地回测、vnpy策略开发：**Parquet**（首选）
- 需要在vnpy图形界面看数据：写入vnpy `data.db`
- 要丢给文华wh8回测：转文华txt
- 需要交互式按时间查询：SQLite
- 只是临时交换：csv（原生）
