import numpy as np
import pandas as pd
from typing import Any, Dict, List
from vnpy.cta.strategy import CtaTemplate, StopOrder, TickData, BarData
from vnpy.trader.constant import Direction, Offset, Interval, Status
from vnpy.trader.utility import BarGenerator, ArrayManager
import talib


class LiquidityStopZoneStrategy(CtaTemplate):
    """
    公共止损流动性回流策略｜以损定开
    底层逻辑：识别高低点聚类模拟散户公共止损区间，等待击穿止损区+回流确认进场
    指标：ATR + ADX + 高低点聚类
    内置绩效统计：胜率、盈亏比、最大回撤、趋势/震荡分段统计
    """
    author = "LiquidityStopZone"

    # ===================== 策略参数（可在界面调参） =====================
    atr_period = 20
    adx_period = 14
    adx_threshold = 20          # ADX>=20判定趋势，否则震荡
    cluster_lookback = 20       # 聚类回看K线数量，用来找公共止损区间
    cluster_tol = 0.005         # 聚类价格容忍度（相对波动，根据品种调，纯碱/燃油改）
    atr_stop_multiple = 1.0     # 止损在止损区外侧几倍ATR
    atr_entry_multiple = 0.5    # 进场偏移：止损区上沿+N*ATR
    risk_reward = 2.0           # 目标盈亏比
    risk_per_trade = 0.01       # 单笔账户风险比例 1%
    intra_day_close_minute = 20 # 收盘前N分钟强制平仓（日内）

    # ===================== 变量 =====================
    atr_value = 0.0
    adx_value = 0.0
    plus_di = 0.0
    minus_di = 0.0

    zone_l = 0.0    # 多头公共止损区间下沿
    zone_h = 0.0    # 多头公共止损区间上沿

    long_entry = 0.0
    long_stop = 0.0
    long_target = 0.0

    short_entry = 0.0
    short_stop = 0.0
    short_target = 0.0

    # 绩效记录
    trade_records: List[Dict[str, Any]] = []
    start_balance = 0.0
    current_balance = 0.0
    max_balance = 0.0

    parameters = [
        "atr_period", "adx_period", "adx_threshold", "cluster_lookback",
        "cluster_tol", "atr_stop_multiple", "atr_entry_multiple",
        "risk_reward", "risk_per_trade", "intra_day_close_minute"
    ]
    variables = [
        "atr_value", "adx_value", "plus_di", "minus_di",
        "zone_l", "zone_h", "long_entry", "long_stop", "long_target",
        "short_entry", "short_stop", "short_target"
    ]

    def __init__(self, cta_engine, strategy_name, vt_symbol, setting):
        super().__init__(cta_engine, strategy_name, vt_symbol, setting)
        self.bg = BarGenerator(self.on_bar, 1, self.on_1min_bar)
        self.am = ArrayManager(size=max(self.cluster_lookback, self.atr_period, self.adx_period) + 50)
        self.last_bar_time = None
        self.trade_records.clear()

    def on_init(self):
        self.write_log("策略初始化：LiquidityStopZone 止损流动性回流策略")
        self.load_bar(10)

    def on_start(self):
        self.write_log("策略启动")

    def on_stop(self):
        self.write_log("策略停止，打印绩效报告")
        self.print_performance()

    def on_tick(self, tick: TickData):
        self.bg.update_tick(tick)

    def on_bar(self, bar: BarData):
        self.bg.update_bar(bar)

    def on_1min_bar(self, bar: BarData):
        """核心K线逻辑，1min周期，可自行改成5min"""
        self.am.update_bar(bar)
        if not self.am.inited:
            return

        # ========== 1. 计算指标 ATR / ADX / +DI / -DI ==========
        high = self.am.high
        low = self.am.low
        close = self.am.close

        self.atr_value = talib.ATR(high, low, close, timeperiod=self.atr_period)[-1]
        adx = talib.ADX(high, low, close, timeperiod=self.adx_period)
        pdi = talib.PLUS_DI(high, low, close, timeperiod=self.adx_period)
        mdi = talib.MINUS_DI(high, low, close, timeperiod=self.adx_period)

        self.adx_value = adx[-1]
        self.plus_di = pdi[-1]
        self.minus_di = mdi[-1]

        is_trend = self.adx_value >= self.adx_threshold
        is_bull = self.plus_di > self.minus_di
        is_bear = self.minus_di > self.plus_di

        # ========== 2. 高低点聚类：计算多头公共止损区间（支撑下方止损集群） ==========
        self.calc_stop_zone()

        # ========== 3. 日内收盘强制平仓 ==========
        if self.is_need_close(bar):
            if self.pos > 0:
                self.sell(bar.close, self.pos, offset=Offset.CLOSE)
            elif self.pos < 0:
                self.cover(bar.close, abs(self.pos), offset=Offset.CLOSE)
            return

        # ========== 4. 多头开仓逻辑：击穿止损区 -> 回流站上zone_h+偏移ATR ==========
        if self.pos == 0 and is_trend and is_bull:
            # 条件：K线曾经下破zone_l，随后回升
            price_break_zone = self.am.low[-2] < self.zone_l
            price_recover = bar.close > self.zone_h + self.atr_entry_multiple * self.atr_value
            if price_break_zone and price_recover:
                self.open_long(bar)

        # ========== 5. 空头开仓逻辑（对称） ==========
        if self.pos == 0 and is_trend and is_bear:
            price_break_zone = self.am.high[-2] > self.zone_h
            price_recover = bar.close < self.zone_l - self.atr_entry_multiple * self.atr_value
            if price_break_zone and price_recover:
                self.open_short(bar)

        # ========== 6. 多头持仓：止损止盈 ==========
        if self.pos > 0:
            if bar.low <= self.long_stop:
                self.sell(self.long_stop, self.pos, offset=Offset.CLOSE)
                self.record_trade("long", "stop", bar.datetime, bar.close, is_trend)
            elif bar.high >= self.long_target:
                self.sell(self.long_target, self.pos, offset=Offset.CLOSE)
                self.record_trade("long", "target", bar.datetime, bar.close, is_trend)

        # ========== 7. 空头持仓：止损止盈 ==========
        if self.pos < 0:
            if bar.high >= self.short_stop:
                self.cover(self.short_stop, abs(self.pos), offset=Offset.CLOSE)
                self.record_trade("short", "stop", bar.datetime, bar.close, is_trend)
            elif bar.low <= self.short_target:
                self.cover(self.short_target, abs(self.pos), offset=Offset.CLOSE)
                self.record_trade("short", "target", bar.datetime, bar.close, is_trend)

        self.put_event()

    def calc_stop_zone(self):
        """
        高低点聚类算法：提取最近cluster_lookback根K线局部低点，聚类得到公共止损区间
        zone_l/zone_h：多头止损集群区间（支撑下方散户多单止损埋单区）
        """
        look = self.cluster_lookback
        lows = self.am.low[-look:]
        highs = self.am.high[-look:]

        # 简单局部低点筛选（局部极小值）
        local_lows = []
        for i in range(2, len(lows)-2):
            if lows[i] < lows[i-1] and lows[i] < lows[i+1]:
                local_lows.append(lows[i])
        if len(local_lows) < 3:
            # 低点不足，用最近低点区间兜底
            self.zone_l = np.min(lows)
            self.zone_h = np.min(lows) * (1 + self.cluster_tol)
            return

        # 简单聚类：取价格密集中心
        arr = np.array(local_lows)
        center = np.median(arr)
        self.zone_l = center * (1 - self.cluster_tol)
        self.zone_h = center * (1 + self.cluster_tol)

    def open_long(self, bar: BarData):
        """多头开仓 + ATR动态仓位"""
        self.long_entry = bar.close
        self.long_stop = self.zone_l - self.atr_stop_multiple * self.atr_value
        risk_points = self.long_entry - self.long_stop
        if risk_points <= 0:
            return
        # vnpy获取合约乘数，CTP品种
        contract = self.cta_engine.get_contract(self.vt_symbol)
        if not contract:
            return
        # 动态仓位计算：单笔风险=资金*风险比例
        capital = self.cta_engine.get_account().balance
        risk_amount = capital * self.risk_per_trade
        one_lot_risk = risk_points * contract.size
        lot = int(risk_amount / one_lot_risk)
        lot = max(1, lot)

        self.buy(self.long_entry, lot, offset=Offset.OPEN)
        self.write_log(f"开多｜价格:{self.long_entry:.2f},止损:{self.long_stop:.2f},止盈:{self.long_target:.2f},手数:{lot}")
        self.long_target = self.long_entry + self.risk_reward * risk_points

    def open_short(self, bar: BarData):
        """空头开仓"""
        self.short_entry = bar.close
        self.short_stop = self.zone_h + self.atr_stop_multiple * self.atr_value
        risk_points = self.short_stop - self.short_entry
        if risk_points <= 0:
            return
        contract = self.cta_engine.get_contract(self.vt_symbol)
        if not contract:
            return
        capital = self.cta_engine.get_account().balance
        risk_amount = capital * self.risk_per_trade
        one_lot_risk = risk_points * contract.size
        lot = int(risk_amount / one_lot_risk)
        lot = max(1, lot)

        self.short(self.short_entry, lot, offset=Offset.OPEN)
        self.short_target = self.short_entry - self.risk_reward * risk_points
        self.write_log(f"开空｜价格:{self.short_entry:.2f},止损:{self.short_stop:.2f},止盈:{self.short_target:.2f},手数:{lot}")

    def is_need_close(self, bar: BarData) -> bool:
        """日内收盘平仓判断"""
        close_time = bar.datetime.replace(hour=14, minute=60 - self.intra_day_close_minute)
        if bar.datetime >= close_time:
            return True
        return False

    def record_trade(self, direction: str, exit_type: str, dt, exit_price, is_trend: bool):
        """记录每一笔交易，用于绩效统计"""
        if direction == "long":
            entry = self.long_entry
            pnl = (exit_price - entry)
        else:
            entry = self.short_entry
            pnl = (entry - exit_price)
        self.trade_records.append({
            "dt": dt,
            "direction": direction,
            "entry": entry,
            "exit": exit_price,
            "pnl": pnl,
            "exit_type": exit_type,
            "is_trend": is_trend
        })

    def print_performance(self):
        """绩效输出：总盈亏、胜率、盈亏比、最大回撤、趋势/震荡分组统计"""
        if len(self.trade_records) == 0:
            self.write_log("暂无交易记录")
            return
        df = pd.DataFrame(self.trade_records)
        # 区分趋势交易 / 震荡交易
        df_trend = df[df["is_trend"] == True]
        df_side = df[df["is_trend"] == False]

        def calc_stats(sub_df: pd.DataFrame, name: str):
            if len(sub_df) == 0:
                return {"count":0}
            win = sub_df[sub_df["pnl"]>0]
            lose = sub_df[sub_df["pnl"]<=0]
            win_rate = len(win)/len(sub_df)
            avg_win = win["pnl"].mean() if len(win)>0 else 0
            avg_loss = lose["pnl"].mean() if len(lose)>0 else 0
            profit_factor = abs(avg_win / avg_loss) if avg_loss !=0 else np.inf
            cum = sub_df["pnl"].cumsum()
            max_dd = self.calc_max_drawdown(cum.values)
            return {
                "name":name,
                "count":len(sub_df),
                "win_rate":round(win_rate,3),
                "avg_win":round(avg_win,2),
                "avg_loss":round(avg_loss,2),
                "profit_factor":round(profit_factor,3),
                "max_drawdown":round(max_dd,2)
            }
        all_stats = calc_stats(df, "全部交易")
        trend_stats = calc_stats(df_trend, "趋势交易")
        side_stats = calc_stats(df_side, "震荡交易")
        self.write_log("===== 绩效报告 =====\n")
        for s in [all_stats, trend_stats, side_stats]:
            self.write_log(str(s))

    @staticmethod
    def calc_max_drawdown(equity_arr: np.ndarray):
        """计算最大回撤"""
        peak = np.maximum.accumulate(equity_arr)
        dd = (peak - equity_arr) / peak
        return np.max(dd)
