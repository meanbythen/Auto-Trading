from vnpy.trader.constant import Direction
import talib

class BoxBreakoutStrategy(CtaTemplate):
    # 参数
    box_lookback = 40        # 箱体回看K线数量
    atr_length = 20
    adx_length = 14
    risk_per_trade = 0.015   # 单笔风险1.5%
    atr_stop_buffer = 0.5
    trail_atr = 2.0

    def on_bar(self, bar):
        am = self.am
        am.update_bar(bar)
        if not am.inited:
            return

        # 指标
        high_array = am.high
        low_array = am.low
        close_array = am.close
        atr = talib.ATR(high_array, low_array, close_array, self.atr_length)[-1]
        adx = talib.ADX(high_array, low_array, close_array, self.adx_length)[-1]

        # 箱体上下沿：取前面box_lookback根K线高低点（不含当前bar）
        box_high = max(am.high[-self.box_lookback-1:-1])
        box_low = min(am.low[-self.box_lookback-1:-1])
        box_height = box_high - box_low

        # 过滤条件
        cond1 = box_height >= atr * 1.2
        cond2 = adx > 20

        # 多头突破：收盘站上箱体上沿
        long_break = bar.close > box_high
        # 空头突破：收盘跌破箱体下沿
        short_break = bar.close < box_low

        # 多头开仓
        if self.pos == 0 and cond1 and cond2 and long_break:
            stop_price = box_high - atr * self.atr_stop_buffer
            # 动态计算手数
            risk_price = bar.close - stop_price
            if risk_price <= 0:
                return
            risk_money = self.capital * self.risk_per_trade
            volume = int(risk_money / (risk_price * self.contract_size))
            if volume <=0:
                return
            self.buy(bar.close, volume)
            self.long_stop = stop_price
            self.trail_start_price = bar.close + box_height

        # 空头开仓
        elif self.pos ==0 and cond1 and cond2 and short_break:
            stop_price = box_low + atr * self.atr_stop_buffer
            risk_price = stop_price - bar.close
            if risk_price <=0:
                return
            risk_money = self.capital * self.risk_per_trade
            volume = int(risk_money / (risk_price * self.contract_size))
            if volume <=0:
                return
            self.short(bar.close, volume)
            self.short_stop = stop_price
            self.trail_start_price = bar.close - box_height

        # 持仓：跟踪止盈
        elif self.pos >0:
            if bar.close >= self.trail_start_price:
                new_trail = max(self.long_stop, am.high[-1] - atr * self.trail_atr)
                self.long_stop = new_trail
            if bar.close <= self.long_stop:
                self.sell(bar.close, abs(self.pos))

        elif self.pos <0:
            if bar.close <= self.trail_start_price:
                new_trail = min(self.short_stop, am.low[-1] + atr * self.trail_atr)
                self.short_stop = new_trail
            if bar.close >= self.short_stop:
                self.cover(bar.close, abs(self.pos))
