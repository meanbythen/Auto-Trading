# ====================== 【模块A：置换检验 Permutation Test for Rank IC】 ======================
# 原假设 H0：因子与未来收益无真实预测关系，观测IC是随机得到
import numpy as np
import pandas as pd
from scipy.stats import spearmanr

def permutation_test_ic(
    df: pd.DataFrame,
    factor_col: str,
    target_col: str,
    n_perm: int = 1000,
    random_seed: int = 42
):
    """
    df: 包含因子、未来收益的DataFrame
    factor_col: 因子列名
    target_col: 未来收益标签列 future_ret_N
    n_perm: 置换次数，机构常用500~1000
    return: 真实ic, 置换IC分布数组, p_value
    """
    np.random.seed(random_seed)
    data = df[[factor_col, target_col]].dropna().copy()
    x = data[factor_col].values
    y_true = data[target_col].values

    # 真实IC
    ic_true, _ = spearmanr(x, y_true)

    perm_ic_list = []
    for _ in range(n_perm):
        y_shuffle = np.random.permutation(y_true)  # 只打乱标签，因子时序不变
        ic_p, _ = spearmanr(x, y_shuffle)
        perm_ic_list.append(ic_p)

    perm_ic_arr = np.array(perm_ic_list)
    # 双侧检验：统计有多少个置换IC比真实IC更极端
    p_value = np.sum(np.abs(perm_ic_arr) >= np.abs(ic_true)) / n_perm

    return ic_true, perm_ic_arr, p_value

print("="*60)

print(f"置换检验 p值 = {p:.4f}")
print(f"判断规则：p < 0.05 认为统计显著；p高说明大概率是随机噪声")
print("="*60)

# ====================== 【模块B：简易WFA Walk Forward 时序滚动检验】 ======================
# 逻辑：滚动训练窗口内计算IC/阈值，滚动窗口外（样本外）评估
# 时序滚动，绝不打乱，适配CTA因子研究
def walk_forward_factor_test(
    df: pd.DataFrame,
    factor_col: str,
    target_col: str,
    train_window: int = 250,    # 训练窗口bar数量
    step: int = 20,             # 滚动步长，每step根bar滚动一次
    quantile_long: float = 0.8, # 分位数做多：因子高于该分位数
    quantile_short: float = 0.2 # 分位数做空：因子低于该分位数
):
    """
    返回：
    wfa_result_df：每一段样本外绩效
    out_full_df：带信号、样本外标记的完整主表
    """
    data = df[[factor_col, target_col]].dropna().copy()
    data = data.sort_index()
    n = len(data)
    results = []
    out_full = data.copy()
    out_full["wfa_signal"] = np.nan
    out_full["is_oos"] = False

    start_idx = 0
    while True:
        train_end = start_idx + train_window
        oos_start = train_end
        oos_end = oos_start + step
        if oos_end >= n:
            break

        train_set = data.iloc[start_idx:train_end]
        oos_set = data.iloc[oos_start:oos_end]

        # 在训练集上计算分位数（只用训练集信息！无未来函数）
        q_long = train_set[factor_col].quantile(quantile_long)
        q_short = train_set[factor_col].quantile(quantile_short)

        # 样本外生成信号
        sig = np.where(
            oos_set[factor_col] >= q_long, 1,
            np.where(oos_set[factor_col] <= q_short, -1, 0)
        )
        oos_ret = sig * oos_set[target_col]

        # 样本外绩效指标
        ret_mean = oos_ret.mean()
        ret_std = oos_ret.std()
        hit_rate = np.sum(oos_ret > 0) / len(oos_ret)
        total_ret = oos_ret.sum()
        if ret_std > 0:
            sharpe = ret_mean / ret_std * np.sqrt(252)
        else:
            sharpe = np.nan

        results.append({
            "oos_start": oos_set.index[0],
            "oos_end": oos_set.index[-1],
            "train_q_long": q_long,
            "train_q_short": q_short,
            "oos_mean_ret": ret_mean,
            "oos_sharpe": sharpe,
            "oos_hitrate": hit_rate,
            "oos_total_ret": total_ret
        })

        # 回填标记，方便画图
        out_full.loc[oos_set.index, "wfa_signal"] = sig
        out_full.loc[oos_set.index, "is_oos"] = True

        start_idx += step

    wfa_result_df = pd.DataFrame(results)
    return wfa_result_df, out_full

# ---------------- 使用示例 ----------------

    df=bar_df,

    target_col="future_ret_5",

    quantile_long=0.8,

)

print("="*60)

print(f"WFA平均分段夏普: {wfa_res['oos_sharpe'].mean():.4f}")
