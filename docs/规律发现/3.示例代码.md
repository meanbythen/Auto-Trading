import warnings
warnings.filterwarnings("ignore")

import numpy as np
import pandas as pd
from scipy.stats import spearmanr
from sklearn.feature_selection import mutual_info_regression
from sklearn.cluster import KMeans
from sklearn.preprocessing import StandardScaler
from sklearn.metrics import silhouette_score

# -----------------------------------------------------------------------------
# 1. Hurst指数 R/S 重标极差算法 (输入：收益率序列 array)
# -----------------------------------------------------------------------------
def hurst(ts: np.ndarray) -> float:
    """
    R/S 重标极差法计算Hurst指数
    H > 0.5 : 趋势性；H=0.5：随机游走；H<0.5：均值回归
    ts: 收益率序列，一维np array
    """
    ts = np.asarray(ts)
    n = len(ts)
    if n < 20:
        return np.nan

    # 分段长度
    max_lag = int(n / 2)
    lags = np.arange(2, max_lag)
    rs = []

    for lag in lags:
        # 拆分
        chunks = np.array_split(ts, lag)
        rs_list = []
        for chunk in chunks:
            if len(chunk) < 2:
                continue
            mean_c = np.mean(chunk)
            dev = chunk - mean_c
            cum_dev = np.cumsum(dev)
            R = np.max(cum_dev) - np.min(cum_dev)
            S = np.std(chunk, ddof=1)
            if S == 0:
                continue
            rs_list.append(R / S)
        if len(rs_list) > 0:
            rs.append(np.mean(rs_list))

    if len(rs) < 4:
        return np.nan
    log_lag = np.log10(lags[:len(rs)])
    log_rs = np.log10(rs)
    slope = np.polyfit(log_lag, log_rs, deg=1)[0]
    return slope

def rolling_hurst(series: pd.Series, window: int = 100) -> pd.Series:
    """滚动Hurst，窗口建议60~200根K线，输入收益率序列"""
    res = []
    arr = series.values
    for i in range(len(arr)):
        if i < window - 1:
            res.append(np.nan)
        else:
            win = arr[i-window+1:i+1]
            res.append(hurst(win))
    return pd.Series(res, index=series.index, name="hurst")

# -----------------------------------------------------------------------------
# 2. 批量计算Spearman相关矩阵
# -----------------------------------------------------------------------------
def spearman_corr_matrix(df: pd.DataFrame) -> pd.DataFrame:
    """df: 因子矩阵，每行样本，每列因子"""
    cols = df.columns.tolist()
    n = len(cols)
    corr_mat = np.zeros((n, n))
    p_mat = np.zeros((n, n))
    for i, c1 in enumerate(cols):
        for j, c2 in enumerate(cols):
            r, p = spearmanr(df[c1].dropna(), df[c2].dropna())
            corr_mat[i,j] = r
            p_mat[i,j] = p
    corr_df = pd.DataFrame(corr_mat, index=cols, columns=cols)
    p_df = pd.DataFrame(p_mat, index=cols, columns=cols)
    return corr_df, p_df

# -----------------------------------------------------------------------------
# 3. 互信息矩阵 mutual_info_regression（非线性关联，对连续标签）
# -----------------------------------------------------------------------------
def mutual_info_matrix(df: pd.DataFrame, target_col: str = None) -> pd.DataFrame:
    """
    如果target_col给定：只算各个因子和目标标签（未来收益）的互信息
    如果target_col=None：因子两两之间互信息矩阵（慢，慎用）
    """
    df_clean = df.dropna().copy()
    if target_col is not None:
        y = df_clean[target_col].values
        X = df_clean.drop(columns=[target_col]).values
        mi = mutual_info_regression(X, y, random_state=42)
        mi_df = pd.DataFrame({"mutual_info": mi}, index=df_clean.drop(columns=[target_col]).columns)
        return mi_df
    else:
        cols = df_clean.columns
        mi_mat = np.zeros((len(cols), len(cols)))
        for i, c1 in enumerate(cols):
            y = df_clean[c1].values
            X = df_clean.drop(columns=[c1]).values
            mi = mutual_info_regression(X, y, random_state=42)
            mi_mat[i, :] = np.insert(mi, i, 0)
        return pd.DataFrame(mi_mat, index=cols, columns=cols)

# -----------------------------------------------------------------------------
# 4. Kmeans行情状态聚类：输入因子矩阵，输出聚类标签 + 轮廓系数
# -----------------------------------------------------------------------------
def kmeans_market_cluster(df_feature: pd.DataFrame, n_clusters: int = 3):
    """
    输入：标准化前的特征df（波动率、ATR、量比、动量等行情特征）
    返回：带cluster标签的df，kmeans模型，轮廓系数
    cluster 0/1/2 对应不同市场状态：趋势/震荡/高波动等
    """
    df_clean = df_feature.dropna().copy()
    scaler = StandardScaler()
    X_scaled = scaler.fit_transform(df_clean.values)

    kmeans = KMeans(n_clusters=n_clusters, random_state=42, n_init="auto")
    labels = kmeans.fit_predict(X_scaled)
    score = silhouette_score(X_scaled, labels)

    out_df = df_clean.copy()
    out_df["cluster"] = labels
    return out_df, kmeans, scaler, score

# 5. 【示例：构造因子 + 主流程，对接vnpy bar数据】
# -----------------------------------------------------------------------------
# --------------------------
# 模拟：你从vnpy载入bar数据，这里替换成你真实的bar df
# bar_df 列：datetime, open, high, low, close, volume
# --------------------------
np.random.seed(42)
idx = pd.date_range(start="2026-01-01", periods=1000, freq="1min")
price = np.cumsum(np.random.randn(1000)*0.2) + 100
bar_df = pd.DataFrame({
    "datetime": idx,
    "close": price,
    "volume": np.random.randint(100, 5000, size=1000)
})
bar_df.set_index("datetime", inplace=True)

# ---------- 构造基础因子（你可以叠加ATR、ADX、RSI、动量等） ----------
bar_df["ret"] = bar_df["close"].pct_change()
bar_df["vol_5"] = bar_df["ret"].rolling(5).std()
bar_df["vol_20"] = bar_df["ret"].rolling(20).std()
bar_df["mom_10"] = bar_df["close"] / bar_df["close"].shift(10) - 1
bar_df["vol_ratio"] = bar_df["volume"] / bar_df["volume"].rolling(20).mean()

# 未来收益标签（预测目标，注意：实盘不能用未来函数！研究用）
bar_df["future_ret_5"] = bar_df["close"].shift(-5) / bar_df["close"] - 1

# 用于分析的因子列表
feature_cols = ["ret", "vol_5", "vol_20", "mom_10", "vol_ratio"]
feature_df = bar_df[feature_cols + ["future_ret_5"]].copy()

# %%
# ========== 模块1：Spearman相关性 ==========
spear_corr, spear_p = spearman_corr_matrix(feature_df)
print("="*50)
print("【Spearman 秩相关矩阵】")
display(spear_corr.round(3))
print("【P值矩阵】")
display(spear_p.round(4))

# %%
# ========== 模块2：互信息：因子 vs 未来收益 ==========
mi_target = mutual_info_matrix(feature_df, target_col="future_ret_5")
print("="*50)
print("【因子对未来收益的互信息（非线性关联）】越高越强")
display(mi_target.sort_values("mutual_info", ascending=False).round(4))

# %%
# ========== 模块3：滚动Hurst指数 ==========
bar_df["hurst_100"] = rolling_hurst(bar_df["ret"], window=100)
print("="*50)
print("【滚动Hurst统计】")
print(bar_df[["close","hurst_100"]].tail(10))

# %%
# ========== 模块4：Kmeans市场状态聚类（只用行情特征，不含未来标签） ==========
cluster_df_in = bar_df[["vol_5","vol_20","mom_10","vol_ratio","hurst_100"]].copy()
cluster_out, kmodel, scaler, sil_score = kmeans_market_cluster(cluster_df_in, n_clusters=3)
print("="*50)
print(f"Kmeans轮廓系数 silhouette_score={sil_score:.3f}，越接近1聚类效果越好")
print("各聚类样本数量：")
print(cluster_out["cluster"].value_counts().sort_index())
print("聚类中心（原始特征空间）：")
cluster_centers = pd.DataFrame(scaler.inverse_transform(kmodel.cluster_centers_),
                               columns=["vol_5","vol_20","mom_10","vol_ratio","hurst_100"])
display(cluster_centers.round(3))

# 把聚类标签合并回主表
bar_df = bar_df.join(cluster_out["cluster"], how="left")

# %%
# ========== 简单分组统计：不同聚类状态下，未来收益分布 ==========
cluster_stats = bar_df.groupby("cluster")["future_ret_5"].agg(["mean","std","count"])
print("="*50)
print("【不同市场状态下未来收益统计】用来检验状态是否有区分度")
display(cluster_stats.round(4))
