# 打分+投票混合版多因子策略

# 打分+投票混合版策略设计说明

核心分层架构：**投票层管「能不能开仓」（鲁棒性，过滤噪声），打分层管「开多大仓位」（信号强度，区分强弱）**

- 第一层（准入门槛）：4因子等权投票，必须满足最低同向票数、票数差过滤、全局市场状态三层筛选，通过才允许开仓，分歧大直接弃权不交易
- 第二层（仓位分级）：同一套4因子输出标准化打分，加权计算总分，按总分绝对值线性映射仓位系数（0.5~1.0），信号越强仓位越高
- 风控底仓：以「单笔风险1%的ATR动态手数」为基准仓位，乘以仓位系数，同时设置最大仓位上限，风险始终可控
- 归因体系：同时保留投票制因子胜率统计 + 打分制盈亏分摊，双维度监控因子衰减

class MultiFactor4HybridStrategy(CtaTemplate):
    """

    author = "study"

    min_position_ratio = 0.5  # 最低仓位系数（刚满足门槛时）
    max_position_ratio = 1.0  # 最高仓位系数（满分时）

    atr_stop_multiple = 2.0

    # 投票层变量
    v_ma = 0

    long_votes = 0

    # 打分层变量

    # 交易记录
    trade_records: List[HybridTradeRecord]

        "atr_len", "adx_len", "atr_window",

    def on_day_bar(self, bar: BarData):
        """日线大周期方向过滤"""
        self.am_day.update_bar(bar)
        if not self.am_day.inited:
            return
        ma60_day = self.am_day.sma(self.ma_long)
        close_day = self.am_day.close[-1]
        self.long_bias = close_day > ma60_day
        self.short_bias = close_day < ma60_day

    def calc_vote_layer(self, close, ma_s, ma_l, h20, l20, boll_mid, boll_upper, boll_lower, boll_slope, sar):
        """计算投票层：输出各因子投票 + 是否通过投票门槛"""
        # 因子1：均线趋势
        self.v_ma = 0
        ma_cross_up = (self.am5.sma(self.ma_short, 1)[-2] <= self.am5.sma(self.ma_long, 1)[-2]) and (ma_s > ma_l)
        ma_cross_down = (self.am5.sma(self.ma_short, 1)[-2] >= self.am5.sma(self.ma_long, 1)[-2]) and (ma_s < ma_l)
        if ma_cross_up and close > ma_l and self.long_bias:
            self.v_ma = 1
        elif ma_cross_down and close < ma_l and self.short_bias:
            self.v_ma = -1

        # 因子2：唐奇安突破
        self.v_don = 0
        if close > h20 and self.long_bias:
            self.v_don = 1
        elif close < l20 and self.short_bias:
            self.v_don = -1

        # 因子3：布林带突破
        self.v_boll = 0
        if boll_slope > 0 and close > boll_upper and self.long_bias:
            self.v_boll = 1
        elif boll_slope < 0 and close < boll_lower and self.short_bias:
            self.v_boll = -1

        # 因子4：SAR转向
        self.v_sar = 0
        if sar < close and self.long_bias:
            self.v_sar = 1
        elif sar > close and self.short_bias:
            self.v_sar = -1

        # 票数统计
        vote_list = [self.v_ma, self.v_don, self.v_boll, self.v_sar]
        self.long_votes = sum(1 for v in vote_list if v == 1)
        self.short_votes = sum(1 for v in vote_list if v == -1)
        vote_diff = abs(self.long_votes - self.short_votes)

        # 投票门槛判断
        self.vote_pass = False
        if self.long_votes >= self.min_votes and vote_diff >= self.min_vote_diff:
            self.vote_pass = True
        if self.short_votes >= self.min_votes and vote_diff >= self.min_vote_diff:
            self.vote_pass = True

    def calc_score_layer(self, close, ma_s, ma_l, h20, l20, mid_don, boll_mid, boll_upper, boll_lower, boll_slope, sar):
        """计算打分层：输出各因子得分 + 总分 + 仓位系数"""
        # 因子1：均线趋势
        self.s_ma = 0.0
        if close > ma_l and ma_s > ma_l:
            cross = (self.am5.sma(self.ma_short, 1)[-2] <= self.am5.sma(self.ma_long, 1)[-2]) and (ma_s > ma_l)
            self.s_ma = 1.0 if cross else 0.5
        elif close < ma_l and ma_s < ma_l:
            cross = (self.am5.sma(self.ma_short, 1)[-2] >= self.am5.sma(self.ma_long, 1)[-2]) and (ma_s < ma_l)
            self.s_ma = -1.0 if cross else -0.5
        if self.s_ma > 0 and not self.long_bias: self.s_ma = 0
        if self.s_ma < 0 and not self.short_bias: self.s_ma = 0

        # 因子2：唐奇安突破
        self.s_don = 0.0
        if close > h20:
            self.s_don = 1.0
        elif close < l20:
            self.s_don = -1.0
        elif close > mid_don:
            self.s_don = 0.5
        elif close < mid_don:
            self.s_don = -0.5
        if self.s_don > 0 and not self.long_bias: self.s_don = 0
        if self.s_don < 0 and not self.short_bias: self.s_don = 0

        # 因子3：布林带趋势
        self.s_boll = 0.0
        if boll_slope > 0 and close > boll_mid:
            self.s_boll = 1.0 if close > boll_upper else 0.5
        elif boll_slope < 0 and close < boll_mid:
            self.s_boll = -1.0 if close < boll_lower else -0.5
        if self.s_boll > 0 and not self.long_bias: self.s_boll = 0
        if self.s_boll < 0 and not self.short_bias: self.s_boll = 0

        # 因子4：SAR转向
        self.s_sar = 0.0
        if sar < close:
            self.s_sar = 1.0
        elif sar > close:
            self.s_sar = -1.0
        if self.s_sar > 0 and not self.long_bias: self.s_sar = 0
        if self.s_sar < 0 and not self.short_bias: self.s_sar = 0

        # 总分合成
        self.total_score = (
            self.s_ma * self.w_ma +
            self.s_don * self.w_don +
            self.s_boll * self.w_boll +
            self.s_sar * self.w_sar
        )

        # 总分映射仓位系数：线性映射 [score_threshold, 1.0] -> [min_ratio, max_ratio]
        abs_score = abs(self.total_score)
        if abs_score <= self.score_threshold:
            self.position_ratio = self.min_position_ratio
        elif abs_score >= 1.0:
            self.position_ratio = self.max_position_ratio
        else:
            # 线性插值
            ratio = (abs_score - self.score_threshold) / (1.0 - self.score_threshold)
            self.position_ratio = self.min_position_ratio + ratio * (self.max_position_ratio - self.min_position_ratio)

    def on_5min_bar(self, bar: BarData):
        self.cancel_all()
        self.am5.update_bar(bar)
        if not self.am5.inited or not self.am_day.inited:
            return

        close = self.am5.close[-1]
        atr = self.am5.atr(self.atr_len)
        adx = self.am5.adx(self.adx_len)
        ma_s = self.am5.sma(self.ma_short)
        ma_l = self.am5.sma(self.ma_long)

        # 唐奇安
        h20 = self.am5.high[-self.donchian_period:].max()
        l20 = self.am5.low[-self.donchian_period:].min()
        mid_don = (h20 + l20) / 2

        # 布林带
        boll_mid = self.am5.sma(self.boll_len)
        boll_std_val = self.am5.std(self.boll_len)
        boll_upper = boll_mid + self.boll_std * boll_std_val
        boll_lower = boll_mid - self.boll_std * boll_std_val
        boll_slope = boll_mid - self.am5.sma(self.boll_len, 1)[-1]

        # SAR
        sar = self.am5.sar(self.sar_af, self.sar_maxaf)

        # ATR分位
        self.atr_list.append(atr)
        if len(self.atr_list) > self.atr_window:
            self.atr_list.pop(0)
        atr_quantile = np.percentile(self.atr_list, 20)

        # ========== 1. 两层计算 ==========
        self.calc_vote_layer(close, ma_s, ma_l, h20, l20, boll_mid, boll_upper, boll_lower, boll_slope, sar)
        self.calc_score_layer(close, ma_s, ma_l, h20, l20, mid_don, boll_mid, boll_upper, boll_lower, boll_slope, sar)

        # ========== 2. 全局市场状态过滤 ==========
        self.is_tradeable = True
        if adx < 20:
            self.is_tradeable = False
        if atr < atr_quantile:
            self.is_tradeable = False
        # 混合版额外：投票不通过，直接不可交易
        if not self.vote_pass:
            self.is_tradeable = False

        # ========== 3. 收盘强制清仓 ==========
        is_close_time = (
            (bar.datetime.hour == 14 and bar.datetime.minute >= 60 - self.close_minute)
            or (bar.datetime.hour == 22 and bar.datetime.minute >= 60 - self.close_minute)
        )
        if is_close_time:
            if self.pos > 0:
                self.sell(close, self.pos)
            elif self.pos < 0:
                self.cover(close, abs(self.pos))
            return

        # ========== 4. 动态仓位计算（基准仓 × 仓位系数）==========
        contract = self.get_contract()
        if not contract:
            return
        multiplier = contract.size
        # 基准仓位：单笔风险1%的ATR手数
        base_volume = (self.risk_per_trade * self.cta_engine.capital) / (atr * self.atr_stop_multiple * multiplier)
        # 实际仓位 = 基准仓 × 仓位系数
        real_volume = max(1, int(base_volume * self.position_ratio))

        # ========== 5. 开仓逻辑 ==========
        if self.is_tradeable and self.pos == 0:
            # 多头方向
            if self.long_votes > self.short_votes and self.total_score > self.score_threshold:
                self.buy(close, real_volume)
                self.stop_loss_price = close - self.atr_stop_multiple * atr
                self.open_record = HybridTradeRecord(
                    trade_id=bar.datetime.strftime("%Y%m%d%H%M"),
                    datetime=bar.datetime,
                    direction=1,
                    v_ma=self.v_ma, v_don=self.v_don, v_boll=self.v_boll, v_sar=self.v_sar,
                    long_votes=self.long_votes, short_votes=self.short_votes,
                    s_ma=self.s_ma, s_don=self.s_don, s_boll=self.s_boll, s_sar=self.s_sar,
                    total_score=self.total_score,
                    position_ratio=self.position_ratio,
                    entry_price=close,
                    volume=real_volume
                )
            # 空头方向
            elif self.short_votes > self.long_votes and self.total_score < -self.score_threshold:
                self.short(close, real_volume)
                self.stop_loss_price = close + self.atr_stop_multiple * atr
                self.open_record = HybridTradeRecord(
                    trade_id=bar.datetime.strftime("%Y%m%d%H%M"),
                    datetime=bar.datetime,
                    direction=-1,
                    v_ma=self.v_ma, v_don=self.v_don, v_boll=self.v_boll, v_sar=self.v_sar,
                    long_votes=self.long_votes, short_votes=self.short_votes,
                    s_ma=self.s_ma, s_don=self.s_don, s_boll=self.s_boll, s_sar=self.s_sar,
                    total_score=self.total_score,
                    position_ratio=self.position_ratio,
                    entry_price=close,
                    volume=real_volume
                )

        # ========== 6. 移动止损 ==========
        if self.pos > 0:
            new_sl = close - self.atr_stop_multiple * atr
            if new_sl > self.stop_loss_price:
                self.stop_loss_price = new_sl
            if close <= self.stop_loss_price:
                self.sell(close, self.pos)
        elif self.pos < 0:
            new_sl = close + self.atr_stop_multiple * atr
            if new_sl < self.stop_loss_price:
                self.stop_loss_price = new_sl
            if close >= self.stop_loss_price:
                self.cover(close, abs(self.pos))

        # ========== 7. 投票反转平仓 ==========
        if self.pos > 0 and self.short_votes >= self.min_votes and abs(self.short_votes - self.long_votes) >= self.min_vote_diff:
            self.sell(close, self.pos)
        if self.pos < 0 and self.long_votes >= self.min_votes and abs(self.long_votes - self.short_votes) >= self.min_vote_diff:
            self.cover(close, abs(self.pos))

        self.put_event()

    def on_trade(self, trade: TradeData):
        """成交回调：平仓时计算盈亏"""
        if self.open_record is None:
            return
        rec = self.open_record
        contract = self.get_contract()
        if not contract:
            return

        if trade.offset == Offset.CLOSE:
            rec.exit_price = trade.price
            mult = contract.size
            if rec.direction == 1:
                pnl = (rec.exit_price - rec.entry_price) * rec.volume * mult
            else:
                pnl = (rec.entry_price - rec.exit_price) * rec.volume * mult
            rec.pnl = pnl
            self.trade_records.append(rec)
            self.open_record = None

        self.put_event()

    def print_hybrid_report(self):
        """输出混合版双维度绩效报告"""
        # ===== 投票维度绩效 =====
        vote_stats = {
            "ma": {"pnl": 0.0, "count": 0, "win": 0},
            "don": {"pnl": 0.0, "count": 0, "win": 0},
            "boll": {"pnl": 0.0, "count": 0, "win": 0},
            "sar": {"pnl": 0.0, "count": 0, "win": 0},
        }
        vote_keys = {"ma": "v_ma", "don": "v_don", "boll": "v_boll", "sar": "v_sar"}

        # ===== 打分维度绩效 =====
        score_stats = {
            "ma": {"pnl": 0.0, "count": 0, "win": 0},
            "don": {"pnl": 0.0, "count": 0, "win": 0},
            "boll": {"pnl": 0.0, "count": 0, "win": 0},
            "sar": {"pnl": 0.0, "count": 0, "win": 0},
        }

        for r in self.trade_records:
            # 投票维度统计
            for name, key in vote_keys.items():
                vote = getattr(r, key)
                if vote == 0:
                    continue
                vote_stats[name]["count"] += 1
                if vote == r.direction:
                    vote_stats[name]["pnl"] += r.pnl
                    if r.pnl > 0:
                        vote_stats[name]["win"] += 1
                else:
                    vote_stats[name]["pnl"] -= r.pnl

            # 打分维度统计（按得分绝对值分摊）
            s_ma = abs(r.s_ma)
            s_don = abs(r.s_don)
            s_boll = abs(r.s_boll)
            s_sar = abs(r.s_sar)
            total_s = s_ma + s_don + s_boll + s_sar
            if total_s < 1e-6:
                continue
            p_ma = r.pnl * s_ma / total_s
            p_don = r.pnl * s_don / total_s
            p_boll = r.pnl * s_boll / total_s
            p_sar = r.pnl * s_sar / total_s

            score_stats["ma"]["pnl"] += p_ma
            score_stats["don"]["pnl"] += p_don
            score_stats["boll"]["pnl"] += p_boll
            score_stats["sar"]["pnl"] += p_sar
            score_stats["ma"]["count"] += 1
            score_stats["don"]["count"] += 1
            score_stats["boll"]["count"] += 1
            score_stats["sar"]["count"] += 1
            if p_ma > 0: score_stats["ma"]["win"] += 1
            if p_don > 0: score_stats["don"]["win"] += 1
            if p_boll > 0: score_stats["boll"]["win"] += 1
            if p_sar > 0: score_stats["sar"]["win"] += 1

        self.write_log("=" * 50)
        self.write_log("【投票维度 因子绩效】")
        for name, st in vote_stats.items():
            wr = st["win"] / st["count"] if st["count"] else 0
            self.write_log(f"{name:4s} | 总盈亏:{st['pnl']:>8.2f} | 投票次数:{st['count']:>3} | 胜率:{wr:.2%}")

        self.write_log("-" * 50)
        self.write_log("【打分维度 因子绩效】")
        for name, st in score_stats.items():
            wr = st["win"] / st["count"] if st["count"] else 0
            self.write_log(f"{name:4s} | 总盈亏:{st['pnl']:>8.2f} | 交易次数:{st['count']:>3} | 胜率:{wr:.2%}")
        self.write_log("=" * 50)

        # 仓位统计
        ratios = [r.position_ratio for r in self.trade_records]
        if ratios:
            self.write_log(f"平均仓位系数: {np.mean(ratios):.3f} | 最高: {np.max(ratios):.3f} | 最低: {np.min(ratios):.3f}")

# 二、配套混合版因子衰减绘图代码

同时绘制**投票维度滚动胜率** + **打分维度滚动盈亏** + **仓位系数时序**，三图联动分析因子衰减与仓位分布。

```
import pandas as pd
import matplotlib.pyplot as plt
import numpy as np

plt.rcParams['font.sans-serif'] = ['SimHei']
plt.rcParams['axes.unicode_minus'] = False

# ====================== 配置 ======================
CSV_PATH = "hybrid_trade_log.csv"
ROLL_WINDOW = 20
# ==================================================


def load_hybrid_data(csv_path):
    df = pd.read_csv(csv_path, parse_dates=["datetime"])
    df = df.sort_values("datetime").reset_index(drop=True)
    factors = ["ma", "don", "boll", "sar"]

    # 投票维度盈亏
    for f in factors:
        col = f"v_{f}"
        df[f"vote_pnl_{f}"] = np.where(
            df[col] == 0,
            0,
            np.where(df[col] == df["direction"], df["pnl"], -df["pnl"])
        )

    # 打分维度盈亏（按得分分摊）
    def calc_score_pnl(row):
        s_ma = abs(row["s_ma"])
        s_don = abs(row["s_don"])
        s_boll = abs(row["s_boll"])
        s_sar = abs(row["s_sar"])
        total_s = s_ma + s_don + s_boll + s_sar
        if total_s < 1e-6:
            return pd.Series([0, 0, 0, 0])
        return pd.Series([
            row["pnl"] * s_ma / total_s,
            row["pnl"] * s_don / total_s,
            row["pnl"] * s_boll / total_s,
            row["pnl"] * s_sar / total_s
        ])

    df[["score_pnl_ma", "score_pnl_don", "score_pnl_boll", "score_pnl_sar"]] = df.apply(calc_score_pnl, axis=1)
    return df


def plot_hybrid_decay(df):
    factors = ["ma", "don", "boll", "sar"]
    colors = ["#1f77b4", "#ff7f0e", "#2ca02c", "#d62728"]

    fig = plt.figure(figsize=(14, 12))
    gs = fig.add_gridspec(3, 1, height_ratios=[3, 2, 1])

    # 图1：打分维度 累计盈亏 + 滚动盈亏
    ax1 = fig.add_subplot(gs[0])
    ax1.set_title(f"混合版策略 因子衰减分析（滚动窗口={ROLL_WINDOW}笔）", fontsize=14)
    for f, c in zip(factors, colors):
        cum = df[f"score_pnl_{f}"].cumsum()
        roll = df[f"score_pnl_{f}"].rolling(ROLL_WINDOW).sum()
        ax1.plot(df["datetime"], cum, label=f"{f} 累计盈亏(打分)", color=c, lw=1.8)
        ax1.plot(df["datetime"], roll, label=f"{f} 滚动盈亏(打分)", color=c, lw=1.2, ls="--")
    ax1.axhline(y=0, color="k", lw=0.8, alpha=0.6)
    ax1.set_ylabel("打分维度盈亏")
    ax1.legend(loc="upper left", ncol=2)
    ax1.grid(alpha=0.3)

    # 图2：投票维度 滚动胜率
    ax2 = fig.add_subplot(gs[1], sharex=ax1)
    for f, c in zip(factors, colors):
        win_flag = df[f"vote_pnl_{f}"] > 0
        vote_flag = df[f"vote_pnl_{f}"] != 0
        roll_win = win_flag.rolling(ROLL_WINDOW).sum()
        roll_count = vote_flag.rolling(ROLL_WINDOW).sum()
        roll_wr = roll_win / roll_count.replace(0, np.nan)
        ax2.plot(df["datetime"], roll_wr, label=f"{f} 滚动胜率(投票)", color=c, lw=1.5)
    ax2.axhline(y=0.5, color="k", lw=0.8, alpha=0.6, ls="--")
    ax2.set_ylabel("投票维度胜率")
    ax2.set_ylim(0, 1)
    ax2.legend(loc="lower left", ncol=2)
    ax2.grid(alpha=0.3)

    # 图3：仓位系数时序
    ax3 = fig.add_subplot(gs[2], sharex=ax1)
    ax3.plot(df["datetime"], df["position_ratio"], color="#9467bd", lw=1.5, label="实际仓位系数")
    ax3.axhline(y=df["position_ratio"].mean(), color="k", lw=0.8, alpha=0.6, ls="--", label="平均仓位")
    ax3.set_ylabel("仓位系数")
    ax3.set_ylim(0, 1.1)
    ax3.legend(loc="lower left")
    ax3.grid(alpha=0.3)
    ax3.set_xlabel("时间")

    plt.tight_layout()
    plt.savefig("hybrid_factor_decay.png", dpi=300, bbox_inches="tight")
    plt.show()

    # 打印汇总表
    print("===== 打分维度绩效汇总 =====")
    score_summary = pd.DataFrame(index=factors)
    for f in factors:
        col = f"score_pnl_{f}"
        score_summary.loc[f, "总盈亏"] = df[col].sum()
        score_summary.loc[f, "交易次数"] = (df[col] != 0).sum()
        score_summary.loc[f, "盈利次数"] = (df[col] > 0).sum()
    score_summary["胜率"] = score_summary["盈利次数"] / score_summary["交易次数"]
    print(score_summary.round(2))

    print("\n===== 投票维度绩效汇总 =====")
    vote_summary = pd.DataFrame(index=factors)
    for f in factors:
        col = f"vote_pnl_{f}"
        vote_summary.loc[f, "总盈亏"] = df[col].sum()
        vote_summary.loc[f, "投票次数"] = (df[col] != 0).sum()
        vote_summary.loc[f, "正确次数"] = (df[col] > 0).sum()
    vote_summary["胜率"] = vote_summary["正确次数"] / vote_summary["投票次数"]
    print(vote_summary.round(2))

    print(f"\n仓位系数：均值={df['position_ratio'].mean():.3f} | 最大={df['position_ratio'].max():.3f} | 最小={df['position_ratio'].min():.3f}")


if __name__ == "__main__":
    df = load_hybrid_data(CSV_PATH)
    plot_hybrid_decay(df)
```

# 三、核心调参建议与优缺点总结

## 1. 关键参数调参方向

| 参数                   | 调大效果             | 调小效果           | 推荐初始值 |
| -------------------- | ---------------- | -------------- | ----- |
| `min_votes`          | 信号更少更准，胜率提升，机会减少 | 信号更多，胜率下降，机会增加 | 2     |
| `min_vote_diff`      | 分歧过滤更严，噪声更少      | 分歧容忍度高，信号更多    | 2     |
| `score_threshold`    | 仓位分级起点更高，弱信号仓位更低 | 弱信号也能拿到较高仓位    | 0.6   |
| `min_position_ratio` | 最低仓位更高，资金利用率高    | 最低仓位更低，弱信号风险小  | 0.5   |
| `max_position_ratio` | 强信号仓位上限更高，盈亏弹性大  | 强信号也轻仓，波动更稳    | 1.0   |

## 2. 混合版核心优势

1. **兼顾鲁棒性与弹性**：投票层过滤掉大部分分歧大的假信号，打分层在强共识时加大仓位，震荡市少亏、趋势市多赚
2. **风险更可控**：仓位上限锁死，单笔风险始终不超过基准风控，不会因为打分高就无限加仓
3. **双维度归因**：既能看每个因子投票准不准，也能看每个因子盈亏贡献，定位衰减更精准
4. **适配性强**：比纯投票版资金利用率高，比纯打分版回撤更稳，适合燃油、纯碱这类趋势和震荡交替的品种

## 3. 注意避坑点

1. 不要把`min_position_ratio`设为0，否则弱信号仓位太小，手续费滑点会吃掉全部收益
2. 投票门槛和打分阈值要匹配，避免投票通过但打分不够、或者打分够但投票不通过的空转情况
3. 仓位映射建议用线性即可，不要搞复杂的非线性映射，参数越多越容易过拟合
4. 三个版本（纯打分、纯投票、混合版）建议用同一份数据回测对比，选最适配你交易品种的版本
