> 说明：
> 1. 本代码基于 vnpy 官方 CtaTemplate，**周线作为趋势周期（定方向）、日线作为交易周期（唐奇安入场）**，符合视频里「大周期定方向、小周期择时」，周线≈日线的4倍，匹配四倍原则；
> 2. 规则：周线多头环境，日线只允许做多；周线空头环境，日线只允许做空；周线无趋势（震荡），直接禁止开仓；
> 3. 底层入场信号：经典唐奇安 55/20；
> 4. 增加 ADX 过滤（周线ADX过滤大震荡）、ATR动态止损；
> 5. 风险提示：仅策略演示代码，**不构成交易建议，实盘务必先充分回测**；
> 6. 注意：vnpy 多周期数据需要同时订阅日线+周线，回测时推荐用复权连续合约。

```python
from vnpy.trader.constant import Interval
from vnpy.trader.utility import BarGenerator, ArrayManager
from vnpy.app.cta_strategy import CtaTemplate, CtaEngine, StopOrder
from vnpy.trader.object import BarData, TickData


class DualCycleDonchianLongStrategy(CtaTemplate):
    """
    双周期长线唐奇安策略
    大周期：周线（趋势方向过滤）
    交易周期：日线（唐奇安入场出场）
    规则：
    1. 周线ADX < adx_threshold：判定震荡，不开任何新仓
    2. 周线价格 > 周线唐奇安上轨：仅允许日线做多
    3. 周线价格 < 周线唐奇安下轨：仅允许日线做空
    4. 日线唐奇安突破上轨开多；突破下轨开空（受上面方向过滤约束）
    5. ATR动态止损
    """
    author = "DualCycle"

    # 参数
    donchian_period = 55        # 唐奇安周期
    exit_period = 20            # 离场唐奇安周期
    adx_period = 14             # ADX周期
    adx_threshold = 22          # 周线ADX阈值，低于则震荡不交易
    atr_period = 14
    atr_multiplier = 2.5        # ATR止损系数，长线可以调到2.5~3.5

    # 变量
    weekly_up = 0.0
    weekly_down = 0.0
    weekly_adx = 0.0
    daily_up = 0.0
    daily_down = 0.0
    daily_atr = 0.0

    parameters = [
        "donchian_period",
        "exit_period",
        "adx_period",
        "adx_threshold",
        "atr_period",
        "atr_multiplier"
    ]
    variables = [
        "weekly_up",
        "weekly_down",
        "weekly_adx",
        "daily_up",
        "daily_down",
        "daily_atr"
    ]

    def __init__(self, cta_engine: CtaEngine, strategy_name, vt_symbol, setting):
        super().__init__(cta_engine, strategy_name, vt_symbol, setting)

        # 日线K线管理
        self.daily_bg = BarGenerator(self.on_daily_bar, interval=Interval.DAY)
        self.daily_am = ArrayManager(100)

        # 周线K线管理
        self.weekly_bg = BarGenerator(self.on_weekly_bar, interval=Interval.WEEK)
        self.weekly_am = ArrayManager(100)

        self.long_stop = 0.0
        self.short_stop = 0.0

    def on_tick(self, tick: TickData):
        """tick推送，给bar generator"""
        self.daily_bg.update_tick(tick)
        self.weekly_bg.update_tick(tick)

    def on_bar(self, bar: BarData):
        """1分钟bar驱动（vnpy常规驱动方式）"""
        self.daily_bg.update_bar(bar)
        self.weekly_bg.update_bar(bar)

    def on_weekly_bar(self, bar: BarData):
        """周线：只用来计算趋势过滤条件"""
        self.weekly_am.update_bar(bar)
        if not self.weekly_am.inited:
            return

        # 周线唐奇安高低轨
        self.weekly_up = self.weekly_am.high[-self.donchian_period:].max()
        self.weekly_down = self.weekly_am.low[-self.donchian_period:].min()
        self.weekly_adx = self.weekly_am.adx(self.adx_period)

        self.write_log(f"【周线过滤】上轨:{self.weekly_up:.2f},下轨:{self.weekly_down:.2f},ADX:{self.weekly_adx:.2f}")

    def on_daily_bar(self, bar: BarData):
        """日线：执行唐奇安开仓、止损逻辑"""
        self.cancel_all()
        self.daily_am.update_bar(bar)

        if not self.daily_am.inited or not self.weekly_am.inited:
            return

        # 日线指标
        self.daily_up = self.daily_am.high[-self.donchian_period:].max()
        self.daily_down = self.daily_am.low[-self.donchian_period:].min()
        self.daily_atr = self.daily_am.atr(self.atr_period)

        # ========== 核心：双周期过滤规则 ==========
        # 条件1：周线ADX不足，大级别震荡，禁止开仓
        if self.weekly_adx < self.adx_threshold:
            self.write_log("周线ADX不足，大级别震荡，不开新仓")
            # 已有持仓继续持有，只禁止新开
            self.exit_signal(bar)
            return

        # 周线趋势判定
        weekly_trend_up = bar.close > self.weekly_up
        weekly_trend_down = bar.close < self.weekly_down

        # 多头逻辑：只有周线多头环境，日线突破才允许开多
        if weekly_trend_up:
            if self.pos == 0:
                self.buy(self.daily_up, self.daily_up, stop=True)
                self.write_log(f"周线多头环境，日线突破上轨开多，价格:{self.daily_up}")
            elif self.pos > 0:
                self.long_stop = self.daily_up - self.atr_multiplier * self.daily_atr
                self.sell(self.long_stop, self.long_stop, stop=True)

        # 空头逻辑：只有周线空头环境，日线突破才允许开空
        elif weekly_trend_down:
            if self.pos == 0:
                self.short(self.daily_down, self.daily_down, stop=True)
                self.write_log(f"周线空头环境，日线突破下轨开空，价格:{self.daily_down}")
            elif self.pos < 0:
                self.short_stop = self.daily_down + self.atr_multiplier * self.daily_atr
                self.cover(self.short_stop, self.short_stop, stop=True)
        else:
            # 周线既不满足多头、也不满足空头区间，只管理已有持仓，不新开
            self.exit_signal(bar)

    def exit_signal(self, bar: BarData):
        """持仓离场：20周期唐奇安反向离场，海龟原版出场"""
        exit_up = self.daily_am.high[-self.exit_period:].max()
        exit_down = self.daily_am.low[-self.exit_period:].min()
        if self.pos > 0:
            self.sell(exit_down, exit_down, stop=True)
        elif self.pos < 0:
            self.cover(exit_up, exit_up, stop=True)

    def on_stop_order(self, so: StopOrder):
        """停止单回调"""
        pass

    def on_init(self):
        self.write_log("策略初始化")
        self.load_bar(200)

    def on_start(self):
        self.write_log("策略启动")

    def on_stop(self):
        self.write_log("策略停止")
```

## 配套说明（贴合视频里「顺大逆小」）
1. **分工**
- 周线（大周期）：判断有没有趋势、方向是什么；震荡直接封死新开仓，过滤逆大周期假突破；
- 日线（交易周期）：唐奇安突破入场、20周期唐奇安离场、ATR止损；
> 和视频里的「大周期定方向，小周期找入场」完全对齐。

2. 参数适配你之前的品种池
- 白糖、尿素、玉米：`atr_multiplier=2.2~2.5`
- 玻璃、生猪、纸浆：`atr_multiplier=3.0~3.5`（波动大、跳空风险高）
- `adx_threshold=22`，周线ADX低于22，直接不做新单子，规避大周期震荡

3. 和单周期唐奇安对比
- ✅ 优点：减少很多逆大周期的假突破，尤其适合玻璃、生猪这种洗盘很重的品种；
- ❌ 缺点：会漏掉一部分日线启动、但周线还没跟上的中等行情；回测信号变少。

4. 部署&回测注意点
- 回测必须用**复权连续合约**；
- 回测驱动：优先1min bar回测，bar generator自动合成日线、周线；
- 如果你想**白糖单独用这套，玉米尿素继续跑简单单周期唐奇安**，可以在cta引擎里新建两个策略实例，分开配置。
