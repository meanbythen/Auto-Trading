> ⚠️ 重要前置说明
1. 缠论本身递归、中枢构建、标准背驰是高度主观的，不存在100%还原人工缠论的完美代码；下面实现的是**工程简化版**：周线定方向 + 日线MACD辅助背驰 + 强弱分型，贴合视频里这套交易框架，适合做趋势跟踪过滤，不是原版严格缠论递归。
2. 框架：**周线（趋势周期）定大势，只允许顺大周期方向开仓；日线（交易周期）识别分型+MACD背驰作为入场触发**
3. 风险提示：仅为学习演示代码，**不构成交易建议，实盘前务必充分回测、调参**
4. 适配：你长线国内期货（白糖/尿素/玻璃/生猪等），基于vnpy CtaTemplate

```python
from vnpy.trader.constant import Interval
from vnpy.trader.utility import BarGenerator, ArrayManager
from vnpy.app.cta_strategy import CtaTemplate, CtaEngine, StopOrder
from vnpy.trader.object import BarData, TickData


class WeeklyDailyChanPriceActionStrategy(CtaTemplate):
    """
    【周线趋势 + 日线缠论分型+MACD背驰 简化版】
    源自视频：价格行为+缠论双周期系统
    规则：
    1. 周线：判断大趋势方向 + 周线ADX过滤震荡
    2. 只允许顺周线大周期方向开仓
    3. 日线：识别强底分型/强顶分型 + MACD辅助背驰作为入场信号
    4. ATR动态止损
    做多：周线多头 + 日线下跌后底背驰 + 强底分型
    做空：周线空头 + 日线上涨后顶背驰 + 强顶分型
    """
    author = "Chan_PA_DualCycle"

    # ========= 参数 =========
    adx_period = 14
    adx_threshold = 22          # 周线ADX低于该值，判定大震荡，不开新仓
    macd_fast = 12
    macd_slow = 26
    macd_signal = 9
    atr_period = 14
    atr_multiplier = 2.8        # 止损倍数，波动大品种调到3.0~3.5
    # 分型需要缓存K线数量
    lookback_k = 3

    # ========= 变量 =========
    weekly_adx = 0.0
    weekly_trend = 0    # 1多头 -1空头 0震荡
    daily_dif = 0.0
    daily_dea = 0.0
    daily_macd = 0.0
    daily_atr = 0.0

    # 背驰缓存（简化MACD高低点记录）
    last_low_price = 0.0
    last_low_dif = 0.0
    last_high_price = 0.0
    last_high_dif = 0.0

    parameters = [
        "adx_period",
        "adx_threshold",
        "macd_fast",
        "macd_slow",
        "macd_signal",
        "atr_period",
        "atr_multiplier"
    ]
    variables = [
        "weekly_adx",
        "weekly_trend",
        "daily_dif",
        "daily_dea",
        "daily_atr"
    ]

    def __init__(self, cta_engine: CtaEngine, strategy_name, vt_symbol, setting):
        super().__init__(cta_engine, strategy_name, vt_symbol, setting)

        # BarGenerator 自动合成日线、周线
        self.daily_bg = BarGenerator(self.on_daily_bar, interval=Interval.DAY)
        self.daily_am = ArrayManager(80)

        self.weekly_bg = BarGenerator(self.on_weekly_bar, interval=Interval.WEEK)
        self.weekly_am = ArrayManager(80)

        self.long_stop = 0.0
        self.short_stop = 0.0

    def on_tick(self, tick: TickData):
        self.daily_bg.update_tick(tick)
        self.weekly_bg.update_tick(tick)

    def on_bar(self, bar: BarData):
        self.daily_bg.update_bar(bar)
        self.weekly_bg.update_bar(bar)

    def on_weekly_bar(self, bar: BarData):
        """周线：判定大趋势方向，ADX过滤震荡"""
        self.weekly_am.update_bar(bar)
        if not self.weekly_am.inited:
            self.weekly_trend = 0
            return

        self.weekly_adx = self.weekly_am.adx(self.adx_period)
        # 周线简单趋势判定：价格在周线20周期均线上方=多头，下方=空头
        ma20_weekly = self.weekly_am.sma(20)
        if self.weekly_adx < self.adx_threshold:
            self.weekly_trend = 0
        elif bar.close > ma20_weekly:
            self.weekly_trend = 1
        else:
            self.weekly_trend = -1

        self.write_log(f"【周线过滤】ADX={self.weekly_adx:.2f}, trend={self.weekly_trend}, ma20={ma20_weekly:.2f}")

    def check_strong_bottom_fractal(self) -> bool:
        """
        强底分型（视频定义）
        三根K：中间K低点最低；第三根向上突破第一根高点
        """
        if self.daily_am.count < 3:
            return False
        h1 = self.daily_am.high[-3]
        l1 = self.daily_am.low[-3]
        h2 = self.daily_am.high[-2]
        l2 = self.daily_am.low[-2]
        h3 = self.daily_am.high[-1]
        l3 = self.daily_am.low[-1]
        # 中间一根低点最低
        if l2 < l1 and l2 < l3:
            # 第三根突破第一根高点 = 强底分型
            if h3 > h1:
                return True
        return False

    def check_strong_top_fractal(self) -> bool:
        """
        强顶分型（视频定义）
        三根K：中间K高点最高；第三根跌破第一根低点
        """
        if self.daily_am.count < 3:
            return False
        h1 = self.daily_am.high[-3]
        l1 = self.daily_am.low[-3]
        h2 = self.daily_am.high[-2]
        l2 = self.daily_am.low[-2]
        h3 = self.daily_am.high[-1]
        l3 = self.daily_am.low[-1]
        # 中间一根高点最高
        if h2 > h1 and h2 > h3:
            # 第三根跌破第一根低点 = 强顶分型
            if l3 < l1:
                return True
        return False

    def check_bear_divergence(self) -> bool:
        """简化：下跌底背驰：价格创新低，DIF不创新低"""
        price_now = self.daily_am.low[-1]
        dif_now = self.daily_dif
        if price_now < self.last_low_price and dif_now > self.last_low_dif:
            return True
        # 更新低点记录
        if price_now < self.last_low_price or self.last_low_price == 0:
            self.last_low_price = price_now
            self.last_low_dif = dif_now
        return False

    def check_bull_divergence(self) -> bool:
        """简化：上涨顶背驰：价格创新高，DIF不创新高"""
        price_now = self.daily_am.high[-1]
        dif_now = self.daily_dif
        if price_now > self.last_high_price and dif_now < self.last_high_dif:
            return True
        if price_now > self.last_high_price or self.last_high_price == 0:
            self.last_high_price = price_now
            self.last_high_dif = dif_now
        return False

    def on_daily_bar(self, bar: BarData):
        """日线：分型 + MACD背驰入场，顺周线方向"""
        self.cancel_all()
        self.daily_am.update_bar(bar)
        if not self.daily_am.inited or self.weekly_trend == 0:
            return

        self.daily_dif, self.daily_dea, self.daily_macd = self.daily_am.macd(
            self.macd_fast, self.macd_slow, self.macd_signal
        )
        self.daily_atr = self.daily_am.atr(self.atr_period)

        # ========== 做多条件：周线多头 + 底背驰 + 强底分型 ==========
        if self.weekly_trend == 1 and self.pos == 0:
            if self.check_bear_divergence() and self.check_strong_bottom_fractal():
                self.buy(bar.close, bar.close, stop=False)
                self.long_stop = bar.close - self.atr_multiplier * self.daily_atr
                self.write_log(f"【做多信号】周线多头，日线底背驰+强底分型，开多 {bar.close}，止损 {self.long_stop:.2f}")
        # ========== 做空条件：周线空头 + 顶背驰 + 强顶分型 ==========
        elif self.weekly_trend == -1 and self.pos == 0:
            if self.check_bull_divergence() and self.check_strong_top_fractal():
                self.short(bar.close, bar.close, stop=False)
                self.short_stop = bar.close + self.atr_multiplier * self.daily_atr
                self.write_log(f"【做空信号】周线空头，日线顶背驰+强顶分型，开空 {bar.close}，止损 {self.short_stop:.2f}")

        # 持仓止损管理
        if self.pos > 0:
            self.sell(self.long_stop, self.long_stop, stop=True)
        elif self.pos < 0:
            self.cover(self.short_stop, self.short_stop, stop=True)

        # 止盈：反向强分型离场（视频规则：小周期反向分型减仓/清仓）
        if self.pos > 0 and self.check_strong_top_fractal():
            self.sell(bar.close, bar.close, stop=False)
            self.write_log("【多头离场】日线强顶分型，多单离场")
        if self.pos < 0 and self.check_strong_bottom_fractal():
            self.cover(bar.close, bar.close, stop=False)
            self.write_log("【空头离场】日线强底分型，空单离场")

    def on_stop_order(self, so: StopOrder):
        pass

    def on_init(self):
        self.write_log("策略初始化：周线趋势+日线缠论分型背驰简化版")
        self.load_bar(250)

    def on_start(self):
        self.write_log("策略启动")

    def on_stop(self):
        self.write_log("策略停止")
```

## 代码局限（一定要清楚）
1. **没有实现缠论中枢递归**：原版缠论中枢、严格递归、线段划分是非常复杂的，无法简单在ArrayManager里低成本实现；这里用MACD代替动力学背驰，是工程简化，和人工手绘缠论有区别。
2. 背驰是简化版：只对比最近一个高低点DIF，不是标准的区间力度对比。
3. 周线趋势判断：用20SMA+ADX，你也可以改成周线唐奇安轨（和上一套双周期唐奇安保持一致）。

## 适配你长线品种的参数参考
- 玉米、尿素、白糖：`atr_multiplier=2.5`
- 玻璃、生猪、纸浆：`atr_multiplier=3.0~3.5`

## 和上一套唐奇安双周期对比
- 上一套：突破跟踪，趋势走出来之后上车，**右侧突破**，信号稳定、回测友好；
- 这套：回调背驰+分型，**顺大周期抓回调拐点**，入场位置更好，但是假信号更多，对背驰判断的容错更低。
