依赖：`numpy, scipy, matplotlib`

```python
import numpy as np
from scipy.stats import norm
from scipy.optimize import minimize
import matplotlib.pyplot as plt

# ===================== 基础复用函数（直接承接前面代码） =====================
N = norm.cdf
n_pdf = norm.pdf

def bs_option(S, K, T, r, sigma, q=0.0, option_type="call"):
    d1 = (np.log(S / K) + (r - q + 0.5 * sigma ** 2) * T) / (sigma * np.sqrt(T))
    d2 = d1 - sigma * np.sqrt(T)
    if option_type == "call":
        price = S * np.exp(-q * T) * N(d1) - K * np.exp(-r * T) * N(d2)
    else:
        price = K * np.exp(-r * T) * N(-d2) - S * np.exp(-q * T) * N(-d1)
    return price

def bs_greeks(S, K, T, r, sigma, q=0.0, option_type="call"):
    d1 = (np.log(S / K) + (r - q + 0.5 * sigma ** 2) * T) / (sigma * np.sqrt(T))
    d2 = d1 - sigma * np.sqrt(T)
    pdf_d1 = n_pdf(d1)
    if option_type == "call":
        delta = np.exp(-q * T) * N(d1)
    else:
        delta = np.exp(-q * T) * (N(d1) - 1)
    gamma = np.exp(-q * T) * pdf_d1 / (S * sigma * np.sqrt(T))
    term1 = - S * np.exp(-q * T) * pdf_d1 * sigma / (2 * np.sqrt(T))
    if option_type == "call":
        term2 = q * S * np.exp(-q * T) * N(d1)
        term3 = - r * K * np.exp(-r * T) * N(d2)
    else:
        term2 = - q * S * np.exp(-q * T) * N(-d1)
        term3 = r * K * np.exp(-r * T) * N(-d2)
    theta = term1 + term2 + term3
    vega = S * np.exp(-q * T) * pdf_d1 * np.sqrt(T)
    if option_type == "call":
        rho = K * T * np.exp(-r * T) * N(d2)
    else:
        rho = -K * T * np.exp(-r * T) * N(-d2)
    return {"delta": delta, "gamma": gamma, "theta": theta, "vega": vega, "rho": rho}

def black_futures_option(F, K, T, r, sigma, option_type="call"):
    d1 = (np.log(F / K) + 0.5 * sigma ** 2 * T) / (sigma * np.sqrt(T))
    d2 = d1 - sigma * np.sqrt(T)
    disc = np.exp(-r * T)
    if option_type == "call":
        price = disc * (F * N(d1) - K * N(d2))
    else:
        price = disc * (K * N(-d2) - F * N(-d1))
    return price

def black_greeks(F, K, T, r, sigma, option_type="call"):
    d1 = (np.log(F / K) + 0.5 * sigma ** 2 * T) / (sigma * np.sqrt(T))
    d2 = d1 - sigma * np.sqrt(T)
    pdf_d1 = n_pdf(d1)
    disc = np.exp(-r * T)
    if option_type == "call":
        delta = disc * N(d1)
    else:
        delta = disc * (N(d1) - 1)
    gamma = disc * pdf_d1 / (F * sigma * np.sqrt(T))
    term1 = - F * disc * pdf_d1 * sigma / (2 * np.sqrt(T))
    if option_type == "call":
        term2 = r * F * disc * N(d1) - r * K * disc * N(d2)
    else:
        term2 = r * K * disc * N(-d2) - r * F * disc * N(-d1)
    theta = term1 + term2
    vega = F * disc * pdf_d1 * np.sqrt(T)
    if option_type == "call":
        rho = - T * disc * (F * N(d1) - K * N(d2))
    else:
        rho = - T * disc * (K * N(-d2) - F * N(-d1))
    return {"delta": delta, "gamma": gamma, "theta": theta, "vega": vega, "rho": rho}

# ===================== 模块1：LSMC 最小二乘蒙特卡洛 美式期权 =====================
def lsmc_american_option(S0, K, T, r, sigma, q=0, option_type="call",
                         n_paths=50000, n_steps=50, seed=42):
    """
    Longstaff-Schwartz 美式期权MC定价
    回归基函数：1, S, S^2 (标准配置)
    """
    np.random.seed(seed)
    dt = T / n_steps
    discount = np.exp(-r * dt)
    # 生成GBM路径
    S = np.zeros((n_paths, n_steps + 1))
    S[:, 0] = S0
    for t in range(1, n_steps+1):
        z = np.random.normal(0, 1, n_paths)
        S[:, t] = S[:, t-1] * np.exp((r - q - 0.5*sigma**2)*dt + sigma*np.sqrt(dt)*z)
    # 到期收益
    if option_type == "call":
        payoff = np.maximum(S[:, -1] - K, 0.0)
    else:
        payoff = np.maximum(K - S[:, -1], 0.0)
    # 倒推
    for t in range(n_steps-1, 0, -1):
        St = S[:, t]
        itm_mask = payoff > 0
        if np.sum(itm_mask) < 5:
            payoff = payoff * discount
            continue
        # 回归：继续持有价值 ~ 当期S
        X = St[itm_mask]
        y = payoff[itm_mask] * discount
        # 基函数
        A = np.vstack([np.ones_like(X), X, X**2]).T
        beta = np.linalg.lstsq(A, y, rcond=None)[0]
        continuation = A @ beta
        # 立即行权价值
        if option_type == "call":
            exercise = np.maximum(X - K, 0.0)
        else:
            exercise = np.maximum(K - X, 0.0)
        # 选择：行权 or 继续持有
        payoff[itm_mask] = np.where(exercise > continuation, exercise, y)
        payoff[~itm_mask] = payoff[~itm_mask] * discount
    # 第一步折现
    price = np.mean(payoff) * discount
    return price

# ===================== 模块2：SVI-JW + 蝶式/价差无套利检验 =====================
def svi_raw(k, a, b, rho, m, sigma):
    return a + b * ( rho * (k - m) + np.sqrt( (k - m)**2 + sigma**2 ) )

def svi_jw_transform(a,b,rho,m,sigma):
    """SVI Raw -> SVI-JW参数：v_t, psi_t, p_t, c_t, tau_t"""
    w = a + b*sigma*np.sqrt(1-rho**2)
    v_t = w
    psi_t = (b*rho)/(2*np.sqrt(w))
    p_t = (b*(1-rho))/np.sqrt(w)
    c_t = (b*(1+rho))/np.sqrt(w)
    tau_t = b*sigma / np.sqrt(w)
    return {"v_t":v_t, "psi_t":psi_t, "p_t":p_t, "c_t":c_t, "tau_t":tau_t}

def svi_obj(params, k_arr, w_arr):
    a,b,rho,m,sig = params
    if b <= 0 or abs(rho)>=1 or sig <=0:
        return 1e12
    w_fit = svi_raw(k_arr, a,b,rho,m,sig)
    return np.sum((w_fit - w_arr)**2)

def fit_svi(F, K_arr, iv_arr, T, init_guess=None):
    k_arr = np.log(F / K_arr)
    w_arr = iv_arr**2 * T
    if init_guess is None:
        init_guess = [0.01, 0.1, -0.3, 0.0, 0.1]
    bounds = [(0.0,1.0),(1e-6,2.0),(-0.999,0.999),(-1.0,1.0),(1e-6,1.0)]
    res = minimize(svi_obj, init_guess, args=(k_arr,w_arr), bounds=bounds, method="L-BFGS-B")
    a,b,rho,m,sig = res.x
    def iv_fit(k):
        w = svi_raw(k,a,b,rho,m,sig)
        w = np.maximum(w,1e-12)
        return np.sqrt(w / T)
    return {"params":(a,b,rho,m,sig), "success":res.success, "iv_func":iv_fit, "k_arr":k_arr}

def butterfly_arbitrage_check(F, Ks, iv_func, T, num=200):
    """
    蝶式无套利检验：d2w/dk2 >=0
    返回 True=无套利；False=存在蝶式套利
    """
    k_test = np.linspace(np.log(F/Ks[-1]), np.log(F/Ks[0]), num)
    w = np.array([iv_func(k)**2 * T for k in k_test])
    # 二阶差分近似
    d2w = np.gradient(np.gradient(w, k_test), k_test)
    # 允许极小数值噪声
    has_arb = np.any(d2w < -1e-6)
    return not has_arb, k_test, d2w

def call_spread_arbitrage_check(F, Ks, iv_func, T):
    """看涨价差无套利：dC/dK <=0"""
    K_test = np.linspace(Ks[0], Ks[-1], 100)
    C = np.array([black_futures_option(F, k, T, 0.03, iv_func(np.log(F/k)), "call") for k in K_test])
    dCdk = np.gradient(C, K_test)
    has_arb = np.any(dCdk > 1e-6)
    return not has_arb, K_test, dCdk

# ===================== 模块3：触发式Delta对冲（Delta Band，阈值调仓） =====================
def delta_band_hedge_simulation(S0, K, T, r, sigma, q=0, option_type="call",
                                band=0.02, n_steps=252, fee_rate=0.0001, seed=42):
    """
    band: Delta阈值，例如0.02：偏离目标Delta超过±0.02才调仓
    卖出期权，对冲目标 = -delta
    """
    np.random.seed(seed)
    dt = T / n_steps
    S = np.zeros(n_steps+1)
    S[0] = S0
    cash = 0.0
    delta_hold = 0.0
    fee_total = 0.0
    V_hist = []
    delta_target_hist = []
    delta_hold_hist = []
    S_hist = []

    V0 = bs_option(S0, K, T, r, sigma, q, option_type)
    cash += V0

    for t in range(n_steps+1):
        tau = T - t*dt
        S_hist.append(S[t])
        if tau < 1e-8:
            V = np.maximum(S[t]-K,0) if option_type=="call" else np.maximum(K-S[t],0)
            V_hist.append(V)
            delta_target_hist.append(0.0)
            delta_hold_hist.append(delta_hold)
            break
        V = bs_option(S[t], K, tau, r, sigma, q, option_type)
        g = bs_greeks(S[t], K, tau, r, sigma, q, option_type)
        delta_target = -g["delta"]
        V_hist.append(V)
        delta_target_hist.append(delta_target)
        delta_hold_hist.append(delta_hold)
        # 触发条件：偏离超过band
        if abs(delta_hold - delta_target) > band:
            trade = delta_target - delta_hold
            fee = abs(trade)*S[t]*fee_rate
            fee_total += fee
            cash -= trade * S[t] + fee
            delta_hold = delta_target
        cash *= np.exp(r*dt)
        # 生成下一期价格
        z = np.random.normal()
        S[t+1] = S[t] * np.exp((r - q - 0.5*sigma**2)*dt + sigma*np.sqrt(dt)*z)

    final_pnl = cash + delta_hold * S[-1] - V_hist[-1]
    out = {
        "S": np.array(S_hist),
        "V": np.array(V_hist),
        "delta_target": np.array(delta_target_hist),
        "delta_hold": np.array(delta_hold_hist),
        "fee_total": fee_total,
        "final_pnl": final_pnl
    }
    return out

# ===================== 模块4：期权组合PnL仿真 + 滚动希腊字母监控 =====================
def portfolio_greeks(underlying, positions, r, q=0):
    """
    positions: list dict: {"type":"call"/"put", "K":K, "T":T, "sigma":iv, "w":权重/手数}
    """
    total = {"delta":0.0, "gamma":0.0, "theta":0.0, "vega":0.0, "rho":0.0, "pv":0.0}
    for p in positions:
        opt_type = p["type"]
        K = p["K"]
        T = p["T"]
        sigma = p["sigma"]
        w = p["w"]
        g = bs_greeks(underlying, K, T, r, sigma, q, opt_type)
        v = bs_option(underlying, K, T, r, sigma, q, opt_type)
        total["delta"] += w * g["delta"]
        total["gamma"] += w * g["gamma"]
        total["theta"] += w * g["theta"]
        total["vega"] += w * g["vega"]
        total["rho"] += w * g["rho"]
        total["pv"] += w * v
    return total

def portfolio_simulation(S0, positions, T_total, r, sigma_underlying, q=0, n_steps=252, seed=42):
    np.random.seed(seed)
    dt = T_total / n_steps
    S = np.zeros(n_steps+1)
    S[0] = S0
    greeks_roll = []
    pv_roll = []
    for t in range(n_steps+1):
        tau_remain = T_total - t*dt
        # 每一个期权剩余时间同步衰减
        pos_rolling = []
        for p in positions:
            new_tau = p["T"] - t*dt
            if new_tau <= 0:
                continue
            pos_rolling.append({"type":p["type"], "K":p["K"], "T":new_tau, "sigma":p["sigma"], "w":p["w"]})
        g = portfolio_greeks(S[t], pos_rolling, r, q)
        greeks_roll.append(g)
        pv_roll.append(g["pv"])
        if t == n_steps:
            break
        z = np.random.normal()
        S[t+1] = S[t] * np.exp((r - q - 0.5*sigma_underlying**2)*dt + sigma_underlying*np.sqrt(dt)*z)
    return {"S":S, "pv":np.array(pv_roll), "greeks":greeks_roll}

# ===================== 模块5：商品期货（生猪）便利收益率 + 季节性模拟 =====================
def commodity_seasonal_future_path(S0, T, r, u, y0, sigma, season_amp=0.08, season_period=1.0, n_steps=252, seed=42):
    """
    F = S * exp( (r + u - \(y(t)\)) * t )
    u:存储成本（连续）
    \(y(t)\):便利收益率，带季节性周期
    season_amp:便利收益季节性振幅（生猪一年周期）
    season_period:周期，单位年，生猪=1
    """
    np.random.seed(seed)
    dt = T / n_steps
    F = np.zeros(n_steps+1)
    F[0] = S0
    for t in range(1, n_steps+1):
        t_year = t * dt
        # 季节性便利收益率：正弦周期
        y_t = y0 + season_amp * np.sin(2 * np.pi * t_year / season_period)
        drift = (r + u - y_t - 0.5 * sigma**2)*dt
        z = np.random.normal()
        F[t] = F[t-1] * np.exp(drift + sigma * np.sqrt(dt)*z)
    return F
```

## 模块要点说明（贴合你的商品期权实盘）
### 1. LSMC 最小二乘蒙特卡洛
- 逻辑：从到期向前倒推，**回归估计继续持有价值**，和立即行权价值对比；美式Put最容易体现提前行权溢价
- 基函数：`1, S, S²`；想要精度更高可以加 $S^3, \ln S$
- 注意：LSMC是有偏估计，路径数、步数越大越稳定；适合没有闭式解的美式、百慕大、路径依赖美式期权

### 2. SVI-JW + 无套利检验
- SVI-JW：把Raw SVI映射成**有直观含义的参数**：ATM波动率、偏斜、左右翼曲率，做跨到期插值、曲面构建行业标准
- 蝶式套利检验：核心条件 $\partial^2 w/\partial k^2 \ge 0$；二阶方差曲率不能为负，负=蝶式套利
- 看涨价差检验：看涨期权价格随K单调递减，否则存在价差套利
> 实盘用途：拿到交易所IV数据，拟合后自动清洗掉套利微笑，给做市/定价用

### 3. Delta Band 触发式对冲（Delta区间对冲）
- 不再固定每日调仓，**Delta偏离目标超过阈值band才调仓**，大幅减少交易次数、手续费
- 权衡：band越大，调仓越少手续费越低，但**Gamma对冲误差变大、PnL波动上升**
- 国内商品期权卖方最常用：band=0.01~0.05，根据手续费、波动率选择

### 4. 期权组合PnL仿真 + 滚动希腊监控
- 支持多期权多头/空头组合，自动随时间衰减剩余到期T
- 每一步输出组合总Delta/Gamma/Vega/Theta/Rho，相当于一个轻量版**组合风险看板**
- 扩展：可以叠加不同行权价、不同到期日、期货期权（把bs换成black）

### 5. 商品期货（生猪）便利收益率+季节性模拟
公式：
$$F(t)=S_0 \exp\left( (r+u-\(y(t)\))t \right)$$
$$\(y(t)\)=y_0 + A \sin(2\pi t / P)$$
- $u$：存储成本；$\(y(t)\)$：时变便利收益率；生猪季节性极强（周期≈1年）
- 适合：生猪、玉米这类农产品，复现淡季/旺季升贴水变化；可以嵌入到你CTA、商品期权定价里
