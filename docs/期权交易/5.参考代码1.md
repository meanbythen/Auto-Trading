```python
import numpy as np
from scipy.stats import norm
import matplotlib.pyplot as plt

# ====================== 1. 工具函数：正态CDF、PDF ======================
N = norm.cdf
n_pdf = norm.pdf


# ====================== 2. Black-Scholes Merton 欧式期权（带连续股息q） ======================
def bs_option(S, K, T, r, sigma, q=0.0, option_type="call"):
    """
    B-S-M 欧式期权定价
    S: 标的现价
    K: 行权价
    T: 剩余到期时间(年)
    r: 无风险连续复利
    sigma: 波动率
    q: 连续股息率
    option_type: "call" / "put"
    """
    d1 = (np.log(S / K) + (r - q + 0.5 * sigma ** 2) * T) / (sigma * np.sqrt(T))
    d2 = d1 - sigma * np.sqrt(T)
    if option_type == "call":
        price = S * np.exp(-q * T) * N(d1) - K * np.exp(-r * T) * N(d2)
    else:
        price = K * np.exp(-r * T) * N(-d2) - S * np.exp(-q * T) * N(-d1)
    return price


def bs_greeks(S, K, T, r, sigma, q=0.0, option_type="call"):
    """B-S 希腊字母：Delta, Gamma, Theta, Vega, Rho"""
    d1 = (np.log(S / K) + (r - q + 0.5 * sigma ** 2) * T) / (sigma * np.sqrt(T))
    d2 = d1 - sigma * np.sqrt(T)
    pdf_d1 = n_pdf(d1)

    # Delta
    if option_type == "call":
        delta = np.exp(-q * T) * N(d1)
    else:
        delta = np.exp(-q * T) * (N(d1) - 1)

    # Gamma (call/put Gamma相同)
    gamma = np.exp(-q * T) * pdf_d1 / (S * sigma * np.sqrt(T))

    # Theta (年化，注意：一般日报Theta要除以365)
    term1 = - S * np.exp(-q * T) * pdf_d1 * sigma / (2 * np.sqrt(T))
    if option_type == "call":
        term2 = q * S * np.exp(-q * T) * N(d1)
        term3 = - r * K * np.exp(-r * T) * N(d2)
    else:
        term2 = - q * S * np.exp(-q * T) * N(-d1)
        term3 = r * K * np.exp(-r * T) * N(-d2)
    theta = term1 + term2 + term3

    # Vega（波动率变动1%的敏感度，这里是σ变动1个单位）
    vega = S * np.exp(-q * T) * pdf_d1 * np.sqrt(T)

    # Rho
    if option_type == "call":
        rho = K * T * np.exp(-r * T) * N(d2)
    else:
        rho = -K * T * np.exp(-r * T) * N(-d2)

    return {
        "delta": delta,
        "gamma": gamma,
        "theta": theta,
        "vega": vega,
        "rho": rho
    }


# ====================== 3. Black 期货期权模型（Hull第15章，国内商品期权用这个！） ======================
def black_futures_option(F, K, T, r, sigma, option_type="call"):
    """
    Black 期货期权定价
    F: 期货价格
    K: 行权价
    T: 到期时间(年)
    r: 无风险连续复利
    sigma: 期货波动率
    """
    d1 = (np.log(F / K) + 0.5 * sigma ** 2 * T) / (sigma * np.sqrt(T))
    d2 = d1 - sigma * np.sqrt(T)
    disc = np.exp(-r * T)
    if option_type == "call":
        price = disc * (F * N(d1) - K * N(d2))
    else:
        price = disc * (K * N(-d2) - F * N(-d1))
    return price


def black_greeks(F, K, T, r, sigma, option_type="call"):
    """Black期货期权希腊字母"""
    d1 = (np.log(F / K) + 0.5 * sigma ** 2 * T) / (sigma * np.sqrt(T))
    d2 = d1 - sigma * np.sqrt(T)
    pdf_d1 = n_pdf(d1)
    disc = np.exp(-r * T)

    if option_type == "call":
        delta = disc * N(d1)
    else:
        delta = disc * (N(d1) - 1)

    gamma = disc * pdf_d1 / (F * sigma * np.sqrt(T))
    term1 = - F * disc * pdf_d1 * sigma / (2 * np.sqrt(T))
    if option_type == "call":
        term2 = r * F * disc * N(d1) - r * K * disc * N(d2)
    else:
        term2 = r * K * disc * N(-d2) - r * F * disc * N(-d1)
    theta = term1 + term2
    vega = F * disc * pdf_d1 * np.sqrt(T)
    if option_type == "call":
        rho = - T * disc * (F * N(d1) - K * N(d2))
    else:
        rho = - T * disc * (K * N(-d2) - F * N(-d1))

    return {
        "delta": delta,
        "gamma": gamma,
        "theta": theta,
        "vega": vega,
        "rho": rho
    }


# ====================== 4. CRR 二叉树（支持美式/欧式，股票标的） ======================
def crr_binomial_tree(S, K, T, r, sigma, N, q=0.0, option_type="call", american=False):
    """
    CRR Cox-Ross-Rubinstein 二叉树
    N: 步数
    american=True → 美式期权
    """
    dt = T / N
    u = np.exp(sigma * np.sqrt(dt))
    d = 1 / u
    p = (np.exp((r - q) * dt) - d) / (u - d)
    disc = np.exp(-r * dt)

    # 叶子节点 payoff
    ST = S * (u ** np.arange(N, -1, -1)) * (d ** np.arange(0, N + 1))
    if option_type == "call":
        payoff = np.maximum(ST - K, 0)
    else:
        payoff = np.maximum(K - ST, 0)

    # 反向递推
    for _ in range(N):
        payoff = disc * (p * payoff[:-1] + (1 - p) * payoff[1:])
        if american:
            # 美式：比较 立刻行权价值 vs 持有价值
            S_t = S * (u ** np.arange(len(payoff)-1, -1, -1)) * (d ** np.arange(0, len(payoff)))
            if option_type == "call":
                exercise = np.maximum(S_t - K, 0)
            else:
                exercise = np.maximum(K - S_t, 0)
            payoff = np.maximum(payoff, exercise)
    return payoff[0]


# ====================== 5. VaR 模块：方差协方差法 + 历史模拟法 ======================
def var_variance_covariance(portfolio_val, mu, sigma, T, conf=0.95):
    """
    方差-协方差 VaR（正态假设）
    mu: 组合年化期望收益
    sigma: 组合年化波动率
    T: 持有期(年)
    conf: 置信度 0.95 / 0.99
    """
    z = norm.ppf(1 - conf)
    var = portfolio_val * (mu * T + z * sigma * np.sqrt(T))
    es = portfolio_val * (mu * T - sigma * np.sqrt(T) * norm.pdf(z) / (1 - conf))
    return {"VaR": -var, "ES": -es}


def var_historical_simulation(returns, portfolio_val, conf=0.95):
    """
    历史模拟法VaR
    returns: 收益率序列
    """
    loss = -portfolio_val * returns
    var = np.quantile(loss, conf)
    es = loss[loss >= var].mean()
    return {"VaR": var, "ES": es}
```

## 代码说明&实务注意点（贴合你的国内商品期权）
1. **Black期货期权是你最常用**：生猪、豆粕、玉米期权都是期货期权，不用B-S，用Black；公式等价 Hull P340。
2. 时间单位：**T 统一为年**；交易日折算：`1/252`，自然日 `1/365`，自己按需切换。
3. 希腊字母单位：
   - Theta：**年化Theta**，单日Theta = theta / 252
   - Vega：σ变动**1（100%）**的价格变化；如果要1个百分点Vega（0.01），`vega * 0.01`
4. CRR二叉树：美式put可以提前行权，代码里做了行权判断；步数N越大越贴近B-S，N=100~300足够。
5. VaR：
   - 方差协方差法：正态假设，低估尾部风险，适合快速监控
   - 历史模拟：无分布假设，实盘风控更推荐；ES（CVaR）一同输出
6. 扩展方向你后面可以自己叠加：
   - IV反解函数（给期权价格，牛顿迭代求隐含波动率）
   - 波动率微笑曲面拟合
   - 蒙特卡洛欧式/亚洲期权
   - 滚动WFA回测框架、Delta动态对冲仿真

```python
def implied_vol_black(price, F, K, T, r, option_type="call", tol=1e-6, max_iter=100):
    sigma = 0.2
    for _ in range(max_iter):
        p = black_futures_option(F, K, T, r, sigma, option_type)
        g = black_greeks(F, K, T, r, sigma, option_type)
        vega = g["vega"]
        if abs(vega) < 1e-12:
            break
        dsigma = (price - p) / vega
        sigma += dsigma
        if abs(dsigma) < tol:
            break
    return sigma

def implied_vol_bs(price, S, K, T, r, q=0, option_type="call", tol=1e-6, max_iter=100):
    sigma = 0.2
    for _ in range(max_iter):
        p = bs_option(S, K, T, r, sigma, q, option_type)
        g = bs_greeks(S, K, T, r, sigma, q, option_type)
        vega = g["vega"]
        if abs(vega) < 1e-12:
            break
        dsigma = (price - p) / vega
        sigma += dsigma
        if abs(dsigma) < tol:
            break
    return sigma
```
