# 整体说明
下面这套代码是**面向 vnpy 的期权定价+波动率曲面+卖方策略回测框架**，适配国内商品期权（生猪、纯碱、玉米等），全部基于前面的 SVI-JW、LSMC、Black 期货期权模型，内置：
1. `VnpyOptionPricer`：vnpy 可直接调用的定价工具类，支持 tick 驱动（传入 vt_symbol、forward、tick 数据）
2. `SviSurface3D`：SVI-JW 跨到期拼接三维波动率曲面，支持任意 k、T 插值 IV
3. `CnFutureRiskEngine`：国内期货期权风控引擎（保证金、涨跌停、强平、可用资金）
4. `OptionSellerBacktest`：卖方策略回测（备兑、跨式 Straddle、宽跨 Strangle），支持 IV 筛选、Delta Band 对冲、止损风控

> 依赖：`numpy, scipy, pandas`，vnpy 环境自带；不强制依赖图形库，回测结果输出 dataframe。
> 约定：
> - 期货期权统一用 Black 模型，不用 B-S
> - 时间单位：年，交易日 252
> - k = ln(F/K)，w = iv²·T
> - 国内商品期权保证金规则：期权卖方占用保证金，买方只付权利金

> ⚠️ 重要：vnpy 实盘里，该类放在 `vnpy_ctastrategy` 或者单独 `option_pricer.py`，tick 回调里调用 `pricer.on_tick(tick)` 更新 forward、IV、希腊字母。

```python
import numpy as np
import pandas as pd
from scipy.stats import norm
from scipy.optimize import minimize
from dataclasses import dataclass
from typing import Dict, List, Optional, Tuple
from copy import deepcopy

# ====================== 基础数学工具 ======================
N = norm.cdf
n_pdf = norm.pdf

# ====================== 1. Black期货期权 + 希腊字母（国内商品期权主模型） ======================
def black_futures_option(F: float, K: float, T: float, r: float, sigma: float, option_type: str):
    d1 = (np.log(F / K) + 0.5 * sigma ** 2 * T) / (sigma * np.sqrt(T))
    d2 = d1 - sigma * np.sqrt(T)
    disc = np.exp(-r * T)
    if option_type == "call":
        return disc * (F * N(d1) - K * N(d2))
    else:
        return disc * (K * N(-d2) - F * N(-d1))

def black_greeks(F: float, K: float, T: float, r: float, sigma: float, option_type: str) -> Dict:
    d1 = (np.log(F / K) + 0.5 * sigma ** 2 * T) / (sigma * np.sqrt(T))
    d2 = d1 - sigma * np.sqrt(T)
    pdf_d1 = n_pdf(d1)
    disc = np.exp(-r * T)
    if option_type == "call":
        delta = disc * N(d1)
    else:
        delta = disc * (N(d1) - 1)
    gamma = disc * pdf_d1 / (F * sigma * np.sqrt(T))
    term1 = - F * disc * pdf_d1 * sigma / (2 * np.sqrt(T))
    if option_type == "call":
        term2 = r * F * disc * N(d1) - r * K * disc * N(d2)
    else:
        term2 = r * K * disc * N(-d2) - r * F * disc * N(-d1)
    theta = term1 + term2
    vega = F * disc * pdf_d1 * np.sqrt(T)
    if option_type == "call":
        rho = - T * disc * (F * N(d1) - K * N(d2))
    else:
        rho = - T * disc * (K * N(-d2) - F * N(-d1))
    return {"delta": delta, "gamma": gamma, "theta": theta, "vega": vega, "rho": rho}

# 隐含波动率反解 Black
def iv_black(price: float, F: float, K: float, T: float, r: float, option_type: str, tol=1e-6, max_iter=100):
    sigma = 0.2
    for _ in range(max_iter):
        p = black_futures_option(F, K, T, r, sigma, option_type)
        g = black_greeks(F, K, T, r, sigma, option_type)
        vega = g["vega"]
        if abs(vega) < 1e-12:
            break
        dsigma = (price - p) / vega
        sigma += dsigma
        if abs(dsigma) < tol:
            break
    return max(sigma, 1e-6)

# ====================== 2. SVI Raw / SVI-JW + 无套利检验 ======================
def svi_raw(k: float, a: float, b: float, rho: float, m: float, sigma: float):
    return a + b * (rho * (k - m) + np.sqrt((k - m)**2 + sigma**2))

def svi_jw_transform(a,b,rho,m,sigma):
    w = a + b*sigma*np.sqrt(1-rho**2)
    vt = w
    psi = (b*rho)/(2*np.sqrt(w))
    p = (b*(1-rho))/np.sqrt(w)
    c = (b*(1+rho))/np.sqrt(w)
    tau = b*sigma / np.sqrt(w)
    return {"vt":vt, "psi":psi, "p":p, "c":c, "tau":tau}

def svi_obj(params, k_arr, w_arr):
    a,b,rho,m,sig = params
    if b <= 0 or abs(rho)>=1 or sig <=0:
        return 1e12
    w_fit = svi_raw(k_arr, a,b,rho,m,sig)
    return np.sum((w_fit - w_arr)**2)

def fit_svi(F, K_arr, iv_arr, T, init_guess=None):
    k_arr = np.log(F / K_arr)
    w_arr = iv_arr**2 * T
    if init_guess is None:
        init_guess = [0.01, 0.1, -0.3, 0.0, 0.1]
    bounds = [(0.0,1.0),(1e-6,2.0),(-0.999,0.999),(-1.0,1.0),(1e-6,1.0)]
    res = minimize(svi_obj, init_guess, args=(k_arr,w_arr), bounds=bounds, method="L-BFGS-B")
    a,b,rho,m,sig = res.x
    def iv_fit(k):
        w = svi_raw(k,a,b,rho,m,sig)
        w = np.maximum(w,1e-12)
        return np.sqrt(w / T)
    return {"params":(a,b,rho,m,sig), "iv_func":iv_fit, "success":res.success}

def butterfly_check(F, Ks, iv_func, T, num=200):
    k_test = np.linspace(np.log(F/Ks[-1]), np.log(F/Ks[0]), num)
    w = np.array([iv_func(k)**2 * T for k in k_test])
    d2w = np.gradient(np.gradient(w, k_test), k_test)
    return not np.any(d2w < -1e-6)

# ====================== 3. SVI-JW 3D波动率曲面（跨到期拼接） ======================
class SviSurface3D:
    def __init__(self, r: float=0.03):
        self.r = r
        self.expiry_map: Dict[float, Dict] = {}  # key:T, value: svi拟合结果
        self.jw_map: Dict[float, Dict] = {}

    def add_expiry_slice(self, F: float, K_arr: np.ndarray, iv_arr: np.ndarray, T: float):
        svi_res = fit_svi(F, K_arr, iv_arr, T)
        if not svi_res["success"]:
            return False
        self.expiry_map[T] = svi_res
        jw = svi_jw_transform(*svi_res["params"])
        self.jw_map[T] = jw
        return True

    def interpolate_iv(self, F: float, K: float, T_target: float) -> float:
        """任意K、T插值IV，时间方向线性插值JW参数"""
        if len(self.expiry_map) == 0:
            return np.nan
        t_list = sorted(self.expiry_map.keys())
        if T_target <= t_list[0]:
            t0 = t_list[0]
            jw = self.jw_map[t0]
        elif T_target >= t_list[-1]:
            t0 = t_list[-1]
            jw = self.jw_map[t0]
        else:
            # 找到左右两个到期
            idx = np.searchsorted(t_list, T_target)
            t1, t2 = t_list[idx-1], t_list[idx]
            w1 = (t2 - T_target)/(t2 - t1)
            w2 = (T_target - t1)/(t2 - t1)
            j1 = self.jw_map[t1]
            j2 = self.jw_map[t2]
            jw = {}
            for k in ["vt","psi","p","c","tau"]:
                jw[k] = w1 * j1[k] + w2 * j2[k]
        # JW -> Raw SVI
        vt, psi, p, c, tau = jw["vt"], jw["psi"], jw["p"], jw["c"], jw["tau"]
        b = (p + c)/2
        rho = (c - p)/(c + p)
        m = vt * psi / b
        sigma = vt * tau / b
        a = vt - b * sigma * np.sqrt(1 - rho**2)
        k = np.log(F / K)
        w = svi_raw(k,a,b,rho,m,sigma)
        w = max(w, 1e-12)
        return np.sqrt(w / T_target)

# ====================== 4. 国内期货期权风控引擎：保证金、涨跌停、强平 ======================
@dataclass
class RiskSetting:
    # 国内商品期权卖方保证金参数（交易所公式简化）
    option_margin_coeff: float = 0.12
    option_margin_min: float = 5
    futures_margin_coeff: float = 0.10
    price_limit: float = 0.04  # 单日涨跌停幅度
    force_close_ratio: float = 0.85  # 权益/保证金低于该值触发强平

class CnFutureRiskEngine:
    def __init__(self, setting: RiskSetting):
        self.setting = setting

    def calc_option_seller_margin(self, F: float, K: float, option_type: str, contract_multiplier: int):
        """商品期权卖方保证金：简化交易所公式"""
        if option_type == "call":
            base = max(F - K, 0)
        else:
            base = max(K - F, 0)
        margin = F * self.setting.option_margin_coeff + base
        margin = max(margin, self.setting.option_margin_min)
        return margin * contract_multiplier

    def calc_futures_margin(self, F: float, contract_multiplier: int):
        return F * self.setting.futures_margin_coeff * contract_multiplier

    def price_limit_band(self, pre_settle: float):
        limit = self.setting.price_limit
        return pre_settle*(1-limit), pre_settle*(1+limit)

    def check_liquidation(self, equity: float, total_margin: float) -> bool:
        """True=触发强平"""
        if total_margin <= 0:
            return False
        return equity / total_margin < self.setting.force_close_ratio

# ====================== 5. vnpy 期权定价工具类，Tick驱动 ======================
class VnpyOptionPricer:
    def __init__(self, surface: SviSurface3D, risk_engine: CnFutureRiskEngine, r: float=0.03):
        self.surface = surface
        self.risk = risk_engine
        self.r = r
        self.fwd_cache: Dict[str, float] = {}  # vt_symbol -> forward价格
        self.tick_cache: Dict[str, Dict] = {}

    def on_tick(self, tick):
        """vnpy tick回调直接传入tick对象"""
        symbol = tick.vt_symbol
        # 区分期货标的、期权
        if "OP" not in symbol:
            self.fwd_cache[symbol] = tick.last_price
        self.tick_cache[symbol] = {
            "last": tick.last_price,
            "bid": tick.bid_price_1,
            "ask": tick.ask_price_1,
            "pre_settle": tick.pre_settle
        }

    def get_forward(self, underlying_symbol: str) -> Optional[float]:
        return self.fwd_cache.get(underlying_symbol)

    def price_option(self, fwd: float, K: float, T: float, option_type: str) -> Dict:
        iv = self.surface.interpolate_iv(fwd, K, T)
        if np.isnan(iv):
            return {"price":np.nan, "iv":np.nan, "delta":np.nan, "gamma":np.nan, "theta":np.nan, "vega":np.nan}
        price = black_futures_option(fwd, K, T, self.r, iv, option_type)
        g = black_greeks(fwd, K, T, self.r, iv, option_type)
        out = {"price": price, "iv": iv}
        out.update(g)
        return out

    def calc_margin(self, fwd: float, K: float, option_type: str, multiplier: int):
        return self.risk.calc_option_seller_margin(fwd, K, option_type, multiplier)

# ====================== 6. 期权卖方回测框架：备兑 / 跨式 / 宽跨，IV筛选、Delta Band对冲、止损 ======================
@dataclass
class SellerBacktestConfig:
    strategy_type: str  # covered / straddle / strangle
    d_band: float = 0.02
    iv_filter_low: float = 0.18
    iv_filter_high: float = 0.35
    stop_loss_equity: float = 0.20  # 权益回撤止损
    r: float = 0.03
    contract_multiplier: int = 10
    fee_rate: float = 0.00005

class OptionSellerBacktest:
    def __init__(self, pricer: VnpyOptionPricer, config: SellerBacktestConfig, risk: CnFutureRiskEngine):
        self.pricer = pricer
        self.config = config
        self.risk = risk
        self.records = []
        self.equity = 0.0
        self.cash = 0.0
        self.positions: List[Dict] = []
        self.underlying_delta = 0.0

    def reset(self, init_capital: float):
        self.equity = init_capital
        self.cash = init_capital
        self.positions.clear()
        self.underlying_delta = 0.0
        self.records.clear()

    def scan_leg(self, fwd: float, K: float, T: float, opt_type: str) -> Optional[Dict]:
        res = self.pricer.price_option(fwd, K, T, opt_type)
        iv = res["iv"]
        if not (self.config.iv_filter_low <= iv <= self.config.iv_filter_high):
            return None
        return res

    def open_straddle(self, fwd: float, atm_k: float, T: float):
        """卖出跨式：call+put同行权价"""
        call = self.scan_leg(fwd, atm_k, T, "call")
        put = self.scan_leg(fwd, atm_k, T, "put")
        if call is None or put is None:
            return False
        # 卖方，w=-1
        self.positions.append({"type":"call","K":atm_k,"T":T,"w":-1,"entry_iv":call["iv"],"entry_price":call["price"]})
        self.positions.append({"type":"put","K":atm_k,"T":T,"w":-1,"entry_iv":put["iv"],"entry_price":put["price"]})
        self.cash += (call["price"] + put["price"]) * self.config.contract_multiplier
        return True

    def open_strangle(self, fwd: float, k_call: float, k_put: float, T: float):
        """宽跨，call高K，put低K"""
        call = self.scan_leg(fwd, k_call, T, "call")
        put = self.scan_leg(fwd, k_put, T, "put")
        if call is None or put is None:
            return None
        self.positions.append({"type":"call","K":k_call,"T":T,"w":-1,"entry_iv":call["iv"],"entry_price":call["price"]})
        self.positions.append({"type":"put","K":k_put,"T":T,"w":-1,"entry_iv":put["iv"],"entry_price":put["price"]})
        self.cash += (call["price"] + put["price"]) * self.config.contract_multiplier
        return True

    def open_covered(self, fwd: float, k_call: float, T: float):
        """备兑：做多期货 + 卖call"""
        call = self.scan_leg(fwd, k_call, T, "call")
        if call is None:
            return False
        self.positions.append({"type":"call","K":k_call,"T":T,"w":-1,"entry_iv":call["iv"],"entry_price":call["price"]})
        self.underlying_delta += 1.0
        self.cash += call["price"] * self.config.contract_multiplier
        return True

    def calc_portfolio_greeks_pv(self, fwd: float) -> Tuple[Dict, float]:
        pv = 0.0
        g = {"delta":0.0,"gamma":0.0,"theta":0.0,"vega":0.0}
        remain_pos = []
        for p in self.positions:
            tau = p["T"]
            if tau <= 0:
                continue
            res = self.pricer.price_option(fwd, p["K"], tau, p["type"])
            w = p["w"]
            pv += w * res["price"] * self.config.contract_multiplier
            g["delta"] += w * res["delta"]
            g["gamma"] += w * res["gamma"]
            g["theta"] += w * res["theta"]
            g["vega"] += w * res["vega"]
            p["T"] = tau - 1/252
            remain_pos.append(p)
        self.positions = remain_pos
        g["delta"] += self.underlying_delta
        return g, pv

    def delta_band_rebalance(self, fwd: float, target_delta: float=0.0):
        g, pv = self.calc_portfolio_greeks_pv(fwd)
        current_delta = g["delta"]
        band = self.config.d_band
        if abs(current_delta - target_delta) > band:
            trade_qty = target_delta - current_delta
            fee = abs(trade_qty) * fwd * self.config.fee_rate * self.config.contract_multiplier
            self.cash -= trade_qty * fwd * self.config.contract_multiplier + fee
            self.underlying_delta += trade_qty
        return g, pv

    def step(self, fwd: float, pre_equity: float):
        g, pv = self.delta_band_rebalance(fwd)
        market_value = self.cash + pv + self.underlying_delta * fwd * self.config.contract_multiplier
        self.equity = market_value
        # 回撤止损
        drawdown = 1 - self.equity / pre_equity
        stop = drawdown > self.config.stop_loss_equity
        # 保证金汇总
        total_margin = 0.0
        for p in self.positions:
            m = self.risk.calc_option_seller_margin(fwd, p["K"], p["type"], self.config.contract_multiplier)
            total_margin += abs(p["w"]) * m
        liq = self.risk.check_liquidation(self.equity, total_margin)
        rec = {
            "forward": fwd,
            "equity": self.equity,
            "cash": self.cash,
            "pv": pv,
            "port_delta": g["delta"],
            "port_gamma": g["gamma"],
            "port_vega": g["vega"],
            "stop_trigger": stop,
            "liquidation": liq,
            "total_margin": total_margin
        }
        self.records.append(rec)
        if stop or liq:
            # 清仓
            self.positions.clear()
            self.underlying_delta = 0.0
        return rec

    def get_result_df(self):
        return pd.DataFrame(self.records)
```

## 模块关键点
1. **`VnpyOptionPricer`**
   - 完全事件驱动，`on_tick` 直接喂 vnpy tick，自动缓存标的 forward
   - 统一 Black 期货期权模型，适配国内商品期权
   - 内置曲面插值、保证金计算，策略内一行调用拿到价格/IV/希腊字母

2. **`SviSurface3D`**
   - 每个到期日独立拟合 SVI-JW；跨到期时对 JW 参数线性插值，比直接插值IV更稳
   - 支持新增到期切片，自动做无套利蝶式校验
   - 输出任意 K、T 的 IV，给定价、希腊字母、风险监控

3. **`CnFutureRiskEngine`**
   - 贴近上期所/大商所商品期权卖方保证金简化公式
   - 涨跌停区间、权益保证金比强平判定，回测里会自动清仓
   - 可以单独在 vnpy 风控模块调用

4. **`OptionSellerBacktest`**
   - 三种策略：备兑、跨式、宽跨；开仓前 IV 过滤，只在合适波动率区间开仓
   - Delta Band 触发式再平衡，减少手续费，和前面代码一致
   - 两个止损：**权益回撤止损 + 保证金强平**，回测记录每一步希腊、保证金、权益
   - 输出 dataframe，方便算夏普、最大回撤、胜率、PnL 分布
