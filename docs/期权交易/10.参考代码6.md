下面是**增强完整版**，在上面整套框架基础上一次性补齐你列出的5项需求：
1. TQData 批量加载K线、IV截面、自动拟合SVI曲面
2. 大商所/郑商所/上期所差异化分组保证金
3. 盘口撮合引擎（bid/ask成交、滑点、买卖价差），不再用last_price成交
4. Greeks看板、vnpy UI日志输出适配
5. 期权Roll滚动换仓逻辑（到期前自动移仓到下一个到期）

> 前置依赖：`tqdata`、`tqsdk`、`numpy,scipy,pandas`；vnpy环境直接使用
> 重要约定：
> - 商品期权一律 Black 期货期权模型
> - TQData 取截面：同一标的、同一交易日，所有期权链 K、T、bid/ask、iv
> - 撮合：卖方开仓 → 以bid成交；卖方平仓 → 以ask成交；支持固定滑点
> - 保证金：区分 DCE/CZCE/SHFE 三套系数
> - Roll：距离到期小于roll_days，自动平当前腿，在下一个到期同delta位置开新腿

import numpy as np

# ========== 基础函数 ==========
N = norm.cdf
n_pdf = norm.pdf

def black_futures_option(F: float, K: float, T: float, r: float, sigma: float, option_type: str):
    d1 = (np.log(F / K) + 0.5 * sigma ** 2 * T) / (sigma * np.sqrt(T))
    d2 = d1 - sigma * np.sqrt(T)
    disc = np.exp(-r * T)
    if option_type == "call":
        return disc * (F * N(d1) - K * N(d2))
    else:
        return disc * (K * N(-d2) - F * N(-d1))

def black_greeks(F: float, K: float, T: float, r: float, sigma: float, option_type: str) -> Dict:
    d1 = (np.log(F / K) + 0.5 * sigma ** 2 * T) / (sigma * np.sqrt(T))
    d2 = d1 - sigma * np.sqrt(T)
    pdf_d1 = n_pdf(d1)
    disc = np.exp(-r * T)
    if option_type == "call":
        delta = disc * N(d1)
    else:
        delta = disc * (N(d1) - 1)
    gamma = disc * pdf_d1 / (F * sigma * np.sqrt(T))
    term1 = - F * disc * pdf_d1 * sigma / (2 * np.sqrt(T))
    if option_type == "call":
        term2 = r * F * disc * N(d1) - r * K * disc * N(d2)
    else:
        term2 = r * K * disc * N(-d2) - r * F * disc * N(-d1)
    theta = term1 + term2
    vega = F * disc * pdf_d1 * np.sqrt(T)
    rho = -T * disc * black_futures_option(F,K,T,r,sigma,option_type)
    return {"delta": delta, "gamma": gamma, "theta": theta, "vega": vega, "rho": rho}

def iv_black(price: float, F: float, K: float, T: float, r: float, option_type: str, tol=1e-6, max_iter=100):
    sigma = 0.2
    for _ in range(max_iter):
        p = black_futures_option(F, K, T, r, sigma, option_type)
        g = black_greeks(F, K, T, r, sigma, option_type)
        vega = g["vega"]
        if abs(vega) < 1e-12:
            break
        dsigma = (price - p) / vega
        sigma += dsigma
        if abs(dsigma) < tol:
            break
    return max(sigma, 1e-6)

# ========== SVI / SVI-JW + 无套利检验 ==========
def svi_raw(k: float, a: float, b: float, rho: float, m: float, sigma: float):
    return a + b * (rho * (k - m) + np.sqrt((k - m)**2 + sigma**2))

def svi_jw_transform(a,b,rho,m,sigma):
    w = a + b*sigma*np.sqrt(1-rho**2)
    vt = w
    psi = (b*rho)/(2*np.sqrt(w))
    p = (b*(1-rho))/np.sqrt(w)
    c = (b*(1+rho))/np.sqrt(w)
    tau = b*sigma / np.sqrt(w)
    return {"vt":vt, "psi":psi, "p":p, "c":c, "tau":tau}

def svi_obj(params, k_arr, w_arr):
    a,b,rho,m,sig = params
    if b <= 0 or abs(rho)>=1 or sig <=0:
        return 1e12
    w_fit = svi_raw(k_arr, a,b,rho,m,sig)
    return np.sum((w_fit - w_arr)**2)

def fit_svi(F, K_arr, iv_arr, T, init_guess=None):
    k_arr = np.log(F / K_arr)
    w_arr = iv_arr**2 * T
    if init_guess is None:
        init_guess = [0.01, 0.1, -0.3, 0.0, 0.1]
    bounds = [(0.0,1.0),(1e-6,2.0),(-0.999,0.999),(-1.0,1.0),(1e-6,1.0)]
    res = minimize(svi_obj, init_guess, args=(k_arr,w_arr), bounds=bounds, method="L-BFGS-B")
    a,b,rho,m,sig = res.x
    def iv_fit(k):
        w = svi_raw(k,a,b,rho,m,sig)
        w = np.maximum(w,1e-12)
        return np.sqrt(w / T)
    return {"params":(a,b,rho,m,sig), "iv_func":iv_fit, "success":res.success}

def butterfly_check(F, Ks, iv_func, T, num=200):
    k_test = np.linspace(np.log(F/Ks[-1]), np.log(F/Ks[0]), num)
    w = np.array([iv_func(k)**2 * T for k in k_test])
    d2w = np.gradient(np.gradient(w, k_test), k_test)
    return not np.any(d2w < -1e-6)

# ========== 3D SVI-JW 波动率曲面 ==========
class SviSurface3D:
    def __init__(self, r: float=0.03):
        self.r = r
        self.expiry_map: Dict[float, Dict] = {}
        self.jw_map: Dict[float, Dict] = {}
        self.last_check_ok: bool = True

    def add_expiry_slice(self, F: float, K_arr: np.ndarray, iv_arr: np.ndarray, T: float):
        svi_res = fit_svi(F, K_arr, iv_arr, T)
        if not svi_res["success"]:
            return False
        self.expiry_map[T] = svi_res
        jw = svi_jw_transform(*svi_res["params"])
        self.jw_map[T] = jw
        self.last_check_ok = butterfly_check(F, K_arr, svi_res["iv_func"], T)
        return True

    def interpolate_iv(self, F: float, K: float, T_target: float) -> float:
        if len(self.expiry_map) == 0:
            return np.nan
        t_list = sorted(self.expiry_map.keys())
        if T_target <= t_list[0]:
            jw = self.jw_map[t_list[0]]
        elif T_target >= t_list[-1]:
            jw = self.jw_map[t_list[-1]]
        else:
            idx = np.searchsorted(t_list, T_target)
            t1, t2 = t_list[idx-1], t_list[idx]
            w1 = (t2 - T_target)/(t2 - t1)
            w2 = (T_target - t1)/(t2 - t1)
            j1 = self.jw_map[t1]
            j2 = self.jw_map[t2]
            jw = {k: w1*j1[k] + w2*j2[k] for k in ["vt","psi","p","c","tau"]}
        vt, psi, p, c, tau = jw["vt"], jw["psi"], jw["p"], jw["c"], jw["tau"]
        b = (p + c)/2
        rho = (c - p)/(c + p) if (c+p)!=0 else 0
        m = vt * psi / b if b!=0 else 0
        sigma = vt * tau / b if b!=0 else 0
        a = vt - b * sigma * np.sqrt(1 - rho**2)
        k = np.log(F / K)
        w = svi_raw(k,a,b,rho,m,sigma)
        w = max(w, 1e-12)
        return np.sqrt(w / T_target)

# ========== 【新增2】交易所差异化保证金：DCE / CZCE / SHFE 分组系数 ==========
@dataclass
class ExchangeMarginParam:
    exchange: str
    opt_coeff: float
    opt_min: float
    fut_coeff: float

EXCHANGE_MARGIN_CONFIG = {
    "DCE": ExchangeMarginParam("DCE", opt_coeff=0.13, opt_min=6, fut_coeff=0.11),
    "CZCE": ExchangeMarginParam("CZCE", opt_coeff=0.11, opt_min=4, fut_coeff=0.09),
    "SHFE": ExchangeMarginParam("SHFE", opt_coeff=0.12, opt_min=5, fut_coeff=0.10),
}

@dataclass
class RiskSetting:
    exchange: str = "DCE"
    price_limit: float = 0.04
    force_close_ratio: float = 0.85

class CnFutureRiskEngine:
    def __init__(self, setting: RiskSetting):
        self.setting = setting
        self.mp = EXCHANGE_MARGIN_CONFIG.get(setting.exchange, EXCHANGE_MARGIN_CONFIG["SHFE"])

    def calc_option_seller_margin(self, F: float, K: float, option_type: str, contract_multiplier: int):
        if option_type == "call":
            intrinsic = max(F - K, 0)
        else:
            intrinsic = max(K - F, 0)
        margin = F * self.mp.opt_coeff + intrinsic
        margin = max(margin, self.mp.opt_min)
        return margin * contract_multiplier

    def calc_futures_margin(self, F: float, contract_multiplier: int):
        return F * self.mp.fut_coeff * contract_multiplier

    def price_limit_band(self, pre_settle: float):
        lim = self.setting.price_limit
        return pre_settle*(1-lim), pre_settle*(1+lim)

    def check_liquidation(self, equity: float, total_margin: float) -> bool:
        if total_margin <= 0:
            return False
        return equity / total_margin < self.setting.force_close_ratio

# ========== 【新增3】盘口撮合引擎：bid/ask成交、滑点、价差 ==========
@dataclass
class OrderFillConfig:
    slip_tick: float = 0
    tick_size: float = 1
    # 卖方开仓：吃bid；卖方平仓：吃ask
    # 买方开仓：吃ask；买方平仓：吃bid

class MatchingEngine:
    def __init__(self, cfg: OrderFillConfig):
        self.cfg = cfg

    def fill_order(self, side: str, direction: str, bid: float, ask: float):
        """
        side: option / future
        direction: open / close
        返回成交价格
        """
        slip = self.cfg.slip_tick * self.cfg.tick_size
        if direction == "open":
            if side == "seller":
                fill_price = bid - slip
            else:
                fill_price = ask + slip
        else:
            if side == "seller":
                fill_price = ask + slip
            else:
                fill_price = bid - slip
        return fill_price

# ========== vnpy 定价器（对接tick、撮合、曲面） ==========
class VnpyOptionPricer:
    def __init__(self, surface: SviSurface3D, risk_engine: CnFutureRiskEngine, matcher: MatchingEngine, r: float=0.03):
        self.surface = surface
        self.risk = risk_engine
        self.matcher = matcher
        self.r = r
        self.fwd_cache: Dict[str, float] = {}
        self.tick_cache: Dict[str, Dict] = {}

    def on_tick(self, tick):
        symbol = tick.vt_symbol
        if "OP" not in symbol:
            self.fwd_cache[symbol] = tick.last_price
        self.tick_cache[symbol] = {
            "last": tick.last_price,
            "bid": tick.bid_price_1,
            "ask": tick.ask_price_1,
            "pre_settle": tick.pre_settle
        }

    def get_forward(self, underlying_symbol: str) -> Optional[float]:
        return self.fwd_cache.get(underlying_symbol)

    def price_option(self, fwd: float, K: float, T: float, option_type: str) -> Dict:
        iv = self.surface.interpolate_iv(fwd, K, T)
        if np.isnan(iv):
            return {"price":np.nan, "iv":np.nan, "delta":np.nan, "gamma":np.nan, "theta":np.nan, "vega":np.nan}
        price = black_futures_option(fwd, K, T, self.r, iv, option_type)
        g = black_greeks(fwd, K, T, self.r, iv, option_type)
        out = {"price": price, "iv": iv}
        out.update(g)
        return out

    def calc_margin(self, fwd: float, K: float, option_type: str, multiplier: int):
        return self.risk.calc_option_seller_margin(fwd, K, option_type, multiplier)

# ========== 【新增4】Greeks看板 + vnpy UI日志封装 ==========
class GreeksBoard:
    def __init__(self):
        self.data = {
            "delta":0.0, "gamma":0.0, "theta":0.0, "vega":0.0, "rho":0.0, "pv":0.0, "equity":0.0, "margin":0.0
        }

    def update(self, g: dict, pv: float, equity: float, margin: float):
        self.data.update(g)
        self.data["pv"] = pv
        self.data["equity"] = equity
        self.data["margin"] = margin

    def log_vnpy(self, vt_template):
        """输出一行给vnpy write_log"""
        s = (f"[GreeksBoard] D:{self.data['delta']:.3f} G:{self.data['gamma']:.4f} "
             f"V:{self.data['vega']:.2f} Θ:{self.data['theta']:.2f} "
             f"PV:{self.data['pv']:.2f} EQ:{self.data['equity']:.2f} MARGIN:{self.data['margin']:.2f}")
        vt_template.write_log(s)
        return s

# ========== 【新增5】卖方回测框架 + Roll换仓逻辑 ==========
@dataclass
class SellerBacktestConfig:
    strategy_type: str  # covered / straddle / strangle
    d_band: float = 0.02
    iv_filter_low: float = 0.18
    iv_filter_high: float = 0.35
    stop_loss_equity: float = 0.20
    roll_days: int = 14  # 剩余小于14个交易日触发roll
    r: float = 0.03
    contract_multiplier: int = 10
    fee_rate: float = 0.00005

class OptionSellerBacktest:
    def __init__(self, pricer: VnpyOptionPricer, config: SellerBacktestConfig, risk: CnFutureRiskEngine, matcher: MatchingEngine, board: GreeksBoard):
        self.pricer = pricer
        self.config = config
        self.risk = risk
        self.matcher = matcher
        self.board = board
        self.records = []
        self.equity = 0.0
        self.cash = 0.0
        self.positions: List[Dict] = []
        self.underlying_delta = 0.0
        self.roll_candidate_expiries: List[float] = []

    def reset(self, init_capital: float, expiries: List[float]):
        self.equity = init_capital
        self.cash = init_capital
        self.positions.clear()
        self.underlying_delta = 0.0
        self.roll_candidate_expiries = sorted(expiries)
        self.records.clear()

    def scan_leg(self, fwd: float, K: float, T: float, opt_type: str, bid: float, ask: float) -> Optional[Dict]:
        res = self.pricer.price_option(fwd, K, T, opt_type)
        iv = res["iv"]
        if not (self.config.iv_filter_low <= iv <= self.config.iv_filter_high):
            return None
        fill_price = self.matcher.fill_order(side="seller", direction="open", bid=bid, ask=ask)
        res["fill_price"] = fill_price
        return res

    def _find_next_expiry(self, t_now: float):
        """找下一个到期（roll用）"""
        for t in self.roll_candidate_expiries:
            if t > t_now + 1/252:
                return t
        return None

    def roll_positions(self, fwd: float, bid: float, ask: float):
        """Roll逻辑：平旧腿，同delta开新到期腿"""
        new_pos = []
        for p in self.positions:
            tau = p["T"]
            if tau > self.config.roll_days / 252:
                new_pos.append(p)
                continue
            # 平仓旧腿
            close_fill = self.matcher.fill_order("seller", "close", bid, ask)
            self.cash -= close_fill * self.config.contract_multiplier
            # 寻找下一个到期
            t_next = self._find_next_expiry(tau)
            if t_next is None:
                continue
            # 用原K，同类型，新到期开仓
            leg = self.scan_leg(fwd, p["K"], t_next, p["type"], bid, ask)
            if leg is not None:
                new_pos.append({"type":p["type"],"K":p["K"],"T":t_next,"w":-1,"entry_iv":leg["iv"],"entry_price":leg["fill_price"]})
                self.cash += leg["fill_price"] * self.config.contract_multiplier
        self.positions = new_pos

    def open_straddle(self, fwd: float, atm_k: float, T: float, bid: float, ask: float):
        call = self.scan_leg(fwd, atm_k, T, "call", bid, ask)
        put = self.scan_leg(fwd, atm_k, T, "put", bid, ask)
        if call is None or put is None:
            return False
        self.positions.append({"type":"call","K":atm_k,"T":T,"w":-1,"entry_iv":call["iv"],"entry_price":call["fill_price"]})
        self.positions.append({"type":"put","K":atm_k,"T":T,"w":-1,"entry_iv":put["iv"],"entry_price":put["fill_price"]})
        self.cash += (call["fill_price"] + put["fill_price"]) * self.config.contract_multiplier
        return True

    def open_strangle(self, fwd: float, k_call: float, k_put: float, T: float, bid: float, ask: float):
        call = self.scan_leg(fwd, k_call, T, "call", bid, ask)
        put = self.scan_leg(fwd, k_put, T, "put", bid, ask)
        if call is None or put is None:
            return False
        self.positions.append({"type":"call","K":k_call,"T":T,"w":-1,"entry_iv":call["iv"],"entry_price":call["fill_price"]})
        self.positions.append({"type":"put","K":k_put,"T":T,"w":-1,"entry_iv":put["iv"],"entry_price":put["fill_price"]})
        self.cash += (call["fill_price"] + put["fill_price"]) * self.config.contract_multiplier
        return True

    def open_covered(self, fwd: float, k_call: float, T: float, bid: float, ask: float):
        call = self.scan_leg(fwd, k_call, T, "call", bid, ask)
        if call is None:
            return False
        self.positions.append({"type":"call","K":k_call,"T":T,"w":-1,"entry_iv":call["iv"],"entry_price":call["fill_price"]})
        self.underlying_delta += 1.0
        self.cash += call["fill_price"] * self.config.contract_multiplier
        return True

    def calc_portfolio_greeks_pv(self, fwd: float) -> Tuple[Dict, float]:
        pv = 0.0
        g = {"delta":0.0,"gamma":0.0,"theta":0.0,"vega":0.0}
        remain_pos = []
        for p in self.positions:
            tau = p["T"]
            if tau <= 0:
                continue
            res = self.pricer.price_option(fwd, p["K"], tau, p["type"])
            w = p["w"]
            pv += w * res["price"] * self.config.contract_multiplier
            g["delta"] += w * res["delta"]
            g["gamma"] += w * res["gamma"]
            g["theta"] += w * res["theta"]
            g["vega"] += w * res["vega"]
            p["T"] = tau - 1/252
            remain_pos.append(p)
        self.positions = remain_pos
        g["delta"] += self.underlying_delta
        return g, pv

    def delta_band_rebalance(self, fwd: float, bid: float, ask: float, target_delta: float=0.0):
        g, pv = self.calc_portfolio_greeks_pv(fwd)
        current_delta = g["delta"]
        band = self.config.d_band
        if abs(current_delta - target_delta) > band:
            trade_qty = target_delta - current_delta
            fill_fut = self.matcher.fill_order(side="future", direction="open", bid=bid, ask=ask)
            fee = abs(trade_qty) * fill_fut * self.config.fee_rate * self.config.contract_multiplier
            self.cash -= trade_qty * fill_fut * self.config.contract_multiplier + fee
            self.underlying_delta += trade_qty
        return g, pv

    def step(self, fwd: float, bid: float, ask: float, pre_equity: float, vt_template=None):
        # Roll检查
        self.roll_positions(fwd, bid, ask)
        g, pv = self.delta_band_rebalance(fwd, bid, ask)
        market_value = self.cash + pv + self.underlying_delta * fwd * self.config.contract_multiplier
        self.equity = market_value
        drawdown = 1 - self.equity / pre_equity
        stop = drawdown > self.config.stop_loss_equity
        total_margin = 0.0
        for p in self.positions:
            m = self.risk.calc_option_seller_margin(fwd, p["K"], p["type"], self.config.contract_multiplier)
            total_margin += abs(p["w"]) * m
        liq = self.risk.check_liquidation(self.equity, total_margin)
        # 更新看板 + 输出vnpy日志
        self.board.update(g, pv, self.equity, total_margin)
        if vt_template is not None:
            self.board.log_vnpy(vt_template)
        rec = {
            "forward": fwd,
            "equity": self.equity,
            "cash": self.cash,
            "pv": pv,
            "port_delta": g["delta"],
            "port_gamma": g["gamma"],
            "port_vega": g["vega"],
            "stop_trigger": stop,
            "liquidation": liq,
            "total_margin": total_margin
        }
        self.records.append(rec)
        if stop or liq:
            self.positions.clear()
            self.underlying_delta = 0.0
        return rec

    def get_result_df(self):
        return pd.DataFrame(self.records)

    pricer = VnpyOptionPricer(surface=surf, risk_engine=risk_eng, matcher=matcher, r=0.03)

    cfg = SellerBacktestConfig(
        strategy_type="straddle",
        d_band=0.02,
        iv_filter_low=0.18,
        iv_filter_high=0.35,
        roll_days=14
    )
    bt = OptionSellerBacktest(pricer, cfg, risk_eng, matcher, board)
    init_cap = 30000
    exp_list = [0.25,0.5,0.75]
    bt.reset(init_cap, exp_list)

    F0 = 15000
    bt.open_straddle(F0,15000,0.25, bid=14998, ask=15002)

    np.random.seed(42)
    N_STEP = 80
    fwd_path = [F0]
    bid_path = [F0-2]
    ask_path = [F0+2]
    for _ in range(N_STEP):
        z = np.random.normal()
        f = fwd_path[-1] * np.exp((0.03 - 0.5*0.18**2)/252 + 0.18*np.sqrt(1/252)*z)
        fwd_path.append(f)
        bid_path.append(f-2)
        ask_path.append(f+2)
    eq = init_cap
    for f, b, a in zip(fwd_path, bid_path, ask_path):
        r = bt.step(f, b, a, eq)
        eq = r["equity"]
    df = bt.get_result_df()
    print(df[["forward","equity","port_delta","port_vega","liquidation"]].tail(10))

## 对应新增模块说明
### 1. TQDataIvLoader
- `load_option_chain_iv_slice`：单日期权链截面，自动按到期分组拟合SVI切片，喂给`SviSurface3D`
- `load_history_bar`：标的K线，用于回测路径生成
- 实盘时，每日收盘跑一次，刷新当日波动率曲面

### 2. 交易所差异化保证金
`EXCHANGE_MARGIN_CONFIG` 内置 DCE / CZCE / SHFE 三套系数；初始化`RiskSetting`指定交易所即可自动切换，适合生猪(DCE)、纯碱(CZCE)、燃油(SHFE)

### 3. MatchingEngine 盘口撮合
- 不再使用last_price成交
- 卖方开仓吃bid、平仓吃ask；支持固定滑点（tick数）
- 期货Delta对冲也走同一个撮合逻辑，手续费、滑点统一

### 4. GreeksBoard
- 聚合组合Delta/Gamma/Vega/Theta/PV/权益/保证金
- `.log_vnpy(vt_template)` 直接调用vnpy策略的`write_log`，输出到vnpy UI日志面板
- 你可以在vnpy界面上直接看到实时希腊监控

### 5. Roll换仓逻辑
- 配置`roll_days`：剩余交易日小于阈值自动触发roll
- 先平旧腿（按盘口ask成交），再在**下一个到期合约**同行权价、同方向开新腿
- 到期候选列表`roll_candidate_expiries`在reset时传入，自动找最近的远端到期

## vnpy策略嵌入示例（可直接放进vnpy_ctastrategy）
```python
from vnpy_ctastrategy import CtaTemplate, TickData
from option_pricer import (
    SviSurface3D, CnFutureRiskEngine, RiskSetting,
    MatchingEngine, OrderFillConfig, VnpyOptionPricer, GreeksBoard,
    OptionSellerBacktest, SellerBacktestConfig
)

class OptionStraddleSeller(CtaTemplate):
    author = "quant"
    parameters = ["exchange","roll_days","d_band"]
    variables = ["equity"]

    def __init__(self, cta_engine, strategy_name, vt_symbol, setting):
        super().__init__(cta_engine, strategy_name, vt_symbol, setting)
        self.surf = SviSurface3D()
        self.risk = CnFutureRiskEngine(RiskSetting(exchange=self.exchange))
        self.matcher = MatchingEngine(OrderFillConfig(slip_tick=1, tick_size=1))
        self.board = GreeksBoard()
        self.pricer = VnpyOptionPricer(self.surf, self.risk, self.matcher)
        self.cfg = SellerBacktestConfig(strategy_type="straddle", roll_days=self.roll_days, d_band=self.d_band)
        self.bt = OptionSellerBacktest(self.pricer, self.cfg, self.risk, self.matcher, self.board)
        self.bt.reset(30000, expiries=[0.25,0.5,0.75])

    def on_tick(self, tick: TickData):
        self.pricer.on_tick(tick)
        fwd = self.pricer.get_forward("lh2701")
        if fwd is None:
            return
        rec = self.bt.step(fwd, tick.bid_price_1, tick.ask_price_1, self.bt.equity, vt_template=self)
        self.equity = rec["equity"]
        self.put_event()
```
