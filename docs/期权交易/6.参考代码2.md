> 统一约定：T 单位为年；252 交易日；手续费按每手名义比例设置。

```python
import numpy as np
from scipy.stats import norm
from scipy.optimize import minimize
import matplotlib.pyplot as plt

# ========== 复用之前基础函数（直接粘贴，保证整套可独立运行） ==========
N = norm.cdf
n_pdf = norm.pdf

def bs_option(S, K, T, r, sigma, q=0.0, option_type="call"):
    d1 = (np.log(S / K) + (r - q + 0.5 * sigma ** 2) * T) / (sigma * np.sqrt(T))
    d2 = d1 - sigma * np.sqrt(T)
    if option_type == "call":
        price = S * np.exp(-q * T) * N(d1) - K * np.exp(-r * T) * N(d2)
    else:
        price = K * np.exp(-r * T) * N(-d2) - S * np.exp(-q * T) * N(-d1)
    return price

def bs_greeks(S, K, T, r, sigma, q=0.0, option_type="call"):
    d1 = (np.log(S / K) + (r - q + 0.5 * sigma ** 2) * T) / (sigma * np.sqrt(T))
    d2 = d1 - sigma * np.sqrt(T)
    pdf_d1 = n_pdf(d1)
    if option_type == "call":
        delta = np.exp(-q * T) * N(d1)
    else:
        delta = np.exp(-q * T) * (N(d1) - 1)
    gamma = np.exp(-q * T) * pdf_d1 / (S * sigma * np.sqrt(T))
    term1 = - S * np.exp(-q * T) * pdf_d1 * sigma / (2 * np.sqrt(T))
    if option_type == "call":
        term2 = q * S * np.exp(-q * T) * N(d1)
        term3 = - r * K * np.exp(-r * T) * N(d2)
    else:
        term2 = - q * S * np.exp(-q * T) * N(-d1)
        term3 = r * K * np.exp(-r * T) * N(-d2)
    theta = term1 + term2 + term3
    vega = S * np.exp(-q * T) * pdf_d1 * np.sqrt(T)
    if option_type == "call":
        rho = K * T * np.exp(-r * T) * N(d2)
    else:
        rho = -K * T * np.exp(-r * T) * N(-d2)
    return {"delta": delta, "gamma": gamma, "theta": theta, "vega": vega, "rho": rho}

def black_futures_option(F, K, T, r, sigma, option_type="call"):
    d1 = (np.log(F / K) + 0.5 * sigma ** 2 * T) / (sigma * np.sqrt(T))
    d2 = d1 - sigma * np.sqrt(T)
    disc = np.exp(-r * T)
    if option_type == "call":
        price = disc * (F * N(d1) - K * N(d2))
    else:
        price = disc * (K * N(-d2) - F * N(-d1))
    return price

def black_greeks(F, K, T, r, sigma, option_type="call"):
    d1 = (np.log(F / K) + 0.5 * sigma ** 2 * T) / (sigma * np.sqrt(T))
    d2 = d1 - sigma * np.sqrt(T)
    pdf_d1 = n_pdf(d1)
    disc = np.exp(-r * T)
    if option_type == "call":
        delta = disc * N(d1)
    else:
        delta = disc * (N(d1) - 1)
    gamma = disc * pdf_d1 / (F * sigma * np.sqrt(T))
    term1 = - F * disc * pdf_d1 * sigma / (2 * np.sqrt(T))
    if option_type == "call":
        term2 = r * F * disc * N(d1) - r * K * disc * N(d2)
    else:
        term2 = r * K * disc * N(-d2) - r * F * disc * N(-d1)
    theta = term1 + term2
    vega = F * disc * pdf_d1 * np.sqrt(T)
    if option_type == "call":
        rho = - T * disc * (F * N(d1) - K * N(d2))
    else:
        rho = - T * disc * (K * N(-d2) - F * N(-d1))
    return {"delta": delta, "gamma": gamma, "theta": theta, "vega": vega, "rho": rho}

# =====================================================
# 模块1：蒙特卡洛定价 + 方差缩减（控制变量 + 对偶变量）
# 标的：几何布朗运动 GBM；支持股票(BS) / 期货(Black)两种模式
# =====================================================
def mc_gbm_paths(S0, T, r, sigma, q=0, n_paths=10000, n_steps=252, seed=42):
    """生成GBM路径矩阵 (n_paths, n_steps+1)"""
    np.random.seed(seed)
    dt = T / n_steps
    drift = (r - q - 0.5 * sigma**2) * dt
    vol = sigma * np.sqrt(dt)
    S = np.zeros((n_paths, n_steps + 1))
    S[:, 0] = S0
    for t in range(1, n_steps+1):
        z = np.random.normal(0,1, n_paths)
        S[:, t] = S[:, t-1] * np.exp(drift + vol * z)
    return S

def mc_european_option(S0, K, T, r, sigma, q=0, option_type="call",
                       n_paths=20000, n_steps=1,
                       control_var=True, antithetic=True):
    """
    蒙特卡洛欧式期权定价
    control_var: 控制变量（用B-S解析解作为控制变量）
    antithetic: 对偶变量方差缩减
    """
    if antithetic:
        n = n_paths // 2
        z = np.random.normal(0,1,n)
        z_all = np.concatenate([z, -z])
    else:
        z_all = np.random.normal(0,1,n_paths)
    ST = S0 * np.exp( (r - q - 0.5*sigma**2)*T + sigma*np.sqrt(T)*z_all )
    if option_type == "call":
        payoff = np.maximum(ST - K, 0.0)
    else:
        payoff = np.maximum(K - ST, 0.0)
    disc = np.exp(-r*T)
    raw_price = disc * np.mean(payoff)
    raw_se = disc * np.std(payoff) / np.sqrt(len(payoff))

    if not control_var:
        return {"price": raw_price, "se": raw_se, "payoff": payoff}

    # 控制变量：BS解析解作为控制变量
    bs_analytic = bs_option(S0,K,T,r,sigma,q,option_type)
    cov = np.cov(payoff, ST)[0,1]
    var_ST = np.var(ST)
    c = - cov / var_ST
    payoff_cv = payoff + c * (ST - S0 * np.exp((r - q)*T))
    cv_price = disc * np.mean(payoff_cv)
    cv_se = disc * np.std(payoff_cv) / np.sqrt(len(payoff_cv))
    return {"price": cv_price, "se": cv_se, "raw_price": raw_price, "raw_se": raw_se}

# =====================================================
# 模块2：Delta动态对冲仿真（带手续费、定期再平衡）
# 场景：卖出期权，持续Delta对冲；统计PnL、对冲误差
# =====================================================
def delta_hedge_simulation(S0, K, T, r, sigma, q=0, option_type="call",
                           n_steps=252, fee_rate=0.0001, seed=42):
    """
    卖出一份欧式期权，逐日Delta对冲仿真
    fee_rate: 每一次调仓，名义金额手续费比例
    返回：时序S、期权理论价值、delta、持仓、现金、累计手续费、PnL
    """
    np.random.seed(seed)
    dt = T / n_steps
    S = np.zeros(n_steps+1)
    S[0] = S0
    cash = 0.0
    delta_hold = 0.0
    fee_total = 0.0
    V_hist = []
    delta_hist = []
    cash_hist = []
    S_hist = []

    # 期初：卖出期权，收到期权费
    V0 = bs_option(S0, K, T, r, sigma, q, option_type)
    cash += V0

    for t in range(n_steps+1):
        tau = T - t * dt
        if tau < 1e-8:
            V = np.maximum(S[t]-K,0) if option_type=="call" else np.maximum(K-S[t],0)
        else:
            V = bs_option(S[t], K, tau, r, sigma, q, option_type)
            g = bs_greeks(S[t], K, tau, r, sigma, q, option_type)
            delta = g["delta"]
        V_hist.append(V)
        S_hist.append(S[t])
        if tau < 1e-8:
            delta_hist.append(0.0)
            break
        delta_hist.append(delta)
        # 调仓：目标delta = -delta（卖出期权，对冲负delta）
        target = -delta
        trade = target - delta_hold
        # 手续费：交易名义 * fee_rate
        fee = abs(trade) * S[t] * fee_rate
        fee_total += fee
        cash -= trade * S[t] + fee
        delta_hold = target
        # 现金无风险计息
        cash *= np.exp(r * dt)
        cash_hist.append(cash)
        # 生成下一期标的价格
        z = np.random.normal()
        S[t+1] = S[t] * np.exp( (r - q - 0.5*sigma**2)*dt + sigma*np.sqrt(dt)*z )

    # 到期平仓：期权payoff支出，股票持仓平仓
    final_pnl = cash + delta_hold * S[-1] - V_hist[-1]
    out = {
        "S": np.array(S_hist),
        "V": np.array(V_hist),
        "delta": np.array(delta_hist),
        "cash": np.array(cash_hist),
        "fee_total": fee_total,
        "final_pnl": final_pnl
    }
    return out

# =====================================================
# 模块3：SVI 波动率微笑拟合（Raw SVI，适合单到期日IV曲面）
# SVI公式：w(k) = a + b*(rho*(k-m)+np.sqrt((k-m)**2 + sigma**2))
# k = log(F/K) 对数行权价；w = iv^2 * T 总方差
# =====================================================
def svi_raw(k, a, b, rho, m, sigma):
    """Raw SVI 总方差 w(k)"""
    return a + b * ( rho * (k - m) + np.sqrt( (k - m)**2 + sigma**2 ) )

def svi_obj(params, k_arr, w_arr):
    """最小二乘目标函数：拟合w = iv²*T"""
    a,b,rho,m,sig = params
    # SVI约束：b>0, |rho|<1, sig>0
    if b <= 0 or abs(rho)>=1 or sig <=0:
        return 1e12
    w_fit = svi_raw(k_arr, a,b,rho,m,sig)
    return np.sum((w_fit - w_arr)**2)

def fit_svi(F, K_arr, iv_arr, T, init_guess=None):
    """
    F: forward/期货价格
    K_arr: 行权价数组
    iv_arr: 对应隐含波动率数组
    T: 到期时间（年）
    return: 拟合参数 + 拟合函数
    """
    k_arr = np.log(F / K_arr)
    w_arr = iv_arr**2 * T
    if init_guess is None:
        init_guess = [0.01, 0.1, -0.3, 0.0, 0.1]
    bounds = [
        (0.0, 1.0),      # a
        (1e-6, 2.0),     # b
        (-0.999,0.999),  # rho
        (-1.0,1.0),      # m
        (1e-6, 1.0)      # sigma
    ]
    res = minimize(svi_obj, init_guess, args=(k_arr,w_arr), bounds=bounds, method="L-BFGS-B")
    a,b,rho,m,sig = res.x
    def iv_fit(k):
        w = svi_raw(k,a,b,rho,m,sig)
        w = np.maximum(w, 1e-12)
        return np.sqrt(w / T)
    return {"params":(a,b,rho,m,sig), "success": res.success, "iv_func": iv_fit, "k_arr":k_arr}

# ====================== 测试入口 ======================
if __name__ == "__main__":
    # ---------- 1. 蒙特卡洛测试 ----------
    print("===== 蒙特卡洛欧式期权（对偶+控制变量） =====")
    mc_res = mc_european_option(S0=100, K=100, T=1, r=0.03, sigma=0.2, q=0.01, option_type="call", n_paths=20000)
    bs_price = bs_option(100,100,1,0.03,0.2,0.01,"call")
    print(f"MC价格：{mc_res['price']:.4f}, 标准误：{mc_res['se']:.6f}")
    print(f"B-S解析解：{bs_price:.4f}")

    # ---------- 2. Delta对冲仿真 ----------

    print(f"期末对冲PnL：{hedge['final_pnl']:.4f}, 总手续费：{hedge['fee_total']:.4f}")
    plt.figure(figsize=(10,4))
    plt.subplot(1,2,1)

    plt.show()

    # ---------- 3. SVI微笑拟合测试 ----------
    print("\n===== SVI波动率微笑拟合 =====")
    F = 100
    Ks = np.array([80,85,90,95,100,105,110,115,120])
    ivs = np.array([0.27,0.24,0.215,0.202,0.20,0.203,0.218,0.245,0.28])

    k_plot = np.linspace(-0.3, 0.3, 100)
    iv_plot = [svi_out["iv_func"](k) for k in k_plot]
    plt.figure()
    plt.plot(k_plot, iv_plot, label="SVI拟合IV")
    plt.scatter(np.log(F/Ks), ivs, c="r", label="市场IV点")
    plt.xlabel("k = log(F/K)")
    plt.ylabel("Implied Vol")
    plt.title("SVI波动率微笑")
    plt.legend()
    plt.show()
```

## 模块要点说明（贴合你做商品期权）
### 1. 蒙特卡洛
- **对偶变量（Antithetic）**：用 $Z$ 和 $-Z$ 成对采样，降低方差；最简单高效。
- **控制变量（Control Variate）**：用解析B-S作为控制变量，适合欧式；如果是期货期权，把底层替换成 `black_futures_option` 即可。
- 路径生成函数 `mc_gbm_paths` 可以拿来做压力测试、路径依赖期权（亚洲、回望）。
> 局限：这里是欧式；美式MC需要LSMC（最小二乘蒙特卡洛），后面可以再加。

### 2. Delta动态对冲仿真
- 场景：**卖出期权，持续对冲Delta**（卖方最常见），目标是对冲到期PnL收敛到期权费减去对冲成本。
- 再平衡：**固定时间步（每日）**，不是连续对冲；现实就是离散再平衡，Gamma带来对冲误差。
- 手续费：`fee_rate` 是**交易名义金额比例**，国内期货期权可以改成按手固定手续费。
- 输出：对冲损益本质就是**Gamma PnL - Theta - 手续费**，是期权卖方回测核心。
> 扩展方向：加入滑点、触发式再平衡（Delta阈值触发，而不是每日）、Gamma对冲。

### 3. Raw SVI（Gatheral SVI）
公式：
$$w(k)=a+b\left(\rho(k-m)+\sqrt{(k-m)^2+\sigma^2}\right)$$
$k=\ln(F/K)$，$w=\text{IV}^2 T$ 总方差
- Raw SVI 是行业标准，**单到期日微笑**，适合拟合国内商品期权同一到期的IV截面；
- 约束：$b>0, |\rho|<1, \sigma>0$，保证无套利基础（不做蝶式套利检验，进阶可以加套利约束）；
- 输出`iv_func`可以直接插值任意行权价IV，喂给Black模型定价。
> 进阶：SVI-JW，把参数分解成atm vol、斜率、凸度，方便曲面跨到期插值。
