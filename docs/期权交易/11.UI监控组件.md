# vnpy GreeksMonitor 监控组件（基于 vnpy 原生 Qt 界面，适配 vnpy_ctastrategy）
> 说明：vnpy 底层是 PyQt5，这个控件是独立 `GreeksMonitorWidget`，可以挂载到策略 UI，**实时表格展示组合希腊、保证金、权益、PV**，自动刷新，不依赖日志。
> 适配前面整套卖方期权框架；和上面 `GreeksBoard` 对接。
> 存放路径：`vnpy_ctastrategy/widgets/greeks_monitor.py`

```python
"""
vnpy_ctastrategy/widgets/greeks_monitor.py
Greeks 实时监控面板，独立Qt控件
"""
from PyQt5.QtWidgets import (
    QWidget, QVBoxLayout, QTableWidget, QTableWidgetItem,
    QHeaderView, QLabel, QFrame
)
from PyQt5.QtCore import Qt, pyqtSlot
from PyQt5.QtGui import QFont, QColor


class GreeksMonitorWidget(QWidget):
    """
    期权组合Greeks监控面板
    字段：Delta / Gamma / Vega / Theta / Rho / PV / Equity / Margin
    """
    def __init__(self, parent=None):
        super().__init__(parent)
        self.setWindowTitle("期权Greeks监控面板")
        self.init_ui()

    def init_ui(self):
        layout = QVBoxLayout()
        self.setLayout(layout)

        # 标题
        title_label = QLabel("【期权卖方组合风险监控】")
        title_label.setFont(QFont("Microsoft YaHei", 12, QFont.Bold))
        title_label.setAlignment(Qt.AlignCenter)
        layout.addWidget(title_label)

        # 分割线
        line = QFrame()
        line.setFrameShape(QFrame.HLine)
        layout.addWidget(line)

        # 表格
        self.table = QTableWidget()
        self.table.setColumnCount(2)
        self.table.setRowCount(8)
        self.table.setHorizontalHeaderLabels(["指标", "数值"])
        self.table.verticalHeader().setVisible(False)

        self.labels = [
            "Delta",
            "Gamma",
            "Vega",
            "Theta",
            "Rho",
            "组合PV(元)",
            "账户权益(元)",
            "占用保证金(元)"
        ]
        for row, name in enumerate(self.labels):
            item_key = QTableWidgetItem(name)
            item_key.setTextAlignment(Qt.AlignCenter)
            self.table.setItem(row, 0, item_key)
            item_val = QTableWidgetItem("--")
            item_val.setTextAlignment(Qt.AlignCenter)
            self.table.setItem(row, 1, item_val)

        # 自适应列宽
        self.table.horizontalHeader().setSectionResizeMode(0, QHeaderView.Stretch)
        self.table.horizontalHeader().setSectionResizeMode(1, QHeaderView.Stretch)
        layout.addWidget(self.table)

        # 状态标签（强平/止损告警）
        self.alert_label = QLabel("状态：正常")
        self.alert_label.setFont(QFont("Microsoft YaHei", 10))
        self.alert_label.setAlignment(Qt.AlignCenter)
        layout.addWidget(self.alert_label)

    @pyqtSlot(dict)
    def update_data(self, data: dict):
        """
        外部调用更新接口
        data 字典结构：
        {
            "delta": float,
            "gamma": float,
            "vega": float,
            "theta": float,
            "rho": float,
            "pv": float,
            "equity": float,
            "margin": float,
            "liquidation": bool,
            "stop_trigger": bool
        }
        """
        row_data = [
            f"{data.get('delta',0):.3f}",
            f"{data.get('gamma',0):.4f}",
            f"{data.get('vega',0):.2f}",
            f"{data.get('theta',0):.2f}",
            f"{data.get('rho',0):.2f}",
            f"{data.get('pv',0):.2f}",
            f"{data.get('equity',0):.2f}",
            f"{data.get('margin',0):.2f}"
        ]
        for row, val in enumerate(row_data):
            item = self.table.item(row, 1)
            item.setText(val)
            # 风险高亮：Delta绝对值过大标黄
            if self.labels[row] == "Delta" and abs(data.get("delta",0)) > 0.3:
                item.setForeground(QColor(200, 80, 0))
            else:
                item.setForeground(QColor(0,0,0))

        # 告警
        liq = data.get("liquidation", False)
        stop = data.get("stop_trigger", False)
        if liq:
            self.alert_label.setText("⚠️ 警告：触发强平！")
            self.alert_label.setStyleSheet("color:red; font-weight:bold;")
        elif stop:
            self.alert_label.setText("⚠️ 警告：触发回撤止损！")
            self.alert_label.setStyleSheet("color:rgb(220,100,0); font-weight:bold;")
        else:
            self.alert_label.setText("✅ 状态：正常")
            self.alert_label.setStyleSheet("color:green;")

    def clear(self):
        """清空面板，策略停止时调用"""
        for row in range(8):
            self.table.item(row,1).setText("--")
        self.alert_label.setText("状态：已停止")
        self.alert_label.setStyleSheet("color:gray;")
```

## 第二步：修改策略文件，把面板挂载进 vnpy CTA 策略
> 策略文件：`vnpy_ctastrategy/strategies/option_straddle_seller.py`
> 重点改动：
> 1. 增加 `create_widget` 方法（vnpy 标准扩展接口，会自动在策略详情页加载自定义UI）
> 2. 把原来 `GreeksBoard` 的数据，转发给 `GreeksMonitorWidget`
> 3. 策略 `on_tick` 回测/实盘每tick自动刷新面板

```python
from vnpy_ctastrategy import CtaTemplate, TickData
from vnpy_ctastrategy.widgets.greeks_monitor import GreeksMonitorWidget
from option_pricer import (
    SviSurface3D, CnFutureRiskEngine, RiskSetting,
    MatchingEngine, OrderFillConfig, VnpyOptionPricer, GreeksBoard,
    OptionSellerBacktest, SellerBacktestConfig
)


class OptionStraddleSeller(CtaTemplate):
    author = "quant"
    # 策略参数
    parameters = [
        "exchange",
        "roll_days",
        "d_band",
        "iv_low",
        "iv_high",
        "init_cap"
    ]
    # 策略变量
    variables = ["equity"]

    def __init__(self, cta_engine, strategy_name, vt_symbol, setting):
        super().__init__(cta_engine, strategy_name, vt_symbol, setting)
        # 初始化曲面、风控、撮合、定价器、回测实例
        self.surf = SviSurface3D()
        self.risk = CnFutureRiskEngine(RiskSetting(exchange=self.exchange))
        self.matcher = MatchingEngine(OrderFillConfig(slip_tick=1, tick_size=1))
        self.board = GreeksBoard()
        self.pricer = VnpyOptionPricer(self.surf, self.risk, self.matcher)

        self.cfg = SellerBacktestConfig(
            strategy_type="straddle",
            roll_days=self.roll_days,
            d_band=self.d_band,
            iv_filter_low=self.iv_low,
            iv_filter_high=self.iv_high
        )
        self.bt = OptionSellerBacktest(
            self.pricer, self.cfg, self.risk, self.matcher, self.board
        )
        # 到期候选列表，根据标的自行修改
        self.expiry_list = [0.25, 0.5, 0.75]
        self.bt.reset(self.init_cap, self.expiry_list)

        self.monitor_widget: GreeksMonitorWidget = None

    def create_widget(self):
        """
        vnpy 标准接口：策略详情页面加载自定义UI
        点击策略【详情】，就会弹出/嵌入这个Greeks面板
        """
        self.monitor_widget = GreeksMonitorWidget()
        return self.monitor_widget

    def on_tick(self, tick: TickData):
        # tick送入定价器缓存
        self.pricer.on_tick(tick)
        fwd = self.pricer.get_forward("lh2701")
        if fwd is None:
            return

        # 执行step，拿到回测/风控记录
        rec = self.bt.step(
            fwd,
            tick.bid_price_1,
            tick.ask_price_1,
            self.bt.equity,
            vt_template=self
        )
        self.equity = rec["equity"]

        # 组装给监控面板的数据
        monitor_data = {
            "delta": rec["port_delta"],
            "gamma": rec["port_gamma"],
            "vega": rec["port_vega"],
            "theta": self.board.data["theta"],
            "rho": self.board.data["rho"],
            "pv": rec["pv"],
            "equity": rec["equity"],
            "margin": rec["total_margin"],
            "liquidation": rec["liquidation"],
            "stop_trigger": rec["stop_trigger"]
        }
        # 更新UI面板
        if self.monitor_widget:
            self.monitor_widget.update_data(monitor_data)

        self.put_event()

    def on_stop(self):
        """策略停止时清空面板"""
        if self.monitor_widget:
            self.monitor_widget.clear()
        super().on_stop()
```

## 使用方式（vnpy客户端操作）
1. 启动 vnpy，添加该策略（参数：exchange=DCE，roll_days=14，d_band=0.02，iv_low=0.18，iv_high=0.35，init_cap=30000）
2. 启动策略
3. 在策略列表，右键这条策略 → **详情**
4. 弹窗里就会出现 Greeks 监控面板，**实时刷新表格**：
   - Delta 绝对值>0.3 自动标橙告警
   - 触发强平 → 红色警告
   - 触发回撤止损 → 橙黄色警告
   - 正常状态绿色✅
