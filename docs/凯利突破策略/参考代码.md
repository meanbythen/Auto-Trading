下面是完整整合版本，在原有升级策略基础上增加开仓前详细日志打印：
打印字段：
● 原始凯利 f*
● 缩放后 f_used
● 凯利算出风险金额（risk_by_kelly）
● 3%封顶后最终单笔风险（risk_per_trade）
● ATR / 价格波动率比值（atr_price_ratio）
● 和冲突品种的相关系数（如果触发相关性拦截，一并打印是哪一个品种+相关系数）
改动说明：
1. 修改 GlobalRiskManager.check_correlation_ok，返回 (ok, corr_info)，不仅返回是否通过，还返回相关系数和对标品种；
2. 在开仓分支前，统一输出复盘日志；
3. 回测开关保留，回测时相关性检查直接放行，不影响批量参数扫描；
4. 日志用 self.write_log，vnpy 日志面板直接可见，也可以导出复盘。

# ===================== 全局风险管理器（支持返回相关系数信息） =====================
class GlobalRiskManager:
    """
    单例全局风险管理器，所有凯利策略实例注册进来
    用途：相关性矩阵 + 组合总风险上限控制
    """
    _instance = None

    def __new__(cls):
        if cls._instance is None:
            cls._instance = super().__new__(cls)
            cls._instance.reset()
        return cls._instance

    def reset(self):
        self.symbol_list = []
        self.symbol_risk = {}
        self.price_data = {}
        self.total_risk_limit_pct = 0.06
        self.corr_threshold = 0.7

    def register_symbol(self, vt_symbol):
        if vt_symbol not in self.symbol_list:
            self.symbol_list.append(vt_symbol)
            self.symbol_risk[vt_symbol] = 0.0
            self.price_data[vt_symbol] = deque(maxlen=100)

    def update_price(self, vt_symbol, price):
        if vt_symbol in self.price_data:
            self.price_data[vt_symbol].append(price)

    def set_symbol_risk(self, vt_symbol, risk_amount):
        self.symbol_risk[vt_symbol] = risk_amount

    def get_total_risk(self):
        return sum(self.symbol_risk.values())

    def check_correlation_ok(self, new_vt_symbol):
        """
        返回 (是否允许开仓, 信息字符串)
        信息字符串：无冲突 / 冲突品种+相关系数
        """
        price_new = np.array(self.price_data.get(new_vt_symbol, []))
        if len(price_new) < 30:
            return True, "样本不足30，相关性暂放行"

        for sym in self.symbol_list:
            if sym == new_vt_symbol:
                continue
            if self.symbol_risk.get(sym, 0) <= 0:
                continue
            price_other = np.array(self.price_data.get(sym, []))
            min_len = min(len(price_new), len(price_other))
            if min_len < 30:
                continue
            corr = np.corrcoef(price_new[:min_len], price_other[:min_len])[0,1]
            if abs(corr) > self.corr_threshold:
                return False, f"与{sym}相关系数={corr:.3f}"
        return True, "相关性检查通过"

    def check_total_risk_ok(self, account_balance):
        total_risk = self.get_total_risk()
        limit = account_balance * self.total_risk_limit_pct
        return total_risk < limit, f"当前组合总风险={total_risk:.2f},上限={limit:.2f}"


g_risk_mgr = GlobalRiskManager()

# 回测开关，回测时关闭相关性检测
BACKTEST_MODE = False
if BACKTEST_MODE:
    def dummy_corr_check(*args, **kwargs):
        return True, "回测模式，跳过相关性"
    def dummy_risk_check(*args, **kwargs):
        return True, "回测模式，跳过组合总风险"
    g_risk_mgr.check_correlation_ok = dummy_corr_check
    g_risk_mgr.check_total_risk_ok = dummy_risk_check


class KellyBreakAtrStrategy(CtaTemplate):
    """
    升级版本：
    1. 波动率过滤
    2. 全局相关性检测 + 组合总风险上限
    3. 单笔风险硬封顶：3%
    4. 交易日自动重置daily_pnl
    5. 【新增】开仓前打印完整复盘日志
    """
    author = "vnpy_kelly_log"
    # -------------------------- 参数（界面可调） --------------------------
    n_window = 20
    atr_window = 20
    atr_stop_multiple = 1.2
    profit_loss_ratio = 2.5
    kelly_scale = 1/3
    trade_stat_window = 200
    max_account_drawdown = 0.20
    max_daily_loss_pct = 0.05
    max_pos_per_symbol = 1

    volatility_filter_min_atr_ratio = 0.008
    single_risk_cap_pct = 0.03
    total_port_risk_pct = 0.06
    corr_threshold = 0.7

    parameters = [
        "n_window",
        "atr_window",
        "atr_stop_multiple",
        "profit_loss_ratio",
        "kelly_scale",
        "trade_stat_window",
        "max_account_drawdown",
        "max_daily_loss_pct",
        "max_pos_per_symbol",
        "volatility_filter_min_atr_ratio",
        "single_risk_cap_pct",
        "total_port_risk_pct",
        "corr_threshold",
    ]
    variables = [
        "high_n", "low_n", "atr_value",
        "win_rate", "b", "f_kelly", "f_used",
        "account_high", "daily_pnl", "in_trade",
        "atr_price_ratio", "today_date"
    ]

    def __init__(self, cta_engine, strategy_name, vt_symbol, setting):
        super().__init__(cta_engine, strategy_name, vt_symbol, setting)
        self.bg = BarGenerator(self.on_bar)
        self.am = ArrayManager(max(self.n_window, self.atr_window) + 10)
        self.trade_records = deque(maxlen=self.trade_stat_window)

        self.long_pos = 0
        self.short_pos = 0
        self.in_trade = False

        self.long_stop = 0.0
        self.long_target = 0.0
        self.short_stop = 0.0
        self.short_target = 0.0

        self.account_high = 0.0
        self.daily_pnl = 0.0
        self.init_balance = 0.0
        self.today_date = None

        self.win_rate = 0.0
        self.b = 0.0
        self.f_kelly = 0.0
        self.f_used = 0.0

        g_risk_mgr.register_symbol(self.vt_symbol)
        g_risk_mgr.total_risk_limit_pct = self.total_port_risk_pct
        g_risk_mgr.corr_threshold = self.corr_threshold

    def on_init(self):
        self.write_log("带复盘日志凯利突破策略初始化")
        self.load_bar(10)

    def on_start(self):
        self.write_log("带复盘日志凯利突破策略启动")
        self.init_balance = self.cta_engine.get_balance()
        self.account_high = self.init_balance
        self.today_date = date.today()

    def on_stop(self):
        self.write_log("带复盘日志凯利突破策略停止")

    def on_tick(self, tick: TickData):
        self.bg.update_tick(tick)

    def on_bar(self, bar: BarData):
        self.cancel_all()
        self.am.update_bar(bar)
        if not self.am.inited:
            return

        # ========= 每日重置单日盈亏 =========
        bar_date = bar.datetime.date()
        if bar_date != self.today_date:
            self.daily_pnl = 0.0
            self.today_date = bar_date
            self.write_log(f"新交易日，重置当日盈亏，日期:{bar_date}")

        self.high_n = self.am.high[-self.n_window:].max()
        self.low_n = self.am.low[-self.n_window:].min()
        self.atr_value = self.am.atr(self.atr_window)
        close_price = bar.close
        if self.atr_value <= 0 or close_price <= 0:
            return

        self.atr_price_ratio = self.atr_value / close_price
        if self.atr_price_ratio < self.volatility_filter_min_atr_ratio:
            self.write_log(f"波动率过滤：ATR/价格={self.atr_price_ratio:.4f}，低于阈值")
            return

        g_risk_mgr.update_price(self.vt_symbol, close_price)

        current_balance = self.cta_engine.get_balance()
        if current_balance > self.account_high:
            self.account_high = current_balance
        drawdown_pct = (self.account_high - current_balance) / self.account_high

        risk_pass = True
        risk_msg_list = []
        if drawdown_pct >= self.max_account_drawdown:
            risk_pass = False
            risk_msg_list.append(f"回撤超标 {drawdown_pct:.2%}")
        if self.daily_pnl <= -self.init_balance * self.max_daily_loss_pct:
            risk_pass = False
            risk_msg_list.append("单日亏损超标")

        corr_ok, corr_msg = g_risk_mgr.check_correlation_ok(self.vt_symbol)
        port_ok, port_msg = g_risk_mgr.check_total_risk_ok(current_balance)
        if not corr_ok:
            risk_pass = False
            risk_msg_list.append(corr_msg)
        if not port_ok:
            risk_pass = False
            risk_msg_list.append(port_msg)

        self.calc_kelly()
        # 预计算风险，用于日志打印
        risk_by_kelly = current_balance * self.f_used
        risk_cap = current_balance * self.single_risk_cap_pct
        risk_per_trade = min(risk_by_kelly, risk_cap)

        # 统一开仓前复盘日志
        log_text = (
            f"【开仓复盘】{self.vt_symbol} | 原始f*={self.f_kelly:.4f} | "
            f"缩放f_used={self.f_used:.4f} | 凯利风险={risk_by_kelly:.2f} | "
            f"封顶后风险={risk_per_trade:.2f} | ATR波动率(ATR/Close)={self.atr_price_ratio:.4f} | "
            f"相关性:{corr_msg} | 组合风险:{port_msg} | "
            f"胜率={self.win_rate:.3f} | 盈亏比b={self.b:.2f}"
        )
        self.write_log(log_text)

        # 多头开仓
        if bar.close > self.high_n and self.long_pos == 0 and risk_pass:
            lots = self.calc_lots(current_balance)
            if lots > 0:
                self.write_log(f"✅ 多头开仓，价格:{bar.close},手数:{lots}")
                self.buy(bar.close, lots, stop=True)
                self.long_stop = bar.close - self.atr_value * self.atr_stop_multiple
                self.long_target = bar.close + self.atr_value * self.atr_stop_multiple * self.profit_loss_ratio
                self.in_trade = True
                contract_multiplier = self.cta_engine.get_multiplier(self.vt_symbol)
                risk_per_lot = self.atr_value * contract_multiplier
                g_risk_mgr.set_symbol_risk(self.vt_symbol, lots * risk_per_lot)

        # 空头开仓
        if bar.close < self.low_n and self.short_pos == 0 and risk_pass:
            lots = self.calc_lots(current_balance)
            if lots > 0:
                self.write_log(f"✅ 空头开仓，价格:{bar.close},手数:{lots}")
                self.short(bar.close, lots, stop=True)
                self.short_stop = bar.close + self.atr_value * self.atr_stop_multiple
                self.short_target = bar.close - self.atr_value * self.atr_stop_multiple * self.profit_loss_ratio
                self.in_trade = True
                contract_multiplier = self.cta_engine.get_multiplier(self.vt_symbol)
                risk_per_lot = self.atr_value * contract_multiplier
                g_risk_mgr.set_symbol_risk(self.vt_symbol, lots * risk_per_lot)

        if self.long_pos > 0:
            self.sell(self.long_stop, self.long_pos, stop=True)
            self.sell(self.long_target, self.long_pos, limit=True)
        if self.short_pos > 0:
            self.cover(self.short_stop, self.short_pos, stop=True)
            self.cover(self.short_target, self.short_pos, limit=True)

        if self.long_pos == 0 and self.short_pos == 0:
            g_risk_mgr.set_symbol_risk(self.vt_symbol, 0.0)

        self.put_event()

    def calc_kelly(self):
        trades = list(self.trade_records)
        if len(trades) < 10:
            self.win_rate = 0
            self.b = 0
            self.f_kelly = 0
            self.f_used = 0
            return
        wins = [x for x in trades if x > 0]
        losses = [x for x in trades if x < 0]
        win_count = len(wins)
        loss_count = len(losses)
        total = len(trades)
        self.win_rate = win_count / total
        avg_win = np.mean(wins) if wins else 0
        avg_loss = abs(np.mean(losses)) if losses else 0
        if avg_loss < 1e-6:
            self.b = 999
        else:
            self.b = avg_win / avg_loss
        q = 1 - self.win_rate
        self.f_kelly = (self.b * self.win_rate - q) / self.b
        self.f_used = self.f_kelly * self.kelly_scale
        if self.f_kelly <= 0:
            self.f_used = 0

    def calc_lots(self, balance):
        if self.f_used <= 0:
            return 0
        risk_by_kelly = balance * self.f_used
        risk_cap = balance * self.single_risk_cap_pct
        risk_per_trade = min(risk_by_kelly, risk_cap)

        contract_multiplier = self.cta_engine.get_multiplier(self.vt_symbol)
        risk_per_lot = self.atr_value * contract_multiplier
        if risk_per_lot < 1e-6:
            return 0
        lots = int(risk_per_trade / risk_per_lot)
        lots = max(0, lots)
        lots = min(lots, self.max_pos_per_symbol)
        return lots

    def on_order(self, order):
        pass

    def on_trade(self, trade):
        pnl = trade.pnl
        self.trade_records.append(pnl)
        self.daily_pnl += pnl

        if trade.direction == Direction.LONG:
            if trade.offset == Offset.OPEN:
                self.long_pos += trade.volume
            else:
                self.long_pos -= trade.volume
        else:
            if trade.offset == Offset.OPEN:
                self.short_pos += trade.volume
            else:
                self.short_pos -= trade.volume

        if self.long_pos == 0 and self.short_pos == 0:
            self.in_trade = False
            g_risk_mgr.set_symbol_risk(self.vt_symbol, 0.0)
        self.put_event()

    def on_stop_order(self, so: StopOrder):
        pass
