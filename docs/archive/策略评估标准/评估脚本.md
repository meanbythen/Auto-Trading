下面是升级后的完整参数扫描脚本，新增：
1. **卡尔玛比率 Calmar**
2. **R期望（单位风险期望：胜率×盈亏比 - 败率）**
3. **盈利覆盖 = 净盈亏 - 最大回撤金额**
4. **拆分趋势组 / 震荡组绩效（趋势、震荡各自：交易笔数、胜率、盈亏比、盈利因子、净盈亏）**
5. **内置自动筛选规则**，输出两张csv：
   - `param_all_result.csv`：全部参数结果
   - `param_filtered_result.csv`：仅保留满足门槛的候选参数
6. 门槛就是前面约定的：
```
盈利因子 >= 1.3
R期望 >= 0.12
Calmar >= 1.0
最大回撤比例 <= 0.15
盈利覆盖 > 0
震荡组盈利因子 <= 1.0 （震荡尽量不赚钱）
```

> 重要前提：
> 你的策略 `LiquidityStopZoneStrategy` 内部绩效统计必须能区分每一笔交易 `is_trend`（之前代码里已经写了这个标记），否则无法拆分趋势/震荡。
> 回测依然是 `use_tq_vp=False`（只用聚类版本）。

```python
import csv
import numpy as np
from vnpy.cta.backtester import CtaBacktester
from vnpy.trader.constant import Interval
from your_strategy_file import LiquidityStopZoneStrategy  # 修改为你的策略文件名

# ====================== 配置区 ======================
symbol = "SHFE.fu2605"
interval = Interval.MINUTE
start = "2026-01-01"
end = "2026-06-01"
capital = 30000
base_slippage = 0.5
rate = 0.00005

# 扫描参数网格，按需增删
param_grid = [
    {"cluster_tol":0.002, "atr_entry_multiple":0.3, "atr_stop_multiple":1.0, "adx_threshold":20, "risk_reward":2.0},
    {"cluster_tol":0.004, "atr_entry_multiple":0.3, "atr_stop_multiple":1.0, "adx_threshold":20, "risk_reward":2.0},
    {"cluster_tol":0.006, "atr_entry_multiple":0.5, "atr_stop_multiple":1.0, "adx_threshold":20, "risk_reward":2.0},
    {"cluster_tol":0.006, "atr_entry_multiple":0.5, "atr_stop_multiple":1.25, "adx_threshold":20, "risk_reward":2.0},
    {"cluster_tol":0.006, "atr_entry_multiple":0.5, "atr_stop_multiple":1.0, "adx_threshold":22, "risk_reward":2.0},
    {"cluster_tol":0.006, "atr_entry_multiple":0.5, "atr_stop_multiple":1.0, "adx_threshold":20, "risk_reward":2.5},
]

# ===================== 筛选门槛（硬规则） =====================
FILTER_RULE = {
    "profit_factor_min": 1.3,
    "r_expect_min": 0.12,
    "calmar_min": 1.0,
    "max_dd_pct_max": 0.15,
    "profit_cover_min": 0,
    "side_profit_factor_max": 1.0
}

# ===================== 工具函数 =====================
def calc_trade_stats(trade_list):
    """
    输入交易记录列表，返回统计字典
    trade_list: [{"pnl":..., "is_trend":...}, ...]
    """
    if not trade_list:
        return {
            "count":0, "win_count":0, "lose_count":0,
            "win_rate":0, "avg_win":0, "avg_loss":0,
            "profit_loss_ratio":0, "profit_factor":0, "net_pnl":0
        }
    arr = [t["pnl"] for t in trade_list]
    win_trades = [p for p in arr if p>0]
    lose_trades = [p for p in arr if p<=0]
    total_win = sum(win_trades)
    total_lose = abs(sum(lose_trades))
    count = len(arr)
    win_count = len(win_trades)
    lose_count = len(lose_trades)
    win_rate = win_count / count if count>0 else 0
    avg_win = np.mean(win_trades) if win_trades else 0
    avg_loss = np.mean(lose_trades) if lose_trades else 0
    pl_ratio = avg_win / abs(avg_loss) if avg_loss !=0 else np.inf
    pf = total_win / total_lose if total_lose>0 else np.inf
    net_pnl = sum(arr)
    return {
        "count":count,
        "win_count":win_count,
        "lose_count":lose_count,
        "win_rate":round(win_rate,4),
        "avg_win":round(avg_win,2),
        "avg_loss":round(avg_loss,2),
        "profit_loss_ratio":round(pl_ratio,3),
        "profit_factor":round(pf,3),
        "net_pnl":round(net_pnl,2)
    }

def calc_calmar(annual_return, max_dd_pct):
    """卡尔玛 = 年化收益 / 最大回撤比例；回撤为0返回inf"""
    if max_dd_pct <= 0:
        return np.inf
    return annual_return / max_dd_pct

def calc_r_expect(win_rate, pl_ratio):
    """R期望 = 胜率*盈亏比 - 败率"""
    return win_rate * pl_ratio - (1 - win_rate)

# ===================== 主循环 =====================
all_result_rows = []
filtered_rows = []

for p in param_grid:
    setting = {
        "atr_period":20,
        "adx_period":14,
        "adx_threshold":p["adx_threshold"],
        "cluster_lookback":20,
        "cluster_tol":p["cluster_tol"],
        "atr_stop_multiple":p["atr_stop_multiple"],
        "atr_entry_multiple":p["atr_entry_multiple"],
        "risk_reward":p["risk_reward"],
        "risk_per_trade":0.01,
        "intra_day_close_minute":20,
        "use_tq_vp": False,   # 回测固定关闭TqVP，只用聚类
        "price_step":1.0,
        "va_ratio":0.7,
        "min_tick_count":3000
    }

    bt = CtaBacktester(
        strategy_class=LiquidityStopZoneStrategy,
        vt_symbol=symbol,
        interval=interval,
        start=start,
        end=end,
        capital=capital,
        rate=rate,
        slippage=base_slippage,
        setting=setting
    )
    bt.run_backtest()
    stats = bt.show_result(display=False)

    # 取出策略内部交易记录（策略里 self.trade_records）
    trade_records = bt.strategy.trade_records
    trend_trades = [t for t in trade_records if t["is_trend"]]
    side_trades  = [t for t in trade_records if not t["is_trend"]]

    all_stats = calc_trade_stats(trade_records)
    trend_stats = calc_trade_stats(trend_trades)
    side_stats  = calc_trade_stats(side_trades)

    # 核心指标
    net_pnl = all_stats["net_pnl"]
    max_dd_pct = stats["max_drawdown"]
    max_dd_value = stats["max_drawdown"] * capital
    profit_cover = net_pnl - max_dd_value

    # 年化收益简易估算（按回测天数，适合你后续自己微调）
    from datetime import datetime
    days = (datetime.strptime(end, "%Y-%m-%d") - datetime.strptime(start, "%Y-%m-%d")).days
    if days <= 0:
        annual_return = 0
    else:
        annual_return = (net_pnl / capital) / days * 252

    calmar = calc_calmar(annual_return, max_dd_pct)
    r_expect = calc_r_expect(all_stats["win_rate"], all_stats["profit_loss_ratio"])

    row = {
        # 参数
        "cluster_tol": p["cluster_tol"],
        "atr_entry_multiple": p["atr_entry_multiple"],
        "atr_stop_multiple": p["atr_stop_multiple"],
        "adx_threshold": p["adx_threshold"],
        "risk_reward": p["risk_reward"],

        # 全局绩效
        "total_trade": all_stats["count"],
        "win_rate": all_stats["win_rate"],
        "profit_loss_ratio": all_stats["profit_loss_ratio"],
        "profit_factor": all_stats["profit_factor"],
        "r_expect": round(r_expect,4),
        "net_pnl": net_pnl,
        "max_dd_pct": round(max_dd_pct,4),
        "max_dd_value": round(max_dd_value,2),
        "profit_cover": round(profit_cover,2),
        "annual_return": round(annual_return,4),
        "calmar": round(calmar,3),

        # 趋势组
        "trend_count": trend_stats["count"],
        "trend_win_rate": trend_stats["win_rate"],
        "trend_pl_ratio": trend_stats["profit_loss_ratio"],
        "trend_pf": trend_stats["profit_factor"],
        "trend_net": trend_stats["net_pnl"],

        # 震荡组
        "side_count": side_stats["count"],
        "side_win_rate": side_stats["win_rate"],
        "side_pl_ratio": side_stats["profit_loss_ratio"],
        "side_pf": side_stats["profit_factor"],
        "side_net": side_stats["net_pnl"],
    }
    all_result_rows.append(row)

    # 自动筛选判断
    pass_flag = True
    if row["profit_factor"] < FILTER_RULE["profit_factor_min"]: pass_flag=False
    if row["r_expect"] < FILTER_RULE["r_expect_min"]: pass_flag=False
    if row["calmar"] < FILTER_RULE["calmar_min"]: pass_flag=False
    if row["max_dd_pct"] > FILTER_RULE["max_dd_pct_max"]: pass_flag=False
    if row["profit_cover"] < FILTER_RULE["profit_cover_min"]: pass_flag=False
    if row["side_pf"] > FILTER_RULE["side_profit_factor_max"]: pass_flag=False

    if pass_flag:
        filtered_rows.append(row)
        print(f"✅ 合格候选：{p} | R期望:{r_expect:.3f} PF:{row['profit_factor']} Calmar:{calmar:.2f}")
    else:
        print(f"❌ 淘汰：{p} | R期望:{r_expect:.3f} PF:{row['profit_factor']} Calmar:{calmar:.2f}")

# ===================== 输出csv =====================
def write_csv(filename, rows):
    if not rows:
        print(f"{filename} 无数据")
        return
    fields = list(rows[0].keys())
    with open(filename, "w", newline="", encoding="utf-8-sig") as f:
        writer = csv.DictWriter(f, fieldnames=fields)
        writer.writeheader()
        writer.writerows(rows)

write_csv("param_all_result.csv", all_result_rows)
write_csv("param_filtered_result.csv", filtered_rows)

print(f"\n扫描完成：全部{len(all_result_rows)}组，筛选合格{len(filtered_rows)}组")
```

## 使用说明
1. 依赖：你的策略 `LiquidityStopZoneStrategy` 里面必须保留 `self.trade_records`，每一笔交易都带上 `is_trend` 标记（之前给你的版本已经写好了）
2. 年化是简易估算：`(总收益率/回测天数)*252`，日内策略通用；如果你想换成交易日计数，可以自行替换
3. 筛选出来的 `param_filtered_result.csv` 里面的参数，**只是过了历史硬门槛，不等于直接实盘**，下一步还要：
   - 样本外单独回测（拿一段没参与扫描的行情）
   - 滑点敏感性测试（+1价位滑点再跑一遍）
   - 参数鲁棒性扰动测试（参数附近小幅改动看绩效是否平滑）
4. 如果你后续想再加一列【滑点+1后的盈利因子】，我可以再加一层嵌套循环，每个参数自动跑两套滑点对比。
